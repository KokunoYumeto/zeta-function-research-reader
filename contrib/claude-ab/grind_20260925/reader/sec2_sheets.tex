\section{Hurwitz sheets, monoids of integers and Euler products}\label{sec:sheets}

\subsection{Zero velocities on the sheets (the copy's Theorems A--F)}

Along the flow $t\mapsto\zeta(s,a)$, $a=1+t$, a simple zero $\rho$ of $\zeta(s,a)$ moves with velocity
\[
c(\rho)=\frac{\rho\,\zeta(\rho+1,a)}{\partial_s\zeta(\rho,a)},
\]
obtained by implicit differentiation from $\partial_a\zeta(s,a)=-s\zeta(s+1,a)$ (Theorem~\ref{thm:jet} below).
The copy studied the sums of these velocities.
Its results were checked in the audit (\note{02\_}, \note{03\_}); Theorem~E had a full referee pass (\note{06\_}); the proof of Theorem~C was completed by claude-ab along the copy's route, relative to Theorem~A (\note{20\_}); a line-by-line re-derivation of Theorem~A itself is still open (Section~\ref{sec:open}).

\begin{theorem}[Theorems A and A$_a$]\label{thm:A}
Let $a>0$, let $c_0$ exceed the real part of every zero of $\zeta(s,a)$, and let $h$ be entire with $\sup_{|\im z|\le c_0+2}(1+|\re z|)^A|h(z)|<\infty$ for every $A$; put $H(s)=h(-i(s-\tfrac12))$ and $\hat h_1(\xi)=\frac1{2\pi}\int(\tfrac12+iu)h(u)e^{-iu\xi}du$.
Put $F_a(s)=s\zeta(s+1,a)/\zeta(s,a)$, and let $\beta_r(a)$ be the coefficients of the sheet's Dirichlet series $a\zeta(s+1,a)/\zeta(s,a)=\sum_{r}\beta_r(a)r^{-s}$ over the multiplicative monoid generated by the numbers $1+n/a$ ($\beta_1=1$).
Then, along good heights $T_k\to\infty$ (where $1/\zeta(\sigma\pm iT_k,a)$ is polynomially bounded on the horizontal segments),
\[
\lim_k\sum_{|\gamma|<T_k}\Res_\rho(F_aH)=\frac1a\,\frac1{2\pi}\int(\tfrac12+iu)h(u)\,du+\frac1a\sum_{r>1}\beta_r(a)\,r^{-1/2}\,\hat h_1(\log r)+L_a(h),
\]
where $L_a(h)=-(2\pi i)^{-1}\int_{\re s=-b}F_aH\,ds$ ($\tfrac32<b<2$) is an explicit left-line Hurwitz term.
\end{theorem}
\status{Proved by the copy (Theorems A, A$_a$; \note{02\_} §4.1, \note{03\_} §2.2). Checked independently at $a=1$ (relative difference $1.8\cdot10^{-18}$). Novelty: none found in the copy's search; standard technique.}

From Theorem~\ref{thm:A} the copy derived the following (all proved by the copy; references in Appendix~\ref{app:catalogue}, items S2--S6).
\begin{itemize}
\item \emph{Smoothed velocity laws} (Corollaries B, B$_a$): for $10\Delta\le T_1<T_2$ the smoothed sums of $\re c$ and $\im c$ grow like $(T_2-T_1)/(4\pi a)$ and $(T_2^2-T_1^2)/(4\pi a)-(T_2-T_1)$, with an error that is explicit at $a=1$; for fixed smoothing width $\Delta$ the error of the horizontal law is of the same order $T$ as its main term.
\item \emph{Sharp sums deviate at order $T$} (Theorem C): under (H), all but finitely many zeros simple and on the line, the errors of the sharp laws are unbounded multiples of $T$, and the normalized mean square has $\limsup\ge0.0373$; part 3 is unconditional.
\item \emph{The exact smoothed spectrum on every sheet} (Theorem F): frequencies $\log r$ for $r$ in the monoid generated by the numbers $1+n/a$, with amplitudes $-\beta_r(a)r^{-1/2}e^{-\Delta^2\log^2r/2}/(2\pi a\log r)$.
\item \emph{The Eulerian characterization} (Theorem E): the velocity spectrum factors over a $\Q$-independent frequency basis if and only if $a\in\{1,\tfrac12\}$; there the primes are exactly the frequencies of maximal amplitude.
\item \emph{The velocity angle is the Gram offset} (Theorem D): assuming the zeros up to $\gamma_n$ are simple and on the line, $\arg c(\rho_n)=\pi/2-\arctan(1/2\gamma_n)+\pi(S_{3/2}(\gamma_n)-S_n)$ with $S_n=n-\tfrac32-\theta(\gamma_n)/\pi$ and $S_{3/2}(t)=\pi^{-1}\arg\zeta(\tfrac32+it)$.
\end{itemize}

\begin{corollary}[only two sheets are Eulerian]\label{cor:eulerian}
$\zeta(s,1+t)$ has an Euler product with local factors of Euler type $(1-P^{-s})^{-1}$ only for $t\in\{0,-\tfrac12\}$.
\end{corollary}
\status{Proved by claude-ab (a note of an earlier folder) and a consequence of Theorem~E, which is strictly stronger (\note{06\_}). Novelty: possibly known; two quick searches found no statement of this kind.}

At the rational points $a=1/q$ Theorem~E reduces to the factoriality criterion for arithmetic congruence monoids (Baginski--Chapman \cite{BaginskiChapman}, Theorem~3.4(2); \note{11\_} §2); the next subsection gives the corresponding statements for the monoids themselves.

\subsection{Monoids of integers and the formal logarithm}

Let $M$ be a multiplicative submonoid of the positive integers.
An \emph{atom} is an element $m>1$ that is not a product of two elements of $M\setminus\{1\}$; a \emph{factorization} of $n\in M$ is a finite multiset of atoms with product $n$; $M$ is \emph{free} if every element has exactly one factorization and \emph{half-factorial} if all factorizations of each element have the same length.
Write $D_M(s)=\sum_{m\in M}m^{-s}=1+E(s)$ and
\[
\log D_M=\sum_{k\ge1}\frac{(-1)^{k+1}}{k}E^k=\sum_{n\ge2}r_n\,n^{-s},\qquad r_n=\sum_{k\ge1}\frac{(-1)^{k+1}}{k}T_k(n),
\]
where $T_k(n)$ is the number of ordered $k$-tuples in $(M\setminus\{1\})^k$ with product $n$.

\begin{theorem}[the formal logarithm detects freeness]\label{thm:formallog}
Let $n\in M$.
\begin{enumerate}[label=(\alph*)]
\item If every element of $M$ that is $\le n$ has exactly one factorization, then $r_n=1/k$ when $n=u^k$ for an atom $u$, and $r_n=0$ otherwise.
\item If $n$ has $j\ge2$ factorizations and every smaller element of $M$ has exactly one, then $r_n=1-j+\varepsilon$, where $\varepsilon=\sum1/k$ over the factorizations of $n$ of the form $u^k$. This number is at most $-1/6$.
\end{enumerate}
Consequently $M$ is free if and only if $r_n\ge0$ for all $n$, and if $M$ is not free the first negative coefficient sits exactly at the least element with two factorizations.
\end{theorem}
\status{Proved here. First derived in the programme (\lbl{CEI} Theorem 3.1 for the criterion; \lbl{ETR1.5--1.7} for the bound $-1/6$); derived independently by claude-ab (\note{11\_} Lemma 11.2); provenance in \note{29\_} §1. Novelty: not located in the published literature (one search); elementary.}

\begin{proof}
\emph{Reduction to free monoids.} Take $k\ge2$ and a tuple counted by $T_k(n)$. Each entry is at most $n/2$, so under the hypothesis of (a) or (b) each entry has exactly one factorization. Concatenating them gives a factorization $\Phi$ of $n$; conversely a factorization $\Phi$ with an ordered splitting of the multiset $\Phi$ into $k$ nonempty sub-multisets gives a tuple, and an entry $m<n$ determines its sub-multiset. Hence $T_k(n)=\sum_\Phi S_k(\Phi)$ for $k\ge2$, where $S_k(\Phi)$ counts ordered splittings of $\Phi$ into $k$ nonempty parts.

\emph{The free value.} Give each atom $u$ occurring in $\Phi$ its own commuting variable $x_u$. Ordered splittings are counted by the expansion of $\log\prod_u(1-x_u)^{-1}=\sum_u\sum_kx_u^k/k$ (the variables are needed because distinct atoms need not be multiplicatively independent, as $4$ and $8$ show). So $\sum_{k\ge1}(-1)^{k+1}S_k(\Phi)/k=\varrho(\Phi)$, where $\varrho(\Phi)=1/k$ if $\Phi$ consists of $k$ copies of one atom and $\varrho(\Phi)=0$ otherwise, and $S_1(\Phi)=1$.

\emph{(a)} There is one $\Phi$ and $T_1(n)=1=S_1(\Phi)$, so $r_n=\varrho(\Phi)$.

\emph{(b)} Now $T_1(n)=1$ while $\sum_\Phi S_1(\Phi)=j$, so $r_n=1+\sum_\Phi(\varrho(\Phi)-1)=1-j+\sum_\Phi\varrho(\Phi)$.
Every factorization of $n$ has length at least $2$, so each pure power $u^k$ among them has $k\ge2$. Two pure powers with the same exponent coincide, since $k$-th roots of positive integers are unique; so the exponents of the $P$ pure powers are distinct and $\varepsilon\le\sum_{i=2}^{P+1}1/i$. If $P\le1$ then $r_n\le1-2+\tfrac12=-\tfrac12$. If $P\ge2$ then $j\ge P$ and $r_n\le1-P+\tfrac12+\tfrac13+\dots+\tfrac1{P+1}\le-\tfrac16$, with equality only for $j=P=2$ and exponents $\{2,3\}$.

\emph{Consequence.} The formal logarithm is supported on $M$. If $M$ is free, (a) applies to every $n$. Otherwise (a) applies below the least element $n_0$ with two factorizations and (b) at $n_0$.
\end{proof}

\begin{corollary}[the first negative coefficient above $-\tfrac12$]\label{cor:firstneg}
Let $M$ be not free, let $N$ be its least element with two factorizations, and suppose $r_N>-\tfrac12$. Then $N$ has exactly two factorizations $a^{j_1}=b^{j_2}$ with $a\ne b$ atoms and $\gcd(j_1,j_2)=1$; $N=c^{j_1j_2}$ with $a=c^{j_2}$, $b=c^{j_1}$; and $r_N=-1+1/j_1+1/j_2$.
Conversely every value $-1+1/u+1/v$ with $2\le u<v$ coprime occurs, at $N=c^{uv}$ in $\langle c^u,c^v\rangle$. So the possible first negative coefficients in $(-\tfrac12,0)$ are exactly
\[
\{-\tfrac12+1/v:\ v\ge3\ \text{odd}\}\ \cup\ \{-\tfrac5{12},\,-\tfrac7{15}\}.
\]
\end{corollary}
\status{Proved here (claude-ab, \note{29\_} Corollary 29.1; refines \lbl{ETR1} and Theorem~\ref{thm:formallog}). Checked exactly on 286 generator sets. Refereed (tenth pass). Novelty: not searched; elementary.}

\begin{proof}
By Theorem~\ref{thm:formallog}(b), $r_N=1-j+\varepsilon$ with $\varepsilon$ a sum of reciprocals of distinct integers $\ge2$, one for each pure-power factorization. If $j\ge3$ then $r_N\le1-j+\sum_{i=2}^{j+1}1/i\le-2+\tfrac{13}{12}<-\tfrac12$; if $j=2$ and at most one factorization is a pure power, $r_N\le-\tfrac12$. So $j=2$ and both factorizations are pure powers, $N=a^{j_1}=b^{j_2}$ with $j_1<j_2$.
If $g=\gcd(j_1,j_2)>1$, then $N'=a^{j_1/g}=b^{j_2/g}<N$ has two distinct factorizations (neither exponent quotient is $1$, since an atom is not a proper power of an atom), contradicting minimality. So $g=1$. Comparing prime exponents, $v_\ell(a)j_1=v_\ell(b)j_2$, so $j_2\mid v_\ell(a)$; with $c=\prod_\ell\ell^{v_\ell(a)/j_2}$ we get $a=c^{j_2}$, $b=c^{j_1}$ and $N=c^{j_1j_2}$, and $r_N=-1+1/j_1+1/j_2$.
Conversely, in $\langle c^u,c^v\rangle$ with $u<v$ coprime, $xu+yv=uv$ with $x,y\ge0$ forces $(x,y)\in\{(v,0),(0,u)\}$, and two representations of an exponent differ by a multiple of $(v,-u)$, so no smaller exponent has two; both generators are atoms since $u\nmid v$.
Finally $1/u+1/v>\tfrac12$ with coprime $2\le u<v$ leaves $u=2$ with $v$ odd, and $u=3$ with $v\in\{4,5\}$.
\end{proof}

\begin{example}
In $M=\{1\}\cup\{2^e:e\ge2\}$ the atoms are $4$ and $8$ and $64=4\cdot4\cdot4=8\cdot8$, so $r_{64}=-\tfrac16$. With $x=2^{-s}$, $D_M=(1-x+x^2)/(1-x)$, whose zeros lie on $\re s=0$: for general monoids the absence of zeros in $\re s>1$ does not imply freeness (\note{11\_} Example 11.5; the monoid appears first in the programme's \lbl{ETR1}).
\end{example}
\status{Proved here (a direct computation; claude-ab, \note{11\_} Example 11.5).}

\subsection{Congruence monoids and the even lattice sums}

Fix $q\ge1$ and a subgroup $H$ of $G=(\ZZ/q)^\times$, and let $M_H=\{n\ge1:\gcd(n,q)=1,\ n\bmod q\in H\}$. By orthogonality, $D_{M_H}(s)=[G:H]^{-1}\sum_{\chi|_H=1}L(s,\chi)$.

\begin{proposition}\label{prop:congruence}
The following are equivalent: (i) $M_H$ is free; (ii) $H=G$; (iii) all $r_n\ge0$; (iv) $D_{M_H}$ has no zeros in $\re s>1$.
Moreover $M_H$ is half-factorial if and only if $[G:H]\le2$.
\end{proposition}
\status{The equivalence (i)$\Leftrightarrow$(ii) and the half-factoriality criterion are the standard factorial and half-factorial criteria for Krull monoids, specialised to $M_H$ (class group $G/H$; \note{11\_} §1.4; for $H=\{1\}$ Baginski--Chapman \cite{BaginskiChapman}, Theorem 3.4, the Krull-monoid analogue of Carlitz's class-number-two theorem \cite{Carlitz}). The direct proofs here and the equivalences with (iii) and (iv) are claude-ab's (\note{11\_} Proposition 11.1, Lemma 11.3; the step (iv)$\Rightarrow$(ii) cites Saias--Weingartner \cite{SaiasWeingartner}). Refereed (third pass).}

\begin{proof}
(i)$\Leftrightarrow$(iii) is Theorem~\ref{thm:formallog}. (ii)$\Rightarrow$(i): $M_G$ is free on the primes not dividing $q$.
(i)$\Rightarrow$(ii): if $c\in G\setminus H$, choose primes $p\ne r$ in the class $c$ and $p'\ne r'$ in the class $c^{-1}$ (Dirichlet); then $pp',rr',pr',rp'$ are atoms of $M_H$ and $(pp')(rr')=(pr')(rp')$.
(iii)$\Rightarrow$(iv): if all $r_n\ge0$, then coefficientwise $\log D\le e^{\log D}-1=D-1$, so $\log D$ converges absolutely on $\re s>1$ and $D=e^{\log D}\ne0$ there.
(iv)$\Rightarrow$(ii): if $H\ne G$, at least two characters trivial on $H$ are induced by distinct primitive characters, so $D_{M_H}$ lies in no single class $E_{q,\psi}$ of Saias--Weingartner's decomposition, and their Theorem~4 gives zeros in $\re s>1$.

Half-factoriality. If $[G:H]=2$, the atoms are the primes in $H$ and the products of two primes in the other coset, so every factorization of $n$ has length $a+b/2$, where $a$ and $b$ count the prime factors of $n$ inside and outside $H$. If $[G:H]\ge3$, either $G/H$ has an element of order $N\ge3$, and with primes $p,p'$ in the classes $c,c^{-1}$ the element $(pp')^N=p^N\cdot p'^N$ has factorizations of lengths $N$ and $2$; or $G/H$ contains $c,d,cd\notin H$, and with primes $p,r,s$ in these classes $(prs)^2=p^2\cdot r^2\cdot s^2$ has factorizations of lengths $2$ and $3$.
\end{proof}

\begin{proposition}[the even sheets at $a=1/q$]\label{prop:evensheets}
For $q\ge3$, $Z_{1/q}(s)=\zeta(s,1/q)+\zeta(s,1-1/q)=q^s\sum_{n\equiv\pm1\ (q)}n^{-s}=q^sD_{M_{\{\pm1\}}}(s)$.
Hence the sheet $a=1/q$ is Eulerian (the monoid $\{n\equiv\pm1\bmod q\}$ is free) if and only if $\varphi(q)\le2$, that is $q\in\{3,4,6\}$, if and only if its formal logarithm is nonnegative, if and only if $Z_{1/q}$ has no zeros with $\re s>1$; and exactly the same $q$ make the one-sided monoid $\{n\equiv1\bmod q\}$ half-factorial.
\end{proposition}
\status{Proved here (claude-ab, \note{08\_} Lemma 2 and \note{11\_} Corollary 11.4). Refereed (first and third passes). Freeness $\iff\varphi(q)\le2$ is the Krull-monoid criterion (class group $(\ZZ/q)^\times/\{\pm1\}$); the analytic packaging as a test on the sheets is not in the Baginski--Chapman survey; not searched further.}

\begin{proof}
$\zeta(s,1/q)=q^s\sum_{n\ge0}(qn+1)^{-s}$ and $\zeta(s,1-1/q)=q^s\sum_{m\equiv-1}m^{-s}$; for $q\ge3$ the two classes are distinct. The rest is Proposition~\ref{prop:congruence} with $H=\{\pm1\}$ and $H=\{1\}$, since $\{\pm1\}=G$ exactly when $\varphi(q)\le2$.
\end{proof}

\subsection{Shifted zeta products and tensor Euler functions}

For finite exponent data, a finitely supported measure $\mu$ on $\C$ with positive integer masses, the programme's tensor Euler function of degree $d$ is $\mathcal{L}_d=\prod_p\det(1-p^{-s}W_p^{\otimes d})^{-1}=\prod_\sigma\zeta(s-\sigma)^{\mu^{*d}(\{\sigma\})}$ (\lbl{ETR4--ETR5}).

\begin{lemma}[shifted zeta functions are multiplicatively independent]\label{lem:shifted}
Let $\sigma_1,\dots,\sigma_\ell\in\C$ be distinct and $C_1,\dots,C_\ell\in\ZZ$. If $\prod_j\zeta(s-\sigma_j)^{C_j}\equiv1$, then all $C_j=0$.
\end{lemma}
\status{Proved in the programme (\lbl{ETR8}); re-derived here (\note{29\_} Lemma 29.2). Refereed (tenth pass). Novelty: elementary; not searched.}

\begin{proof}
For $\re s$ large the Euler logarithm $\sum_p\sum_r\frac1r\big(\sum_jC_jp^{r\sigma_j}\big)p^{-rs}$ converges absolutely and tends to $0$ as $s\to+\infty$; its exponential is $1$, so it vanishes identically. Each prime power has a unique base, so $\sum_jC_jp^{r\sigma_j}=0$ for all $p$ and $r\ge1$. Two support points with $p^{\sigma_i}=p^{\sigma_j}$ differ by $2\pi ia/\log p$ with $a\in\ZZ\setminus\{0\}$, and the same pair cannot collide at two primes (that would give $p^b=q^a$); so some prime $p$ has the $x_j=p^{\sigma_j}$ distinct. For that $p$ the equations with $r=1,\dots,\ell$ have determinant $\prod_jx_j\prod_{i<j}(x_j-x_i)\ne0$, so all $C_j=0$.
\end{proof}

\begin{lemma}[uniqueness of convolution roots]\label{lem:convroot}
Let $\mu,\nu$ be finitely supported complex measures on $\C$ whose total masses are positive reals. If $\mu^{*d}=\nu^{*d}$ for some $d\ge1$, then $\mu=\nu$.
\end{lemma}
\status{Proved in the programme (\lbl{ETR9}); re-derived here in this generality (\note{29\_} Lemma 29.3).}

\begin{proof}
Put $A(z)=\int e^{\omega z}d\mu$, $B(z)=\int e^{\omega z}d\nu$. Then $A^d=B^d$, and $A(0),B(0)>0$ have equal $d$-th powers, so $A(0)=B(0)$. Near $0$ the quotient $A/B$ is continuous with values in the $d$-th roots of unity and equals $1$ at $0$, so $A=B$ near $0$ and hence everywhere. Distinct exponentials are linearly independent, so $\mu=\nu$.
\end{proof}

\begin{corollary}[one tensor degree recovers the exponent measure]
For finite exponent data, a single $\mathcal{L}_d$ determines $\mu$.
\end{corollary}
\status{Proved in the programme (\lbl{ETR4--ETR9}); re-derived (\note{29\_} Lemma 29.3). The programme also proves, for conjugation-stable data, that the real pole of $\mathcal{L}_{2k}$ at $1+2kB$ has order $h_{2k}$ with $h_{2k}^{1/2k}$ increasing to the dimension of the maximal face (\lbl{ETR5--ETR7}); that positivity of even tensor powers is Deligne's Weil~II §1.5 mechanism, as the programme credits.}

\begin{proof}
By Lemma~\ref{lem:shifted}, $\mathcal{L}_d$ determines the measure $\mu^{*d}$; by Lemma~\ref{lem:convroot} it determines $\mu$.
\end{proof}

The global character of Lemma~\ref{lem:shifted} is necessary: no finite set of primes fixed in advance has Euler factors that determine every finite shifted-zeta product (Section~\ref{sec:negative}, item N-Eu).
