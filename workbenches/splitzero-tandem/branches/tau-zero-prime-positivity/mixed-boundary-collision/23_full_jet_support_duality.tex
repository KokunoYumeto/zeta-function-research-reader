\section{Full jet duality and all prime-compatible positive forms}


\subsection{The original coefficients and the complete local pairing}
Keep
\[
 g(s)=2\xi(s)=s(s-1)\pi^{-s/2}\Gamma(s/2)\zeta(s),\qquad
 \iota\rho=1-\bar\rho.
 \tag{FJ1}\label{fj:fj:objects}
\]
Let $Z$ be a nonempty finite set of distinct nontrivial zeros, invariant
under $\iota$. Every multiplicity $m_\rho$ is the actual multiplicity.
Write $t_\rho=s-\rho$ and
\[
 g(\rho+t)=u_\rho(t)t^{m_\rho},\quad
 u_\rho(t)=2\lambda_\rho e^{h_\rho(t)},\quad
 \lambda_\rho=\xi^{(m_\rho)}(\rho)/m_\rho!,\quad h_\rho(0)=0.
 \tag{FJ2}
\]
All equalities below use this coordinate and this unit. The finite algebra is
$A=A_Z=\prod_{\rho\in Z}\mathcal O_\rho/(g)$; its coordinate identification
with $\prod_\rho\C[t_\rho]/(t_\rho^{m_\rho})$ follows because $u_\rho$
is invertible, and does not replace $g$ in any residue.
These are the finite jets in the supplied calculation \cite{FJprime},
Sections 2--4, and the original tau coefficient model \cite{FJtau},
Sections 5--6. Connes and Consani's original author source
\cite{ConnesConsaniHochschild}, Proposition \texttt{laplacian}, describes the finite
spectral quotient by jets and the triangular multiplication action;
the arguments below give the pairing and support maps directly.

For $f\in A$, put
$f^\dagger_\rho(t)=\overline{f_{\iota\rho}(-\bar t)}$.
The functional equation and reality give $g^\dagger=g$ and
$m_{\iota\rho}=m_\rho$, hence this is a well-defined antilinear algebra
involution. Define
\[
 R(f,h)=\sum_{\rho\in Z}\Res_{t=0}
       \frac{f^\dagger_\rho(t)h_\rho(t)}{u_\rho(t)t^{m_\rho}}\,dt,
 \qquad (\Pi_p f)_\rho(t)=p^\rho e^{t\log p}f_\rho(t)\pmod {t^{m_\rho}}.
 \tag{FJ3}\label{fj:fj:raw}
\]
Here $p$ is any positive prime. Multiplication by $p^s$ is a linear
automorphism of $A$, not a unital algebra homomorphism $A\to A$.

\begin{theorem}[Exact coefficient matrix, flags, and duality]
\label{fj:thm:perfect}
Write $u_\rho(t)^{-1}=\sum_{r\ge0}v_{\rho,r}t^r$ and
$f_\rho=\sum_{j<m_\rho}f_{\rho,j}t^j$. Then
\[
 R(f,h)=\sum_{\rho\in Z}\ \sum_{j+k\le m_\rho-1}
 (-1)^jv_{\rho,m_\rho-1-j-k}\,
 \overline{f_{\iota\rho,j}}h_{\rho,k}.
 \tag{FJ4}\label{fj:fj:matrix}
\]
It is perfect and skew-Hermitian. The linear isomorphism
\[
 D_R:\overline A\longrightarrow A^*,\qquad
 D_R(\bar f)(h)=R(f,h)
 \tag{FJ5}\label{fj:fj:dual}
\]
obeys, without a change of the prime factor,
\[
 R(\Pi_p f,\Pi_p h)=pR(f,h),\qquad
 D_R\overline{\Pi_p}=p(\Pi_p^{-1})^*D_R.
 \tag{FJ6}\label{fj:fj:similitude}
\]
For $F^a_\rho=t_\rho^aA_\rho$, $0\le a\le m_\rho$, the annihilator
of $F^b_\rho$ in its reflected partner is $F^{m_\rho-b}_{\iota\rho}$.
The resulting perfect pairing of complementary associated quotients is
\[
 (F^a_{\iota\rho}/F^{a+1}_{\iota\rho})\times
 (F^{m_\rho-1-a}_\rho/F^{m_\rho-a}_\rho)\longrightarrow\C,
 \quad (ct^a,dt^{m_\rho-1-a})\longmapsto
       \frac{(-1)^a\bar c d}{2\lambda_\rho}.
 \tag{FJ7}\label{fj:fj:graded}
\]
On every quotient $F^a_\rho/F^{a+1}_\rho$, the prime eigenvalue is
exactly $p^\rho$, at every jet order $a$.
\end{theorem}
\begin{proof}
Taking the coefficient of $t^{-1}$ proves \eqref{fj:fj:matrix}.
Changing a representative by a multiple of $g$ adds a holomorphic
integrand, so the formula is well defined. Its antidiagonal coefficient
is $(-1)^jv_{\rho,0}=(-1)^j/(2\lambda_\rho)\ne0$.
Reversing the order of the columns makes this block triangular with
nonzero diagonal. Thus every local block pairs perfectly with its
reflected block and their direct sum is perfect.
For a meromorphic germ $q$, reflection and conjugation send the
coefficient of $t^{-1}$ to minus its conjugate. Applying this to
$q=f^\dagger h/g$ and summing the reflected points proves
$R(h,f)=-\overline{R(f,h)}$.
Furthermore $(p^s)^\dagger=p^{1-s}$, so multiplication of the two
factors inside the residue gives exactly $p$. Replacing $h$ by
$\Pi_p^{-1}h$ gives the second identity in \eqref{fj:fj:similitude}.
Products of elements of orders $a,b$ have order at least $a+b$.
Consequently $F^{m-b}$ annihilates $F^b$. If $f$ has lowest nonzero
order $a<m-b$, choose $h=u_\rho(t)t^{m-1-a}$ times a scalar at the
partner point. It belongs to $F^b$ and its residue against $f$ is
nonzero. This proves the reverse containment. At orders adding to
$m-1$ only $v_{\rho,0}$ occurs, giving \eqref{fj:fj:graded}.
Finally $e^{t\log p}=1+t\log p+\cdots$ proves the action on each quotient.
\end{proof}

The equality $\overline{p^{\iota\rho}}p^\rho=p$ is the eigenvalue
identity in \eqref{fj:fj:graded}. It pairs two reflected points; it does
not force either point individually to be fixed by $\iota$.
All higher jets remain in \eqref{fj:fj:matrix}, with the same eigenvalue
and their nonzero Jordan entries $p^\rho(\log p)^j/j!$.

\begin{theorem}[The Jacobian defines its exact quotient and image]
\label{fj:thm:jacobian}
Let $N=\prod_\rho(t_\rho)$, $E=A/N=\C^Z$, and
$\epsilon:A\to E$ be evaluation. Set $J_g=M_{g'}$ and
\[
 j_g:E\longrightarrow A,\qquad
 (a_\rho)\longmapsto
 (2m_\rho\lambda_\rho a_\rho t_\rho^{m_\rho-1})_\rho.
 \tag{FJ8}\label{fj:fj:jg}
\]
Then $J_g=j_g\epsilon$, $\ker J_g=N$, and
$\im J_g=\prod_\rho\C t_\rho^{m_\rho-1}$, including the whole factor
when $m_\rho=1$. All maps are prime equivariant. The Hermitian form
\[
 T(f,h)=R(f,J_gh)
       =\sum_\rho m_\rho\overline{f(\iota\rho)}h(\rho)
       =\Tr(M_{f^\dagger h}\mid A)
 \tag{FJ9}\label{fj:fj:trace}
\]
has radical exactly $N$, and $T(\Pi_pf,\Pi_ph)=pT(f,h)$.
\end{theorem}
\begin{proof}
Differentiating the unaltered local equation gives
$g'=mu(t)t^{m-1}+u'(t)t^m\equiv2m\lambda t^{m-1}\pmod {t^m}$.
This proves \eqref{fj:fj:jg}, its kernel, and its image. Multiplication
operators commute, and the prime acts by $p^\rho$ both on values and
on the displayed image, proving equivariance. Also
$g'/g=m/t+u'/u$; the second term is holomorphic. Its residue against
$f^\dagger h$ is zero, proving the sum in \eqref{fj:fj:trace} with every
multiplicity. Multiplication by a germ on the $m$-dimensional local
factor has its value on all $m$ diagonal entries, proving the trace
identity. The involution permutes equal multiplicities, proving
Hermitian symmetry. Test against the value vector supported at the
reflected point to detect any nonzero value of $f$; this proves the
radical assertion. The first identity of \eqref{fj:fj:similitude} and
commutation with $J_g$ prove the final assertion.
\end{proof}

\subsection{The actual tau chart and every retained support label}
Use the original spaces $V,\mathscr B$ of \cite{FJtau}, Section 2:
$V$ consists of even Schwartz functions $\phi$ on $\mathbb R$ with
$\phi(0)=\widehat\phi(0)=0$, with Fourier kernel $e^{-2\pi ix\xi}$;
$\mathscr B$ consists of smooth functions on $\mathbb R_{>0}$ for
which every $x^b(-x\partial_x)^kF$ is bounded, $b\in\mathbb Z$, $k\ge0$.
Set $\Theta\phi(x)=\sum_{n\ne0}\phi(nx)$,
$\mathcal MF(s)=\int_0^\infty F(x)x^s\,dx/x$, and
$J F(x)=x^{-1}F(1/x)$. The four-point chart has
$+<\eta<\sigma$ and $-<\eta<\sigma$ with upper-set topology.
The sheaf has $V,V,\mathscr B,0$ at these points and maps
$\Theta,J\Theta=\Theta\mathcal F$ to $\eta$.
Its cohomology complex and actual prime action are
\[
 C_\tau=[V\oplus V\xrightarrow{d}\mathscr B],\quad
 d(\phi,\psi)=\Theta(\phi-\widehat\psi),\quad
 U_p^0(\phi,\psi)=(U_p\phi,pU_{1/p}\psi),\quad
 U_p^1F(x)=F(x/p).
 \tag{FJ10}\label{fj:fj:tau}
\]
For each nonempty $B\subseteq\{+,-\}$ retain exactly its source
legs, the same $\mathscr B$ at $\eta$, and their original signs.
The empty label has zero complex. Transports insert zero legs and
are the identity at $\eta$. This is the complete original label
lattice $L=\mathcal P(\{+,-\})$, not a chosen branch.

\begin{theorem}[Surjection and dual injection over the whole tau lattice]
\label{fj:thm:tau}
Every nonempty degree-one fibre is $Q=\mathscr B/\Theta V$.
There is a surjective morphism of support diagrams
\[
 q_Z:\{Q_B\}_{B\in L}\longrightarrow\{A_B\}_{B\in L},\quad
 q_Z[F]_\rho=\sum_{j<m_\rho}
       \frac{(\mathcal MF)^{(j)}(\rho)}{j!}t_\rho^j,
 \tag{FJ11}\label{fj:fj:jet}
\]
where $A_B=A$ for $B\ne\varnothing$, $A_\varnothing=0$,
and every nonempty transport is the identity.
It intertwines \eqref{fj:fj:tau} with $\Pi_p$ at every label.
On opposite diagrams, precomposition and \eqref{fj:fj:dual} give an injection
\[
 q_Z^*D_R:\{\overline A_B\}_{L^{\rm op}}\hookrightarrow
        \{\Hom(Q_B,\C(1))\}_{L^{\rm op}},\quad
 \bar f\longmapsto([F]\mapsto R(f,q_Z[F])).
 \tag{FJ12}\label{fj:fj:dual-tau}
\]
Here $\C(1)$ is the actual moment line with prime action $p$, and
the dual action is $\ell\mapsto p\ell U_p^{-1}$.
For the chart's coefficient diagram $\mathcal T$ and right adjoint
$K_\tau(\C(1))$, the target fibre is exactly
\[
 H^{-1}R\Hom(\mathcal T,K_\tau(\C(1)))=\Hom(Q,\C(1)).
\]
The contracted map $\bar f\mapsto([F]\mapsto R(f,J_gq_Z[F]))$
has kernel $\overline N$ in every nonempty source fibre. Labels of
these zero functionals remain distinct.
\end{theorem}
\begin{proof}
Fourier inversion on even Schwartz functions makes $\mathcal F:V\to V$
an automorphism, so every nonempty leg has image $\Theta V$.
The change of variable $x=py$ gives $\mathcal MU_p=p^s\mathcal M$.
The original Fourier convention gives
$\widehat{U_p\phi}=pU_{1/p}\widehat\phi$, proving $dU_p^0=U_p^1d$.
Every one of these operators is invertible on its retained leg.
For $\Re s>1$, absolute summation gives
$\mathcal M\Theta\phi(s)=2\zeta(s)\int_0^\infty\phi(x)x^{s-1}\,dx$.
The integral is holomorphic for $\Re s>-2$ because evenness and
$\phi(0)=0$ give $\phi(x)=O(x^2)$ at zero. Near any nontrivial
zero the other factors of $g$ are nonzero and holomorphic. Analytic
continuation therefore shows that $\mathcal M\Theta\phi$ is divisible
there by the actual $g$. Thus \eqref{fj:fj:jet} kills the relations.

To prove surjectivity already on compact smooth tests, use $y=\log x$.
Any functional annihilating all their jets would give a finite sum
$\sum_\rho e^{\rho y}P_\rho(y)=0$ on an arbitrary open interval,
where $\deg P_\rho<m_\rho$. To see this, the integral of that sum
against every compact smooth test on the interval is zero; continuity
forces the sum itself to vanish. Analyticity extends the equality to
all $y$. For a fixed $\rho$, apply
$\prod_{\eta\ne\rho}(\partial_y-\eta)^{m_\eta}$.
All other terms vanish, while its operator on $P_\rho$ is
$\prod_{\eta\ne\rho}(\partial_y+\rho-\eta)^{m_\eta}$.
It is invertible on polynomials of degree $<m_\rho$, being triangular
with nonzero constant diagonal. Hence $P_\rho=0$ for every $\rho$.
The jet map has zero annihilator in a finite-dimensional target, so
it is onto. Differentiating $p^s\mathcal MF$ proves its equivariance.

All nonempty support transports are identities on these quotients,
which proves the diagram assertion. Precomposition reverses the arrows.
Surjectivity and the perfect pairing prove the injection; FJ6 proves
its equivariance. For completeness, the projective resolution of the
constant chart diagram is
$0\to P_\eta\to P_+\oplus P_-\to\C\to0$, with maps
$v\mapsto(v,-v)$ and $(u,v)\mapsto u+v$.
It is exact at each of the four points. Applying $\Hom$ gives
\eqref{fj:fj:tau}. Its dual injective complex is
$[I_\eta\C(1)\to I_+\C(1)\oplus I_-\C(1)]$ in degrees $-1,0$,
with maps $(+\mathrm{res},-\mathrm{res})$.
Hom into this complex is the dual of \eqref{fj:fj:tau}; its degree $-1$
kernel is precisely the functionals annihilating $\im d$, namely
$\Hom(Q,\C(1))$. Finally FJ8--9 identify the contracted kernel.
\end{proof}

Reconstruction gives $\mathcal Q=\bigsqcup_{B\in L}Q_B$ and
$\mathcal A=\bigsqcup_{B\in L}A_B$, retaining each label even at zero.
The actual full base
\[
 G_L(\C)=\{(r,1_L):r\in\C\}\cup\{z_B=(0,B):B\in L\}
 \tag{FJ13}
\]
acts by $(r,1_L)(B,v)=(B,rv)$ and
$z_C(B,v)=(B\cap C,0)$. Addition transports to the union label
and adds there. The maps $q_Z,J_g,\epsilon,j_g$ preserve these actions:
each is linear within a fibre and carries its zero to the zero with
the same label. Thus they are maps over the entire lattice base.
Dual reconstruction uses $L^{\rm op}$ and precomposition. Its least
element is the original full-support label and has a nonzero dual
fibre; replacing it by zero would destroy FJ12.

\subsection{All finite support cochains and their exact dual receiver}
Let $S$ be any nonempty finite $T_0$ support space, ordered so that
opens are lower sets, as in \cite{FJsupport}, formulas
(FL14)--(FL16). Its full constant-coefficient complex is
\[
 C^n(S,A)=\prod_{s_0<\cdots<s_n}A,\qquad
 (dc)_{s_0<\cdots<s_{n+1}}
   =\sum_{a=0}^{n+1}(-1)^ac_{s_0<\cdots<\widehat{s_a}<\cdots<s_{n+1}}.
 \tag{FJ14}\label{fj:fj:support}
\]
There is no degree truncation. Empty products in degrees with no
chains mean the zero vector space. Define $C_n(S,A)$ by the same
finite coordinate sets and the boundary matrix which is the transpose
of this displayed differential. The scalar incidence coefficients
are the original integers $0,1,-1$.

\begin{theorem}[Perfect support cohomology--homology duality]
\label{fj:thm:support}
The prime acts on every coefficient in FJ14 by $\Pi_p$ and commutes
with every differential and boundary. The pairing
\[
 \mathcal R_n(a,c)=\sum_{s_0<\cdots<s_n}R(a_{s_0<\cdots<s_n},
                                           c_{s_0<\cdots<s_n})
 \tag{FJ15}\label{fj:fj:chain-pair}
\]
descends to a perfect pairing
$H_n(S,A)\times H^n(S,A)\to\C$ and obeys the exact prime
similitude $\mathcal R_n(\Pi_pa,\Pi_pc)=p\mathcal R_n(a,c)$.
There are canonical isomorphisms
\[
 H^n(S,A)=H^n(S,\C)\otimes_\C A,\qquad
 H_n(S,A)=H_n(S,\C)\otimes_\C A.
 \tag{FJ16}\label{fj:fj:kunneth}
\]
The Jacobian acts as $1\otimes J_g$. Its kernel and image on either
side are, respectively, $H^{n}(S,\C)\otimes N$ and
$H^{n}(S,\C)\otimes\im J_g$, with $H_n$ in the homological formula.
Consequently the contraction of FJ15 by $J_g$ has exactly these
radicals and retains the factor $m_\rho$ on its value quotient.

For an order-preserving map $f:S'\to S$, use
\[
 (f^*c)_{t_0<\cdots<t_n}=
 \begin{cases}
 c_{f(t_0)<\cdots<f(t_n)},&\text{all image points distinct},\\
 0,&\text{otherwise},
 \end{cases}
 \tag{FJ17}
\]
and its transpose chain map $f_*$. They commute with all prime and
Jacobian maps, and
$\mathcal R_n(f_*a,c)=\mathcal R_n(a,f^*c)$ on chains and homology.
\end{theorem}
\begin{proof}
Expanding $d^2$ pairs every two deleted vertices in its two possible
orders with opposite signs; thus it is zero. The chain assertion is
its transpose. Scalar incidence matrices commute with coefficient
multiplication. Transposition and the real signs give
$\mathcal R_n(a,db)=\mathcal R_{n-1}(\partial a,b)$.
It follows that cycles pair trivially with boundaries, on either side.
Each coordinate pairing is perfect by Theorem~\ref{fj:thm:perfect}.
In a finite-dimensional perfect pairing, the annihilator of a kernel
is the image of the adjoint matrix: containment follows by evaluation,
and equality follows because both dimensions equal the rank of that
matrix. Apply this once to each of the two adjacent differentials.
The induced pairing on the two homology quotients is perfect.

Every chain module in FJ14 is its scalar chain module tensored with
$A$. Choosing any vector-space complements to its kernels and images
proves that tensoring with $A$ preserves both kernels and images.
The resulting isomorphisms FJ16 are canonical because their maps send
$[c]\otimes a$ to $[c\otimes a]$ independently of those complements.
Apply the same exactness to FJ8 and use perfectness of the homology
pairing; this proves all kernel, image, and radical assertions.
Coefficientwise FJ6 gives the prime factor $p$.

For FJ17, if there is an adjacent repetition in the image chain,
the two terms deleting its equal members cancel and every other term
still has a repetition. Without a repetition the ordinary face
formula gives $df^*=f^*d$. Transposition gives the chain identity.
The defining finite sums give the adjointness, and coefficientwise
operators commute with these scalar matrices. All identities therefore
descend to the stated groups.
\end{proof}

\begin{corollary}[The exact augmented support quotient]
\label{fj:cor:augmented}
Put $K^0=A$, $K^n=C^{n-1}(S,A)$ for $n\ge1$, with
$d^0a=(a)_{s\in S}$ and $d^n=d_C^{n-1}$. If $S$ has $c$ connected
components, then
\[
 H^0(K)=0,\quad H^1(K)=A^c/\Delta A,
 \quad H^n(K)=H^{n-1}(S,\C)\otimes A\quad(n\ge2).
 \tag{FJ18}\label{fj:fj:augmented}
\]
The first nonzero group is paired perfectly with
$\{(a_1,\ldots,a_c)\in A^c:\sum_i a_i=0\}$ by
$\sum_iR(a_i,b_i)$. This pairing, its Jacobian contraction, every
support pullback, and the prime similitude follow from FJ15 without
discarding a support component.
\end{corollary}
\begin{proof}
The diagonal is injective because $S\ne\varnothing$.
Degree-zero cocycles of the support complex are exactly functions
constant on each connected component, proving FJ18. Changing a
representative $b$ by $\Delta h$ changes the pairing by
$R(\sum_i a_i,h)=0$. Conversely this annihilator of $\Delta A$
is exactly the stated kernel by perfectness of $R$. The quotient
pairing is therefore perfect. The remaining assertions are the
cochain identities already proved, including the augmentation.
\end{proof}

For any declared support-trajectory lattice of the incidence matrices,
restrict FJ14 to its upward-closed coordinate trajectories. Extension
by zero gives the original cochain maps: closure ensures that no
nonzero matrix entry leaves the next support. Tensoring each such
complex with $A$, as in FJ16, preserves its entire cohomology diagram.
Its dual diagram reverses arrows by matrix transpose. In particular
an inclusion on cochains need not be injective on cohomology; its
dual map is the transpose of the actual induced map, with the full
kernel preserved. This is the precise receiver for the full support
construction of \cite{FJsupport}, not a replacement by Boolean stalks.

\subsection{The spaces defined by the positivity and descent defects}
\begin{theorem}[Every positive semidefinite prime-similitude form]
\label{fj:thm:psd}
The Hermitian positive semidefinite forms $B$ on the entire $A_Z$
satisfying $B(\Pi_pf,\Pi_ph)=pB(f,h)$ for both $p=2,3$ are exactly
\[
 B(f,h)=\sum_{\substack{\rho\in Z\\\Re\rho=1/2}}
             b_\rho\overline{f(\rho)}h(\rho),\qquad b_\rho\ge0.
 \tag{FJ19}\label{fj:fj:psd}
\]
Thus their common forced radical is the prime-invariant ideal
\[
 \mathfrak D_Z=
 \left(\prod_{\Re\rho\ne1/2}A_\rho\right)
 \oplus\left(\prod_{\Re\rho=1/2}(t_\rho)\right),
 \qquad A_Z/\mathfrak D_Z=\C^{Z\cap(1/2+i\mathbb R)}.
 \tag{FJ20}\label{fj:fj:defect}
\]
The quotient map is a prime-equivariant algebra homomorphism and is
the universal quotient through which all forms FJ19 factor. A positive
definite prime-similitude form on $A_Z$ exists exactly when every point
of $Z$ lies on the critical line and every $m_\rho=1$.
\end{theorem}
\begin{proof}
Let $N_B$ be the radical of a positive semidefinite Hermitian form.
Cauchy--Schwarz follows by evaluating $B(v+zw,v+zw)\ge0$ for complex
$z$, and proves that $B(v,v)=0$ is equivalent to $v\in N_B$.
The two prime similitudes and invertibility of $\Pi_p$ make $N_B$
invariant under $\Pi_p$ and its inverse. On $A/N_B$ each
$p^{-1/2}\Pi_p$ is unitary. A finite-dimensional unitary operator is
diagonalizable: an eigenvector exists over $\C$, its orthogonal
complement is invariant under the operator and its inverse, and
induction produces an orthonormal eigenbasis. Its eigenvalues have
modulus one by its norm identity.

The local image of $A_\rho$ in this quotient is annihilated by
$(\Pi_2-2^\rho)^{m_\rho}$. If $\Re\rho\ne1/2$, the eigenvalue
$2^\rho$ has the wrong modulus, so the image is zero. On a critical
factor, diagonalizability forces $\Pi_2-2^\rho$ to annihilate the
image. But
$\Pi_2-2^\rho=2^\rho t_\rho a(t_\rho)$, with
$a(0)=\log2\ne0$, so its image ideal is exactly $(t_\rho)$.
Thus every nonconstant jet in a critical factor also maps to zero.

For remaining value vectors at $\rho,\eta$, equivariance gives
$(\overline{p^\rho}p^\eta-p)B(e_\rho,e_\eta)=0$ for $p=2,3$.
At critical points a nonzero cross term requires
$p^\rho=p^\eta$ for both primes. Then
$(\rho-\eta)\log2=2\pi ik$ and
$(\rho-\eta)\log3=2\pi il$. A zero integer forces $\rho=\eta$;
otherwise division gives a rational relation between $\log2$ and
$\log3$, contradicting unique factorization after exponentiation.
Therefore distinct critical values are orthogonal. Their diagonal
weights are nonnegative, proving FJ19. Conversely that formula gives
a positive semidefinite form and its prime similitudes directly.
Choosing every weight strictly positive proves that the intersection
of all their radicals is exactly FJ20. The final claim follows because
FJ20 is zero exactly in the stated case.
\end{proof}

\begin{corollary}[The full support multiplicity spaces]
For a finite-dimensional complex vector space $W$, all positive
semidefinite Hermitian forms on $W\otimes A$ compatible with
$1\otimes\Pi_2$ and $1\otimes\Pi_3$ are exactly
\[
 B(v,w)=\sum_{\Re\rho=1/2}B_\rho(v(\rho),w(\rho)),
\]
where each $B_\rho$ is an arbitrary positive semidefinite Hermitian
form on $W$. Their common forced radical is $W\otimes\mathfrak D_Z$.
This applies to each of $W=H^n(S,\C)$,
$W=H_n(S,\C)$ and $W=\C^c/\Delta\C$ in FJ16--18.
\end{corollary}
\begin{proof}
The quotient-by-radical argument of Theorem~\ref{fj:thm:psd} annihilates
each off-line factor and each $W\otimes(t_\rho)$ in a critical factor.
Distinct critical eigenvalue pairs are orthogonal by the same two-prime
calculation. Within a fixed eigenvalue pair the prime action is scalar
on $W$, so it imposes no further condition on the positive semidefinite
form on that space. Conversely any displayed family gives both prime
similitudes. Choosing a positive definite form on every nonzero $W$
proves the common-radical assertion.
\end{proof}

For the original $T$, each critical point contributes the weight
$m_\rho$. A reflected nonfixed pair contributes
$m\left(\begin{smallmatrix}0&1\\1&0\end{smallmatrix}\right)$ on
its values and has a negative vector $(1,-1)$. Thus the exact
similitude FJ6 and FJ9 exists before any positivity. FJ19 specifies
all possible positive replacements and their precise lost data; it
does not establish that the actual $T$ belongs to that set for all
zeros. In particular a finite positive restriction of the original
trace proves no universal positivity assertion.

There is also an exact obstruction to obtaining a scalar-valued
pairing from support labels alone. For a diagram $M_B$ on a finite
join lattice with top $1$, let $\mathcal M=\bigsqcup_BM_B$ with
addition by transport to joins. Its universal map to an additive
group is
\[
 \varepsilon_{\rm top}:\mathcal M\to M_1,
       \quad(B,v)\longmapsto h_{B1}v.
 \tag{FJ21}\label{fj:fj:top}
\]
Indeed every label zero $(B,0)$ is additively idempotent and must
map to zero in a group, while
$(B,v)+(1,0)=(1,h_{B1}v)$. This proves both factorization and its
uniqueness. Consequently any complex-valued pairing additive in each
entry factors through FJ21 in both entries. The exact object killed
in fibre $B$ is $\ker h_{B1}$; retaining its diagram is necessary
to preserve all support cohomology. The opposite dual diagram used
in FJ12 and FJ15 does preserve it and supplies the proved connecting
maps. At the still coarser coefficient-support observation of
\cite{FJdivisor}, vectors $(1,1)$ and $(1,-1)$ on a reflected pair
have identical support but opposite values $2m,-2m$ under $T$.
This computes a failure of factorization, while FJ11--18 give the
full coefficient and cohomological maps which do exist.

\subsection{The prime action on every declared jet support}
Fix an actual factor $A_\rho$ of multiplicity $m$ and its unaltered
coordinate basis $1,t_\rho,\ldots,t_\rho^{m-1}$. A declared support is
a subset $B\subseteq\{0,\ldots,m-1\}$. Its carrier element $(B,x)$
has $x\in\C^B$, and zero coordinates in $B$ remain supported zeros.
Outside $B$ there is external absence. Put
$\uparrow B=\{k:\text{some }j\in B\text{ has }j\le k\}$.

\begin{theorem}[Full prime support lift and its cancellation cells]
\label{fj:thm:prime-support}
The actual prime matrix has entries
\[
 (P_{p,\rho})_{kj}=
 \begin{cases}p^\rho(\log p)^{k-j}/(k-j)!,&k\ge j,\\0,&k<j.
 \end{cases}
 \tag{FJ22}\label{fj:fj:prime-matrix}
\]
Every entry in its lower triangle is nonzero. Its declared-support
lift to $G(\C)^m$ is the exact semimodule map
\[
 \widetilde\Pi_{p,\rho}(B,x)
   =(\uparrow B,\ (P_{p,\rho}x)|_{\uparrow B}).
 \tag{FJ23}\label{fj:fj:supported-prime}
\]
For positive integers $a,b>1$ these lifts satisfy
$\widetilde\Pi_a\widetilde\Pi_b=\widetilde\Pi_{ab}$.
For the action with parameter $1/p$, the composite with FJ23 is
$(B,x)\mapsto(\uparrow B,x^{\rm ext})$, where $x^{\rm ext}$ adds
supported zeros. It is the identity precisely on the subdiagram of
upward-closed supports, whose coefficient fibres are the full ideals
$F^a_\rho=t_\rho^aA_\rho$.

There is an exact linear isomorphism to the source's derivative stalk
\[
 s_\rho:A_\rho\longrightarrow E_\rho=t_\rho\C[t_\rho]/(t_\rho^{m+1}),
       \quad f\longmapsto t_\rho f,
 \tag{FJ24}
\]
intertwining multiplication by $p^\rho e^{t_\rho\log p}$ on both
spaces. Its composition with the original pointed map $\sigma_\rho$
records the actual nonzero coefficients, with index shifted by one.
For $m\ge2$ no map on these numerical support values alone
intertwines this prime action for all $f\in A_\rho$.

The exact replacement is the finite family of coefficient spaces
\[
 \mathcal C_{p;B,C}=\left\{x\in\C^m:
 \begin{array}{ll}
 x_j\ne0\ \Longleftrightarrow\ j\in B,\\
 \displaystyle p^\rho\sum_{j=0}^k
     \frac{(\log p)^{k-j}}{(k-j)!}x_j\ne0
       \ \Longleftrightarrow\ k\in C
 \end{array}\right\}.
 \tag{FJ25}\label{fj:fj:cells}
\]
They are disjoint locally closed algebraic subsets, some possibly
empty, which partition $A_\rho$. Multiplication by $P_{p,\rho}$
gives the explicit isomorphism
$\mathcal C_{p;B,C}\to\mathcal C_{1/p;C,B}$ with inverse
$P_{1/p,\rho}$. Thus the failed support descent determines a
specified family of coefficient spaces and exact maps.
\end{theorem}
\begin{proof}
Expanding the original exponential gives FJ22. Matrix multiplication
in $G(\C)$ gives support $\uparrow B$ even if an amplitude cancels;
this proves FJ23 including all its supported zeros. The exponential
identity gives $P_aP_b=P_{ab}$, with every prime factor and logarithm
unchanged. Since $\uparrow\uparrow B=\uparrow B$, the supports of
the two lifted sides also agree for $a,b>1$. The inverse-parameter
matrix is lower triangular with every lower entry nonzero as well.
Its ring composite is the identity matrix, but its declared composite
retains every lower triangular path. Thus its value on $(B,x)$ has
exactly the support and amplitudes stated after FJ23. Equality with
$(B,x)$ is precisely $B=\uparrow B$. Such subsets are the terminal
segments $\{a,\ldots,m-1\}$ and the empty set, with coefficient
spaces exactly $F^a_\rho$. Their restriction maps are inclusions;
FJ7 is the full reflected duality of this flag diagram.

The map $s_\rho$ is well defined and its inverse divides its polynomial
representative by $t_\rho$, modulo $t_\rho^m$. Commutation with
coefficient multiplication proves its equivariance. For failure of
the numerical support action, put $L=\log p$ and compare
$f=1+Lt_\rho$ and $h=1-Lt_\rho$, both having the same nonzero
coefficient indices $\{0,1\}$. The coefficient of $t_\rho$ in their
prime images is respectively $2p^\rho L\ne0$ and zero. This remains
valid for every $m\ge2$, regardless of later coefficients.
Therefore no function of the input numerical support can give the
output numerical support for both inputs.
Finally FJ25 consists of linear equations together with inequalities
that finitely many linear polynomials are nonzero, proving local
closedness. Every $x$ has one input and one output support, proving
the partition. Invertibility of $P_{p,\rho}$ and its explicit inverse
give both the asserted bijection and its regular inverse.
\end{proof}

The closure $B\mapsto\uparrow B$ preserves joins and bottom; it
does not preserve every meet. For $m\ge2$, $B=\{0\}$ and
$C=\{1\}$ have empty intersection, whereas their upper closures
intersect. The exact general comparison is
$\uparrow(B\cap C)\subseteq\uparrow B\cap\uparrow C$.
Consequently the unrestricted support lift is a map of the original
$G(\C)$ semimodules with this support transport; it is not silently
a scalar-base automorphism of $G_{\mathcal P(\{0,\ldots,m-1\})}(\C)$.
On the complete terminal-flag diagram the support map is the identity,
the entire prime action is invertible, and every full jet is still
present in its largest coefficient fibre. The products of these
constructions over the actual $\rho\in Z$ retain every multiplicity,
reflection partner, and support coordinate.
