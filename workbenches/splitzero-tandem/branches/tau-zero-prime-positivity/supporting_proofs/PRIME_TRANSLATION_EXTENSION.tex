\subsection{The exact Mellin and source-space dictionary}
Retain the new source's constants and operators:
\begin{equation}
 r=1/64,\quad a_j=r2^{-j},\quad
 b_r=\mathop{*}_{j\ge1}\frac{\mathbf1_{[-a_j,a_j]}}{2a_j},\quad
 G_r(z)=\prod_{j\ge1}\frac{\sinh(a_jz)}{a_jz},\quad
 T=\partial_v^2-1/4.                                        \tag{PWG1}\label{PWG1}
\end{equation}
The entire product and smooth probability density are the classical construction of Arias de Reyna \cite{Arias}, with the exact scale used by UP. Their proof, reproduced and attributed in UP, gives
$\supp b_r\subset[-r,r]$, $\int b_r=1$, and zeros of $G_r$ only at
$128\pi i k$, of order $1+\nu_2(|k|)$ for nonzero integers $k$.
Let
\begin{equation}
 f_r=Tb_r,\quad k_r=b_r*b_r,\quad h_r=T^2k_r=f_r*f_r,
 \quad M_f(s)=\int_\R f(v)e^{-(s-1/2)v}\,dv.                 \tag{PWG2}\label{PWG2}
\end{equation}
The source proves
\begin{equation}
 M_{f_r}(s)=s(s-1)G_r(s-1/2),\qquad
 M_{f_r}(0)=M_{f_r}(1)=0,                                   \tag{PWG3}\label{PWG3}
\end{equation}
and this transform is nonzero at every possible actual off-critical zeta zero.

The map to the earlier multiplicative Mellin convention is explicitly
\begin{equation}
 (\mathcal Vf)(x)=x^{-1/2}f(-\log x),\qquad
 \int_0^\infty(\mathcal Vf)(x)x^s\frac{dx}{x}=M_f(s).
                                                               \tag{PWG4}\label{PWG4}
\end{equation}
Substitution $v=-\log x$ proves both the Mellin formula and
$\|\mathcal Vf\|_{L^2(dx)}=\|f\|_{L^2(dv)}$.
It also proves
\[
 \mathcal V(f^\#)(x)=x^{-1}\overline{(\mathcal Vf)(1/x)},\qquad
 \mathcal V(f*g)=(\mathcal Vf)*_\times(\mathcal Vg),
 \quad f^\#(v)=\bar f(-v).
\]
Translation $f_a(v)=f(v-a)$ becomes
$\mathcal V f_a(x)=e^{a/2}(\mathcal Vf)(e^ax)$. Thus a translation of length $\log p$ is the unitary dilation $k(x)\mapsto\sqrt p\,k(px)$ on this stated source space. Moreover
\begin{equation}
 \mathcal V\partial_v\mathcal V^{-1}=-x\partial_x-\tfrac12.
                                                               \tag{PWG5}\label{PWG5}
\end{equation}
This identifies every factor, the physical measure, and the adjoint convention. It is not an identification of this space with a finite $G_N$ without its original source map.

The complete original translation pairing is
\begin{equation}
 K_r(a)=B((f_r)_a,f_r)
       =\sum_\rho w_\rho e^{c_\rho a},\quad
 c_\rho=\rho-\tfrac12,\quad
 w_\rho=m_\rho M_{f_r}(\rho)^2.                              \tag{PWG6}\label{PWG6}
\end{equation}
The weights are complex squares off the line. UP18 proves
$\sum_\rho|w_\rho|(1+|\Im\rho|)^j<\infty$ for every $j$, and
$|\Re c_\rho|\le1/2$. The source's unsmoothed $K_r$ is smooth, real and even; a holomorphic extension was not asserted there. We do not add one without a convergence proof.

\subsection{Gaussian translation heat preserves the complete positivity question}
For $\epsilon>0$ put
\begin{equation}
 \gamma_\epsilon(a)=\frac1{\sqrt{4\pi\epsilon}}e^{-a^2/(4\epsilon)},
 \quad f_{r,\epsilon}=\gamma_{\epsilon/2}*f_r,
 \quad K_{r,\epsilon}=\gamma_\epsilon*K_r.                  \tag{PWG7}\label{PWG7}
\end{equation}
The factor $\epsilon/2$ in the test is essential: smoothing both slots adds their variances. Gaussian convolution is an injective contraction on $L^2(dv)$, with Fourier multiplier $e^{-\epsilon\xi^2}$ for time $\epsilon$. Its inverse on its range need not be bounded on $L^2$. Under \eqref{PWG5} its generator is $(x\partial_x+1/2)^2$, not the original operator $(1/4)\partial_s^2$ moving $g_t$.

\begin{theorem}[Entire translation heat and a fixed-time RH criterion]
For every fixed $\epsilon>0$,
\begin{align}
 M_{f_{r,\epsilon}}(s)&=e^{\epsilon(s-1/2)^2/2}M_{f_r}(s),\notag\\
 \boxed{K_{r,\epsilon}(a)=
 B((f_{r,\epsilon})_a,f_{r,\epsilon})
 =\sum_\rho w_\rho e^{\epsilon c_\rho^2+c_\rho a}.}          \tag{PWG8}\label{PWG8}
\end{align}
The last function is entire in $a$, is real and even on the real axis, and satisfies
\begin{equation}
 |K_{r,\epsilon}(a)|\le C_r
 \exp\left(\frac\epsilon4+\frac{|\Re a|}{2}
                         +\frac{(\Im a)^2}{4\epsilon}\right),
 \quad C_r=\sum_\rho|w_\rho|<\infty.                        \tag{PWG9}\label{PWG9}
\end{equation}
It solves $\partial_\epsilon K_{r,\epsilon}=\partial_a^2K_{r,\epsilon}$, with the original real-time correlation as its initial Gaussian-convolution limit. Most importantly,
\begin{equation}
 \boxed{\mathrm{RH}\ \Longleftrightarrow\
 K_{r,\epsilon}\text{ is bounded on }[0,\infty)
 \ \Longleftrightarrow\ |K_{r,\epsilon}(a)|\le K_{r,\epsilon}(0)
                    \text{ for all }a\ge0.}                \tag{PWG10}\label{PWG10}
\end{equation}
These equivalences hold at every positive translation-heat time, not only in a limit as $\epsilon\downarrow0$.
\end{theorem}
\begin{proof}
All Gaussian Mellin integrals converge absolutely, and completing the square gives the multiplier $e^{\epsilon(s-1/2)^2/2}$. The two endpoint values remain exactly zero. Convolution of the two smoothed slots gives $\gamma_\epsilon*h_r$. The full zero series in \eqref{PWG8} is absolutely convergent. If $a=A+iB$ and $c_\rho=d+i\gamma$, its exponential modulus is at most
\[
 \exp(\epsilon/4+|A|/2-\epsilon\gamma^2+|B|\,|\gamma|)
 \le\exp(\epsilon/4+|A|/2+B^2/(4\epsilon)).
\]
This proves \eqref{PWG9}, normal convergence on every complex compact set, and termwise differentiation. It also permits the original Gaussian integral in $a$ to be interchanged with the full zero series, giving exactly the multiplier $e^{\epsilon c_\rho^2}$. Reflection and conjugation of the actual zero multiset prove the real-even assertion.

Every off-critical coefficient remains nonzero, because both $M_{f_r}(\rho)$ and the Gaussian multiplier are nonzero. If $K_{r,\epsilon}$ is bounded on the positive half-line, its Laplace transform is holomorphic on $\Re w>0$. On $\Re w>1/2$ it equals
\[
 \sum_\rho\frac{w_\rho e^{\epsilon c_\rho^2}}{w-c_\rho}.
\]
The sum is meromorphic on the whole plane, with each displayed nonzero off-critical residue. The normal-convergence proof is the same absolute-tail estimate just given, or the stronger Gaussian decay at high ordinates. A right-hand off-critical zero would give a nonremovable pole in the holomorphic Laplace half-plane, a contradiction by the identity theorem; reflection then excludes left-hand off-critical zeros. Conversely, under RH every $c_\rho=i\gamma_\rho$ and $M_{f_r}(\rho)$ is real. All coefficients become $m_\rho M_{f_r}(\rho)^2e^{-\epsilon\gamma_\rho^2}\ge0$, so the exact triangle inequality proves the last bound. It implies boundedness directly. This proves the full equivalence, using the UP nonvanishing and whole-zero-set mechanism with the new Gaussian unit retained.
\end{proof}

This is not the Newman heat flow. Translation heat moves the test and its prime window while fixing every $\rho$; Newman heat moves the zeros of $g_t$. The original all-zero obstruction survives translation heat at every positive time. No assertion that it must survive the different Newman evolution is made.

\subsection{All translations become a single-point coefficient tower}
For $j,k\ge0$ define
\begin{equation}
 (\mathcal C_\epsilon)_{jk}
   =\frac{(-1)^j}{j!k!}K_{r,\epsilon}^{(j+k)}(0).
                                                               \tag{PWG11}\label{PWG11}
\end{equation}
These are real Hermitian coefficients: the odd derivatives vanish, and for an even total degree $(-1)^j=(-1)^k$. There is no numerical positivity assumption on them.

\begin{theorem}[Complete one-point positivity criterion after translation heat]
For any one fixed $\epsilon>0$,
\begin{equation}
 \boxed{\mathrm{RH}\ \Longleftrightarrow\
 [ (\mathcal C_\epsilon)_{jk}]_{0\le j,k\le N}\succeq0
 \quad\text{for every }N\ge0.}                            \tag{PWG12}\label{PWG12}
\end{equation}
The entries obey the exact two equivalent evolution laws
\begin{align}
 \partial_\epsilon(\mathcal C_\epsilon)_{jk}
 & =-(j+1)(k+1)(\mathcal C_\epsilon)_{j+1,k+1}\notag\\
 & =\tfrac12\left[(j+1)(j+2)(\mathcal C_\epsilon)_{j+2,k}
                  +(k+1)(k+2)(\mathcal C_\epsilon)_{j,k+2}\right].
                                                               \tag{PWG13}\label{PWG13}
\end{align}
A finite block does not close under these equations without its displayed higher rows.
\end{theorem}
\begin{proof}
Under RH, \eqref{PWG8} represents the coefficient matrix as the full Gram
\[
 (\mathcal C_\epsilon)_{jk}=
 \sum_\rho m_\rho M_{f_r}(\rho)^2e^{-\epsilon\gamma_\rho^2}
 \frac{(-i\gamma_\rho)^j(i\gamma_\rho)^k}{j!k!}.
\]
It converges absolutely and is positive. Conversely, the entire function proved in \eqref{PWG9} has the absolutely convergent two-variable Taylor expansion
\[
 K_{r,\epsilon}(b-a)
 =\sum_{j,k\ge0}(\mathcal C_\epsilon)_{jk}a^jb^k.
\]
For any finite real translation set and complex coefficient vector, each square truncation is a nonnegative quadratic form if all principal blocks in \eqref{PWG12} are positive. Absolute convergence permits passing to its full value. Thus $[K_{r,\epsilon}(a_j-a_i)]$ is positive for every finite translation set. Its two-point submatrix gives $|K_{r,\epsilon}(a)|\le K_{r,\epsilon}(0)$, and \eqref{PWG10} proves RH. Finally, differentiate \eqref{PWG11} using $\partial_\epsilon K=K''$ and compare factorials and signs. This gives both formulas in \eqref{PWG13}. No finite truncation has been substituted for the whole tower.
\end{proof}

The passage from a merely smooth UP correlation to this entire correlation is why the global Taylor argument is justified here. This is a new propagation of the just-published fixed-test criterion; it is not a proof that every block in \eqref{PWG12} is positive.


\subsection{The complete negative index is unchanged by translation heat}
The preceding criterion can be strengthened without supposing that off-critical zeros exist. Let $\kappa$ be the number of distinct two-point orbits of $\rho\mapsto1-\bar\rho$ in the actual zero set, with infinity allowed. The multiplicities remain in every Hilbert-space weight; one two-point orbit contributes one negative direction regardless of that common positive multiplicity.

\begin{theorem}[Exact index of the smoothed coefficient tower]
For every $\epsilon>0$,
\begin{equation}
 \boxed{\sup_{N\ge0}\operatorname{ind}_{-}
       [ (\mathcal C_\epsilon)_{jk}]_{0\le j,k\le N}=\kappa.}
                                                               \tag{PWG24}\label{PWG24}
\end{equation}
In particular the entire negative index is independent of the positive translation-heat time. This statement is about the full tower, not the eigenvalues of any one fixed block.
\end{theorem}
\begin{proof}
Let $\mathcal Z_*$ contain the actual zeros at which $M_{f_r}(\rho)\ne0$, and put
\[
 \mathcal H_* =\ell^2(\mathcal Z_*,m),\qquad
 (Jh)(\rho)=h(\rho^\#),\qquad
 a_\rho=M_{f_r}(\rho)e^{\epsilon c_\rho^2/2}.
\]
This set is reflection stable. Every off-critical zero belongs to it by UP15. Omitting any of its complementary critical zeros changes only positive directions, not $\kappa$. For a polynomial $P$, define $V_P(\rho)=a_\rho P(c_\rho)$. These vectors belong to $\mathcal H_*$ by the Gaussian and original rapid-decay bounds. The original reflection gives $\bar a_{\rho^\#}=a_\rho$ and $c_{\rho^\#}=-\bar c_\rho$. Thus the form $\langle JV_P,V_Q\rangle$ in the divided power basis $P(c)=c^j/j!$ is exactly \eqref{PWG11}.

The vectors $V_P$ are dense in $\mathcal H_*$. To prove it, suppose $h$ is orthogonal to all of them and form
\[
 F_h(z)=\sum_{\rho\in\mathcal Z_*}m_\rho\bar h(\rho)a_\rho e^{c_\rho z}.
\]
Cauchy--Schwarz and the Gaussian factor give normal convergence on every complex compact set: the second squared factor is bounded by a convergent sum of $m_\rho|a_\rho|^2e^{2|\Im\rho|\,|\Im z|+|\Re z|}$. All derivatives exist normally and are zero at $z=0$ by the assumed orthogonality. Hence $F_h$ is identically zero. For real $z\ge0$, its absolute value is bounded by
$e^{z/2}\|h\|\,(\sum m_\rho|a_\rho|^2)^{1/2}$.
Its Laplace transform on $\Re w>1/2$ can therefore be integrated termwise and gives
\[
 0=\sum_{\rho\in\mathcal Z_*}
       \frac{m_\rho\bar h(\rho)a_\rho}{w-c_\rho}.
\]
The series is normally convergent off the discrete exponent set: Cauchy--Schwarz bounds its tails, and away from bounded heights its denominators grow at least linearly with height. It is meromorphic, with residue $m_\rho\bar h(\rho)a_\rho$ at $c_\rho$. The complement of a discrete subset of the plane is connected, so the identity theorem and the nonzero $a_\rho$ force $h(\rho)=0$ for all $\rho$. This proves density without assuming a real spectrum.

The bounded self-adjoint involution $J$ has one negative eigenvector for each two-point reflection orbit and none for a fixed point. Every negative polynomial subspace injects into that negative spectral subspace, proving the upper bound in \eqref{PWG24}. Conversely approximate any finite orthonormal set of negative spectral vectors by $V_P$ using the proved density. For sufficiently close approximations, the complete Gram matrix remains negative definite, by the explicit perturbation bound used in \eqref{CW9}. Finitely many polynomials fit in one finite degree block. This proves the lower bound, including when $\kappa$ is infinite.
\end{proof}

The functions corresponding to these polynomial vectors are the actual derivatives $\partial_v^j f_{r,\epsilon}/j!$, since integration by parts in the retained minus-exponent Mellin convention multiplies the transform by $(s-1/2)^j/j!$. Their endpoint values still vanish. This connects the index theorem to an actual fixed source family, not merely abstract polynomial sequences. The new density proof extends the source's Cauchy-rational density mechanism to this Gaussian-weighted polynomial observation; it is not an assertion that unsmoothed polynomial moments would have the same convergence.

\subsection{Full prime and Archimedean expressions after smoothing}
Set $h_{r,\epsilon}=\gamma_\epsilon*h_r=T^2(\gamma_\epsilon*k_r)$. Gaussian convolution of the \emph{whole} PW12 formula gives
\begin{align}
 K_{r,\epsilon}(a)={}&-(\gamma_{\rm E}+\log\pi)h_{r,\epsilon}(a)\notag\\
 &+\int_0^\infty\frac{e^{-x}h_{r,\epsilon}(a)
   -\tfrac12e^{-x/4}\{h_{r,\epsilon}(a+x/2)+h_{r,\epsilon}(a-x/2)\}}
                       {1-e^{-x}}\,dx\notag\\
 &-\sum_{n\ge2}\frac{\Lambda(n)}{\sqrt n}
       \{h_{r,\epsilon}(a+\log n)+h_{r,\epsilon}(a-\log n)\}.
                                                               \tag{PWG14}\label{PWG14}
\end{align}
The constant is Euler's constant $\gamma_{\rm E}$, not a zero ordinate or the heat Gaussian. All derivative and translation factors in this expression are the original ones.

For justification, near $x=0$ the numerator can be rewritten as
\[
 (e^{-x}-e^{-x/4})h(a)
 -\tfrac12e^{-x/4}[h(a+x/2)+h(a-x/2)-2h(a)].
\]
The first term is $O(x)$ and the second $O(x^2)$ with the actual derivative bounds. At infinity the exponential factors are integrable. In the prime term, convolution of compact $h_r$ by a Gaussian gives Gaussian decay in $\log n$ at each fixed translation; the bound below proves its absolute sum and the Fubini interchange. Thus \eqref{PWG14} is an extension obtained from the original complete distribution by a justified limit, not a declaration that compact support survives smoothing. In particular the outer-window simplification $a>2r$ in PW19 cannot be copied after Gaussian convolution: \eqref{PWG14} is the required replacement.

Here is a finite error at every derivative order. Set
\begin{equation}
 B_m(\epsilon)=\frac{m!}{2^m}
 \sum_{j=0}^{\lfloor m/2\rfloor}
 \frac{\epsilon^{-(m-j)}}{j!(m-2j)!}
 \left(\frac{4\epsilon(m-2j)}{\mathrm e}\right)^{(m-2j)/2},
                                                               \tag{PWG15}\label{PWG15}
\end{equation}
where the factor with $m-2j=0$ is one. This convention is a value of a continuous nonnegative scalar maximum, not a change to supported zero. The finite derivative expansion of a Gaussian proves
\begin{equation}
 |\gamma_\epsilon^{(m)}(x)|\le
 (4\pi\epsilon)^{-1/2}B_m(\epsilon)e^{-x^2/(8\epsilon)}.
                                                               \tag{PWG16}\label{PWG16}
\end{equation}
Indeed its $j$th coefficient is
$m!(-1)^{m-j}x^{m-2j}/[2^m\epsilon^{m-j}j!(m-2j)!]$ times the Gaussian. Split its exponential in half and use the exact maximum
$\sup_x|x|^d e^{-x^2/(8\epsilon)}=(4\epsilon d/\mathrm e)^{d/2}$ for $d>0$.
Put
\begin{equation}
 C_{\epsilon,m}=(4\pi\epsilon)^{-1/2}
 \left[B_{m+4}(\epsilon)+\tfrac12B_{m+2}(\epsilon)
                                      +\tfrac1{16}B_m(\epsilon)\right].
                                                               \tag{PWG17}\label{PWG17}
\end{equation}
Since $k_r$ is a probability measure on $[-2r,2r]$, for $|a|\le A$ and $\log n>A+2r$,
\[
 |h_{r,\epsilon}^{(m)}(a\pm\log n)|
 \le C_{\epsilon,m}
        e^{-(\log n-A-2r)^2/(8\epsilon)}.
\]
For $B=A+2r$, $c_B=B+2\epsilon$, and any integer $N$ with
$\log N\ge\max(B,2)$, define
\begin{align}
 J_{\epsilon,B}(N)=e^{B/2+\epsilon/2}
 \bigg[&4\epsilon e^{-(\log N-c_B)^2/(8\epsilon)}\notag\\
 &+c_B\sqrt{2\pi\epsilon}\,
  \operatorname{erfc}\left(\frac{\log N-c_B}{\sqrt{8\epsilon}}\right)\bigg].
                                                               \tag{PWG18}\label{PWG18}
\end{align}
The complete prime-tail bound is
\begin{equation}
 \boxed{\sup_{|a|\le A}|\partial_a^m(P_\epsilon-P_{\epsilon,N})(a)|
       \le2C_{\epsilon,m}J_{\epsilon,B}(N).}                 \tag{PWG19}\label{PWG19}
\end{equation}
For proof use $0\le\Lambda(n)\le\log n$. The scalar majorant is decreasing once $\log n\ge\max(B,2)$, since the logarithmic derivative in $u=\log n$ is
$1/u-1/2-(u-B)/(4\epsilon)\le0$. Its sum is bounded by
\[
 \int_{\log N}^\infty u\exp\left(u/2-(u-B)^2/(8\epsilon)\right)\,du.
\]
Completing the square gives exactly \eqref{PWG18}. This retains every integer, hence every omitted prime power, without a prime-counting or RH estimate.

 C_A=|\gamma_{\rm E}+\log\pi|+\frac32+

To verify its Archimedean part, integrate the preceding cancellation formula on $0<x\le1$. The inequalities $1-e^{-x}\ge x/2$ and

 2C_{\epsilon,m},\qquad

 \mathfrak Q_{\epsilon,m}(d)={}&2C_{\epsilon,m}A_2

 \frac{(X-d)^3}{24n^2}\mathfrak Q_{\epsilon,m}(d).

 F_n(x)=\frac1{(n-1)!}\int_{-\infty}^x(x-y)^{n-1}\gamma_\epsilon(y)\,dy
 \quad(n\ge1),\qquad F_0=\gamma_\epsilon.

 (\gamma_\epsilon*k_{r,J})(x)=

 I_0(x)=\tfrac12[1+\operatorname{erf}(x/(2\sqrt\epsilon))],\quad
