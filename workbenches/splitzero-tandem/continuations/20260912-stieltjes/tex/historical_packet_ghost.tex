We use the original coordinate $s$, without replacing its multiplicative

\[
 L_Z=\bigoplus_{\rho\in Z}\mathbb C[t_\rho]/(t_\rho^{m_\rho}),
 \qquad e_{\rho,k}=t_\rho^k\quad(0\leq k<m_\rho)

\begin{proposition}[Original-coordinate identification with all jets]
\label{prop:ghost-crt}
The map
\[
 T:E_h\longrightarrow L_Z,\qquad
 T[f]_\rho=\sum_{k=0}^{m_\rho-1}
             \frac{f^{(k)}(\rho)}{k!}t_\rho^k
\]
is a complex-algebra isomorphism.  In these coordinates
\[
 TAT^{-1}=A_{\mathrm{ss}}+N,\quad
 A_{\mathrm{ss}}e_{\rho,k}=\rho e_{\rho,k},\quad
 Ne_{\rho,k}=\begin{cases}e_{\rho,k+1}&k+1<m_\rho,\\0&k+1=m_\rho.
 \end{cases}
\]
\end{proposition}
\begin{proof}
Taylor multiplication modulo $t_\rho^{m_\rho}$ proves multiplicativity.
Its kernel consists exactly of polynomials divisible by every
$(s-\rho)^{m_\rho}$ and hence by their product $h$, since the factors are
pairwise coprime.  For completeness an inverse is explicit.  Set
$h_\rho=h/(s-\rho)^{m_\rho}$, and let $a_\rho$ be the Taylor polynomial
of $1/h_\rho(s)$ at $\rho$ through degree $m_\rho-1$.  Then
$E_\rho=[a_\rho h_\rho]$ has residue $1$ at the $\rho$ factor and $0$
at every other factor.  Thus
\[
 T^{-1}e_{\rho,k}=E_\rho[(s-\rho)^k]
\]
gives the inverse.  Multiplication by $s=\rho+t_\rho$ gives the displayed
operator, including its last nilpotent column.
\end{proof}

In every subsequent operator formula on $L_Z$, the symbol $A$ denotes
the transported operator $T A T^{-1}:L_Z\to L_Z$ of
Proposition~\ref{prop:ghost-crt}.  The polynomial-coordinate quotient at

Define complex-linear permutations $C_0,S_0$ and a complex-linear parity

\begin{proposition}[Typed arithmetic involutions]
\label{prop:ghost-involutions}
Under $T$, ordinary conjugation of polynomial coefficients is the
conjugate-linear map $\mathcal C=C_0K$, and pullback by $s\mapsto1-s$ is
the complex-linear map $\mathcal S=S_0H$.  In particular
\[
 C_0=\mathcal C K,\qquad S_0=\mathcal S H,\qquad
 \mathcal C A=A\mathcal C,\qquad
 \mathcal S A\mathcal S=I-A.
\]
Here the equality involving $\mathcal C$ is an equality of conjugate-linear
maps.  Moreover $C_0N=NC_0$, $S_0N=NS_0$, and $HN=-NH$.
\end{proposition}
\begin{proof}
If $f^\#(s)=\overline{f(\bar s)}$, then
$(f^\#)^{(k)}(\rho)/k!=\overline{f^{(k)}(\bar\rho)/k!}$, giving
$\mathcal C=C_0K$.  The $k$th derivative of $f(1-s)$ at $\rho$ is
$(-1)^kf^{(k)}(1-\rho)$, giving $\mathcal S=S_0H$.  The ideal $(h)$
is preserved by both operations: conjugation fixes the monic polynomial
$h$, and $h(1-s)=(-1)^{\deg h}h(s)$.  Conjugating coefficients fixes the
indeterminate $s$, whereas substituting $1-s$ replaces multiplication by
$s$ by multiplication by $1-s$.  This proves the two identities for $A$.
The assertions about $N$ follow on each basis vector, including the vector
of largest jet degree, which both sides send to zero.
\end{proof}

\begin{theorem}[Historical ghost with complete multiplicities]
\label{thm:historical-ghost}
Set
\[
 R=C_0S_0,\qquad \Gamma=C_0-S_0,\qquad
 P_-=\frac{I-R}{2},\qquad P_+=\frac{I+R}{2}.
\]
These satisfy
\[
 \Gamma^*=\Gamma,\qquad \Gamma^2=4P_-,\qquad
 \Gamma e_{\rho,k}=0\ \Longleftrightarrow\ \Re\rho=\tfrac12.
\]
On a nonreal off-axis quartet of common multiplicity $m$, $\Gamma$ has
eigenvalues $-2$ and $2$ each with multiplicity $m$, and $0$ with
multiplicity $2m$.  Furthermore
\[
 \operatorname{rank}P_-=2m,\qquad
 \operatorname{Tr}(\Gamma^*\Gamma)=8m.
\]
On a critical-line orbit $\Gamma=P_-=0$ at every jet degree.
\end{theorem}
\begin{proof}
The permutations $C_0,S_0$ are commuting selfadjoint involutions in the
declared reference inner product.  Consequently
$(C_0-S_0)^2=2I-2C_0S_0$.  The equality
$C_0e_{\rho,k}=S_0e_{\rho,k}$ says exactly that the two labelled basis
vectors have the same root, namely $\bar\rho=1-\rho$, which is equivalent
to $2\Re\rho=1$.

On an off-axis nonreal quartet order the roots as
$(\rho,\bar\rho,1-\rho,1-\bar\rho)$.  At every fixed jet degree the
four joint $(C_0,S_0)$ eigenspaces have characters
$(1,1),(1,-1),(-1,1),(-1,-1)$.  Explicit joint eigenvectors are
\[
 e_{\rho,k}+\epsilon e_{\bar\rho,k}
       +\delta e_{1-\rho,k}+\epsilon\delta e_{1-\bar\rho,k},
 \qquad \epsilon,\delta\in\{1,-1\}.
\]
They form a basis: their pairwise inner products are zero and each has
squared norm $4$.  The eigenvalue of $\Gamma$ is $\epsilon-\delta$.
Thus its squared eigenvalues are $0,4,4,0$, giving the rank and trace
with every one of the $m$ jet degrees included.  On a critical orbit the
two permutations agree on every basis vector.
\end{proof}

For real off-axis orbits the same proof gives rank $m$ and squared trace
$4m$, since the orbit has two elements.  Actual nontrivial zeta zeros
in $0<\Re s<1$ are nonreal: for real $0<s<1$ the alternating series
$\eta(s)=\sum_{n\geq1}(-1)^{n-1}n^{-s}$ is strictly positive by grouping
successive pairs, and
$\zeta(s)=\eta(s)/(1-2^{1-s})<0$.  For completeness the paired terms satisfy
\[
 (2n-1)^{-s}-(2n)^{-s}
       =s\int_{2n-1}^{2n}x^{-s-1}\,dx.
\]
They converge uniformly on each compact subset of $\Re s>0$, because
on such a subset $|s|\leq M$, $\Re s\geq\delta>0$, and the absolute
value is at most $M(2n-1)^{-1-\delta}$.  Thus $\eta$ is analytic there.
The product identity $\eta(s)=(1-2^{1-s})\zeta(s)$ follows first in
$\Re s>1$ by splitting even terms and then by analytic continuation.
Division is used here only on the real interval $0<s<1$, where the
denominator is strictly negative.
Hence the quartet formula applies to off-axis arithmetic packets.

\begin{proposition}[The exact group-algebra quotient]
\label{prop:ghost-group-quotient}
For $W=\langle c,\sigma\mid c^2=\sigma^2=1,\ c\sigma=\sigma c\rangle$
the homomorphism
\[
 q:\mathbb C[W]\longrightarrow\mathbb C[u]/(u^2-1),\qquad
 c\longmapsto u,\quad\sigma\longmapsto u
\]
is surjective, with
\[
 I_q:=\ker q=(c-\sigma)\mathbb C[W]
  =\mathbb C(c-\sigma)\oplus\mathbb C(1-c\sigma).
\]
Under the representation $c\mapsto C_0,\sigma\mapsto S_0$,
$I_qL_Z=P_-L_Z$.  Thus $L_Z/P_-L_Z$ is the largest quotient of the
$W$-representation on which $C_0=S_0$.
\end{proposition}
\begin{proof}
In the basis $(1,c,\sigma,c\sigma)$, $q$ sends
$a+bc+d\sigma+ec\sigma$ to $(a+e)+(b+d)u$.  Its kernel is therefore
spanned by $c-\sigma$ and $1-c\sigma$.  Multiplication by $c$ sends
$c-\sigma$ to $1-c\sigma$, and multiplication by any generator stays
in their span.  This proves the ideal formula.  Since
$P_-=\Gamma^2/4$ and $\Gamma=\Gamma P_-$, their images agree and equal
$I_qL_Z$.  A quotient kills $I_q$ precisely when it kills this image;
this proves its universal property without any invariance assumption
about the arithmetic generator.
\end{proof}

\begin{theorem}[Arithmetic saturation of the historical ghost quotient]
\label{thm:ghost-arithmetic-saturation}

 Z_{\mathrm{crit}}=\{\rho\in Z:\Re\rho=\tfrac12\},\qquad

\]

\[

\]

which both the original generator $A$ descends and $C_0=S_0$ is therefore
the exact algebra and $W$-module quotient
\[
 L_Z\longrightarrow L_{Z_{\mathrm{crit}}},
 \qquad (f_\rho)_\rho\longmapsto(f_\rho)_{\rho\in Z_{\mathrm{crit}}}.
\]
Its kernel is $L_{\mathrm{off}}$, including every local multiplicity.
In polynomial coordinates this is the quotient
\[
 \mathbb C[s]/(h)\longrightarrow
 \mathbb C[s]/(h_{\mathrm{crit}}),\qquad [f]_h\longmapsto[f]_{h_{\mathrm{crit}}},
 \quad h_{\mathrm{crit}}=\prod_{\rho\in Z_{\mathrm{crit}}}
                         (s-\rho)^{m_\rho},
\]
whose kernel is $(h_{\mathrm{crit}})/(h)$; the target is zero when
$h_{\mathrm{crit}}=1$.
\end{theorem}
\begin{proof}

\[

 Ye_{\rho,k}=\Im\rho\,e_{\rho,k},\qquad

\]

then $(\tfrac12I+iY+N)v\in P_-L_Z$ whereas $Bv\in P_+L_Z$.
Therefore
\[
 P_+Av=Bv\qquad(v\in P_-L_Z).
\]

$B:P_-L_Z\to P_+L_Z$ is an isomorphism on that orbit, its inverse being
the restriction of the original diagonal operator $B^{-1}$.

$P_-L_Z+A(P_-L_Z)$.  If $Av\in P_-L_Z$ with $v\in P_-L_Z$, then

\begin{proposition}[Exact displacement and nilpotent contribution]

 \frac{W_{\mathrm{ref}}+RW_{\mathrm{ref}}R}{2}=N+N^*.
\]

\begin{proof}

identities proves both formulas, including the entire nilpotent term.
The tesserine identification uses $c_{-+}=2(2\Re\rho-1)=4B$ with its
original factor $4$.
\end{proof}
