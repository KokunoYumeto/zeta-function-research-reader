\section{Exact contiguous Gamma determinants and positive Christoffel evolution}
\label{sec:HCT}

\paragraph{HCT1. Original measure and the typed square map.}
Throughout this section
\[
 d\sigma(y)=\frac{|\Gamma(1/4+iy/2)|^2}{2\pi}\,dy,
 \qquad \gamma_0=\sqrt{2\pi},\qquad y\in\mathbb R.
\]
Let $s:\mathbb R\longrightarrow[0,\infty)$ be $s(y)=y^2$ and
let $\nu=s_*\sigma$. For every nonnegative Borel function $g$,
change of variable on each half of the original real line gives
\begin{equation}\tag{HCT1}
 \int_{\mathbb R}g(y^2)\,d\sigma(y)
 =\int_0^\infty g(t)\,
 \frac{|\Gamma(1/4+i\sqrt t/2)|^2}{2\pi\sqrt t}\,dt.
\end{equation}
Thus the factor $t^{-1/2}$ includes both preimages of $t>0$.
For integers $a,n\geq0$ put
\[
 d\nu_a(t)=t^a\,d\nu(t),\qquad
 \mu_n=\int_{\mathbb R}y^{2n}\,d\sigma(y)=\int_0^\infty t^n\,d\nu(t),
 \qquad H_n^{(a)}=\det[\mu_{a+i+j}]_{i,j=0}^{n-1},\quad H_0^{(a)}=1.
\]
No factor $\gamma_0$ is removed from any $H_n^{(a)}$.

All moments exist. Indeed, rotation by $\theta\operatorname{sgn}(y)$
in Euler's Gamma integral, with $0<\theta<\pi/2$, gives
\[
 |\Gamma(1/4+iy/2)|
 \leq \Gamma(1/4)(\cos\theta)^{-1/4}e^{-\theta|y|/2}.
\]
The rotation is justified on a truncated sector: the small circular
arc is $O(\epsilon^{1/4})$, and the large arc tends to zero because
$\cos\theta>0$. Squaring proves exponential decay of $\sigma$.
The Gamma function is continuous and nonzero at $1/4$, so the density
is positive on an interval about zero. Every nonzero polynomial has
positive squared integral against every $\nu_a$; consequently
\begin{equation}\tag{HCT2}
 H_n^{(a)}>0\qquad(a\geq0, n\geq0).
\end{equation}

\paragraph{HCT2. Gamma mass, moments, and the full real-line polynomial norms.}
The constants needed below can be derived directly from the original
Gamma weight. For $\lambda=1/4$ set $g(x)=e^{\lambda x-e^x}$.
With the Fourier convention
$\widehat g(\xi)=\int_{\mathbb R}g(x)e^{i\xi x}\,dx$, Euler's
integral gives $\widehat g(\xi)=\Gamma(\lambda+i\xi)$.
The function $g$ and all its derivatives decay faster than every
inverse power at both ends. Fourier inversion of its autocorrelation
therefore gives
\[
 \int_{\mathbb R}e^{iu\xi}|\Gamma(\lambda+i\xi)|^2\,d\xi
 =2\pi\int_{\mathbb R}g(x+u/2)g(x-u/2)\,dx
 =\frac{2\pi\Gamma(2\lambda)}{(2\cosh(u/2))^{2\lambda}}.
\]
The last equality follows by $v=e^x$ in the integral on the right.
Substituting $y=2\xi$ and $u=2w$ gives the exact characteristic
function
\[
 \int_{\mathbb R}e^{iwy}\,d\sigma(y)
 =\gamma_0(\cosh w)^{-1/2}\qquad(w\in\mathbb R).
\]
The exponential bound in HCT1 permits its analytic continuation
to the open strip $|\operatorname{Im}w|<\pi/2$. In particular
\begin{equation}\tag{HCT3}
 \int_{\mathbb R}e^{zy}\,d\sigma(y)
 =\gamma_0(\cos z)^{-1/2}\quad(|z|<\pi/2),\qquad
 \mu_n=\gamma_0(2n)![z^{2n}](\cos z)^{-1/2}.
\end{equation}
The square root in (HCT3) is positive at $z=0$.

Define monic polynomials $p_j(y)$ by the convergent generating function
on $|z|<1$, with the branches analytic at zero and equal to their
real values there,
\begin{equation}\tag{HCT4}
 \sum_{j=0}^\infty p_j(y)\frac{z^j}{j!}
 =(1+z^2)^{-1/4}\exp(y\arctan z).
\end{equation}
For sufficiently small real $z,w$, (HCT3) and
\[
 \cos(\arctan z+\arctan w)
 =\frac{1-zw}{\sqrt{1+z^2}\sqrt{1+w^2}}
\]
show that the integral of the product of the two generating functions
is $\gamma_0(1-zw)^{-1/2}$. Derivatives may pass under the integral
by the exponential bound in HCT1. Comparison of coefficients proves
\begin{equation}\tag{HCT5}
 \int_{\mathbb R}p_i(y)p_j(y)\,d\sigma(y)
 =\mathbf1_{i=j}\,\gamma_0\,j!(1/2)_j.
\end{equation}
Here $(b)_j=b(b+1)\cdots(b+j-1)$ and $(b)_0=1$.
Differentiating (HCT4) gives, with $p_{-1}=0$,
\begin{equation}\tag{HCT6}
 y p_j(y)=p_{j+1}(y)+j(j-1/2)p_{j-1}(y),\qquad p_0=1.
\end{equation}
In particular $p_{2n}$ is even and $p_{2n+1}$ is odd. This also
follows directly from $p_j(-y)=(-1)^j p_j(y)$ in (HCT4).

\paragraph{HCT3. Monic polynomials on the actual square-pushforward.}
Let $P_n^{(a)}(t)$ be the monic polynomial orthogonal to
$1,t,\ldots,t^{n-1}$ in $L^2(\nu_a)$, and write
$h_n^{(a)}=\int(P_n^{(a)})^2\,d\nu_a$.
Positivity (HCT2) gives existence and uniqueness by finite
Gram elimination. The determinant formula and its constant term are
\begin{equation}\tag{HCT7}
 P_n^{(a)}(t)=\frac1{H_n^{(a)}}
 \det\begin{bmatrix}
 \mu_a&\mu_{a+1}&\cdots&\mu_{a+n}\\
 \mu_{a+1}&\mu_{a+2}&\cdots&\mu_{a+n+1}\\
 \vdots&\vdots&&\vdots\\
 \mu_{a+n-1}&\mu_{a+n}&\cdots&\mu_{a+2n-1}\\
 1&t&\cdots&t^n
 \end{bmatrix},\qquad
 P_n^{(a)}(0)=(-1)^n\frac{H_n^{(a+1)}}{H_n^{(a)}}.
\end{equation}
For $n=0$ this means $P_0^{(a)}=1$. The last-row expansion in the
first column proves the displayed sign $(-1)^n$. Orthogonality
follows by integrating against $t^j$, which duplicates a row for
$0\leq j<n$. A monic triangular change from monomials to the
$P_j^{(a)}$ has determinant one, so
\begin{equation}\tag{HCT8}
 H_n^{(a)}=\prod_{j=0}^{n-1}h_j^{(a)},\qquad
 h_n^{(a)}=\frac{H_{n+1}^{(a)}}{H_n^{(a)}}>0.
\end{equation}

Define
\begin{align}
 c_n^{(a)}&=\frac{H_{n+1}^{(a+1)}H_n^{(a)}}
 {H_{n+1}^{(a)}H_n^{(a+1)}}>0 \quad(n\geq0),\tag{HCT9}\\
 d_n^{(a)}&=\frac{H_{n+1}^{(a)}H_{n-1}^{(a+1)}}
 {H_n^{(a)}H_n^{(a+1)}}>0 \quad(n\geq1),\qquad d_0^{(a)}=0.\tag{HCT10}
\end{align}
The exact Christoffel and inverse connection identities are
\begin{align}
 tP_n^{(a+1)}(t)&=P_{n+1}^{(a)}(t)+c_n^{(a)}P_n^{(a)}(t),\tag{HCT11}\\
 P_n^{(a)}(t)&=P_n^{(a+1)}(t)+d_n^{(a)}P_{n-1}^{(a+1)}(t)
 \quad(n\geq1).\tag{HCT12}
\end{align}
For $n=0$ the second identity reads $1=1$.
To prove (HCT11), (HCT7) gives
$P_{n+1}^{(a)}(0)+c_n^{(a)}P_n^{(a)}(0)=0$.
Division by $t$ is consequently polynomial division with zero
remainder. The quotient is monic of degree $n$ and has zero
$\nu_{a+1}$ inner product with $t^j$ for $j<n$; uniqueness proves
the identity. Taking its $\nu_a$ inner product with
$P_n^{(a+1)}$ proves
\begin{equation}\tag{HCT13}
 h_n^{(a+1)}=c_n^{(a)}h_n^{(a)}.
\end{equation}
Indeed the $P_{n+1}^{(a)}$ term has zero inner product, while
$P_n^{(a+1)}-P_n^{(a)}$ has degree at most $n-1$.
For (HCT12), $P_n^{(a)}-P_n^{(a+1)}$ has degree at most $n-1$.
Its $\nu_{a+1}$ inner products with all polynomials of degree at
most $n-2$ vanish, because multiplication of such a polynomial by
$t$ has degree at most $n-1$. It is therefore a multiple of
$P_{n-1}^{(a+1)}$. The numerator of that coefficient is
\[
 \int P_n^{(a)}P_{n-1}^{(a+1)}\,d\nu_{a+1}
 =\int P_n^{(a)}\,tP_{n-1}^{(a+1)}\,d\nu_a=h_n^{(a)},
\]
using (HCT11) at $n-1$. Dividing by $h_{n-1}^{(a+1)}$ gives
exactly (HCT10), including its positive sign.

\paragraph{HCT4. Positive discrete Toda identities.}
Combining (HCT11) and (HCT12) twice, entirely within the original
polynomial spaces, gives
\begin{equation}\tag{HCT14}
 tP_n^{(a)}=P_{n+1}^{(a)}+
 (c_n^{(a)}+d_n^{(a)})P_n^{(a)}
 +d_n^{(a)}c_{n-1}^{(a)}P_{n-1}^{(a)}.
\end{equation}
At $n=0$ the last term is absent. Expanding instead in the
$a+1$ basis gives
\[
 tP_n^{(a+1)}=P_{n+1}^{(a+1)}+
 (c_n^{(a)}+d_{n+1}^{(a)})P_n^{(a+1)}
 +c_n^{(a)}d_n^{(a)}P_{n-1}^{(a+1)}.
\]
Coefficient comparison with (HCT14) at $a+1$ proves
\begin{align}
 c_n^{(a+1)}+d_n^{(a+1)}
 &=c_n^{(a)}+d_{n+1}^{(a)} &&(n\geq0),\tag{HCT15}\\
 c_{n-1}^{(a+1)}d_n^{(a+1)}
 &=c_n^{(a)}d_n^{(a)} &&(n\geq1).\tag{HCT16}
\end{align}
All nonzero factors in these identities are positive. In the
notation $b_n^{(a)}=c_n^{(a)}+d_n^{(a)}$ and
$u_n^{(a)}=d_n^{(a)}c_{n-1}^{(a)}$, the two recurrences give the
complete $b,u$ coefficients and their shifted values, including
$b_n^{(a+1)}=c_n^{(a)}+d_{n+1}^{(a)}$ and
$u_n^{(a+1)}=c_n^{(a)}d_n^{(a)}$.

Here is the exact polynomial meaning of the matrix factorization.
Use column vectors $\mathbf P^{(a)}=(P_0^{(a)},P_1^{(a)},\ldots)^T$
as formal polynomial sequences. Let $L_a$ have diagonal $1$ and
subdiagonal $(L_a)_{n,n-1}=d_n^{(a)}$, and let $U_a$ have diagonal
$c_n^{(a)}$ and superdiagonal $1$. Every row has finite support.
Then (HCT11)--(HCT12) state
\begin{equation}\tag{HCT17}
 \mathbf P^{(a)}=L_a\mathbf P^{(a+1)},\qquad
 t\mathbf P^{(a+1)}=U_a\mathbf P^{(a)},\qquad
 J_a=L_aU_a,\quad J_{a+1}=U_aL_a.
\end{equation}
Consequently $J_{a+1}U_a=U_aJ_a$ and $J_aL_a=L_aJ_{a+1}$.
These are row-finite polynomial identities; no infinite-dimensional
bounded-operator assertion or truncation of an endpoint row is
needed. At a finite degree the identities retain every displayed
$P_{n+1}$ term. The scalar multiplication map is literally
$t:\mathbb R[t]_{\leq n}\to\mathbb R[t]_{\leq n+1}$.

\paragraph{HCT5. Exact initial Gamma data and a subtraction-free finite evolution.}
Parity, (HCT1), and uniqueness give the exact source identifications
\begin{equation}\tag{HCT18}
 p_{2n}(y)=P_n^{(0)}(y^2),\qquad
 p_{2n+1}(y)=yP_n^{(1)}(y^2).
\end{equation}
Thus (HCT5) gives, with the original mass retained,
\begin{align}
 h_n^{(0)}&=\gamma_0(2n)!(1/2)_{2n},&
 H_n^{(0)}&=\gamma_0^n\prod_{j=0}^{n-1}(2j)!(1/2)_{2j},\tag{HCT19}\\
 h_n^{(1)}&=\gamma_0(2n+1)!(1/2)_{2n+1},&
 H_n^{(1)}&=\gamma_0^n\prod_{j=0}^{n-1}(2j+1)!(1/2)_{2j+1}.\tag{HCT20}
\end{align}
Equations (HCT9)--(HCT10) therefore yield the exact positive
rational starting values
\begin{equation}\tag{HCT21}
 c_n^{(0)}=(2n+1)(2n+1/2),\qquad
 d_n^{(0)}=2n(2n-1/2),\qquad n\geq0.
\end{equation}

For a fixed step $a\to a+1$ introduce auxiliary numbers $e_n$ by
\begin{align}
 e_0&=c_0^{(a)},&d_0^{(a+1)}&=0,&
 c_0^{(a+1)}&=e_0+d_1^{(a)},\tag{HCT22}\\
 e_n&=\frac{c_n^{(a)}e_{n-1}}{e_{n-1}+d_n^{(a)}},&
 d_n^{(a+1)}&=\frac{c_n^{(a)}d_n^{(a)}}{e_{n-1}+d_n^{(a)}},&
 c_n^{(a+1)}&=e_n+d_{n+1}^{(a)}\quad(n\geq1).\tag{HCT23}
\end{align}
Every denominator and every $e_n,c_n^{(a+1)}$ is strictly positive;
$d_n^{(a+1)}>0$ for $n\geq1$. These formulas use only positive
sums, products, and quotients. To verify them, (HCT15) at zero
gives (HCT22). If $c_{n-1}^{(a+1)}=e_{n-1}+d_n^{(a)}$, then
(HCT16) gives the middle formula in (HCT23). The first formula
implies $e_n+d_n^{(a+1)}=c_n^{(a)}$; substituting this in
(HCT15) gives the last formula. This proves the recurrence and
the claimed positivity by induction.

It follows, without determinant cancellation, that for any finite
integer $a,n\geq0$
\begin{equation}\tag{HCT24}
 H_n^{(a)}=H_n^{(0)}
 \prod_{s=0}^{a-1}\prod_{j=0}^{n-1}c_j^{(s)}.
\end{equation}
Indeed $H_n^{(a+1)}/H_n^{(a)}=\prod_{j<n}c_j^{(a)}$ follows
from (HCT8) and (HCT13), and then telescopes in $a$.
An initial array through index $N+A$ suffices to compute all
$c_n^{(a)},d_n^{(a)}$ with $a\leq A$, $n\leq N$: one may
retain indices $0\leq n\leq N+A-a$ after step $a$ and shrink
the retained interval by one at each subsequent step.
Thus the formulas give a finite exact calculation for arbitrarily
specified original ranks and shifts. The assertion here concerns
an exact finite recurrence, with no unproved assertion about its
computational cost or large-rank limit.

\paragraph{HCT6. Sign-correct contiguous determinants.}
For $n\geq1$, apply the Desnanot--Jacobi identity to the
$(n+1)\times(n+1)$ matrix $[\mu_{a+i+j}]_{i,j=0}^{n}$,
deleting its first and last rows and columns. The two principal
minors are $H_n^{(a+2)}$ and $H_n^{(a)}$, the two cross minors
are both $H_n^{(a+1)}$, and the central minor is
$H_{n-1}^{(a+2)}$. Their original index order is retained. Hence
\begin{equation}\tag{HCT25}
 H_n^{(a)}H_n^{(a+2)}-(H_n^{(a+1)})^2
 =H_{n+1}^{(a)}H_{n-1}^{(a+2)}>0.
\end{equation}
For completeness, the determinant identity used here follows by
Schur-complement elimination of the central block, whose determinant
is positive by (HCT2). The remaining $2\times2$ determinant is
the product of its diagonal entries minus the product of its
off-diagonal entries; multiplication by the central determinant
squared gives (HCT25). For $n=1$ the central block has size zero
and determinant one, giving the same identity directly.

Put $\kappa_0^{(a)}=0$ and, for $n\geq1$,
\begin{equation}\tag{HCT26}
 \kappa_n^{(a)}=
 \frac{H_{n+1}^{(a)}H_{n-1}^{(a+2)}}{(H_n^{(a+1)})^2}>0.
\end{equation}
Then the exact logarithmic second difference is
\begin{equation}\tag{HCT27}
 \log H_n^{(a+2)}-2\log H_n^{(a+1)}+\log H_n^{(a)}
 =\log(1+\kappa_n^{(a)}).
\end{equation}
For $n\geq1$ its equivalent positive Toda expression is
\begin{equation}\tag{HCT28}
 0<\frac{d_n^{(a)}}{c_{n-1}^{(a+1)}}
 =\frac{\kappa_n^{(a)}}{1+\kappa_n^{(a)}}<1,\qquad
 \log(1+\kappa_n^{(a)})
 =-\log\left(1-\frac{d_n^{(a)}}{c_{n-1}^{(a+1)}}\right).
\end{equation}
Substitution of (HCT9)--(HCT10), followed by (HCT25), proves
every equality and inequality in (HCT28).
In particular, if $R_n^{(a)}=H_n^{(a+1)}/H_n^{(a)}$, then
\begin{equation}\tag{HCT29}
 R_n^{(a)}=\prod_{j=0}^{n-1}c_j^{(a)},\qquad
 \frac{R_n^{(a+1)}}{R_n^{(a)}}=1+\kappa_n^{(a)}.
\end{equation}
For a finite integer $l\geq1$, exact telescoping gives
\begin{equation}\tag{HCT30}
 \log\frac{H_n^{(a+l+1)}/H_n^{(a+1)}}
 {H_n^{(a+l)}/H_n^{(a)}}
 =\sum_{s=0}^{l-1}\log(1+\kappa_n^{(a+s)}).
\end{equation}
This quantity is strictly positive when $n\geq1$.
The signed claim in (HCT27)--(HCT30) concerns $a$-differences
of the indicated determinant ratios. The degree difference is
the different exact identity
$\log R_{n+1}^{(a)}-\log R_n^{(a)}=\log c_n^{(a)}$;
no sign or asymptotic size of this last logarithm is needed.

\paragraph{HCT7. Return to both original full matrices.}
Retain the full matrices from the original real coordinate:
\begin{equation}\tag{HCT31}
 M_{a,r}[i,j]=\int_{\mathbb R}y^{2a+i+j}\,d\sigma(y)
 \quad(0\leq i,j<r),\qquad \mathcal D_{a,r}=\det M_{a,r}.
\end{equation}
Let $n_0=\lceil r/2\rceil$, $n_1=\lfloor r/2\rfloor$.
Permute the original indices to $0,2,4,\ldots,1,3,5,\ldots$
using the same permutation in the source and its dual. Odd
moments vanish by evenness of $\sigma$. The even-even block is
$[\mu_{a+i+j}]_{i,j<n_0}$ and the odd-odd block is
$[\mu_{a+1+i+j}]_{i,j<n_1}$. Thus the exact congruence and
its determinant are
\begin{equation}\tag{HCT32}
 P_r M_{a,r}P_r^T=
 \operatorname{diag}\big([\mu_{a+i+j}]_{i,j<n_0},
 [\mu_{a+1+i+j}]_{i,j<n_1}\big),\qquad
 \mathcal D_{a,r}=H_{n_0}^{(a)}H_{n_1}^{(a+1)}.
\end{equation}
The determinant sign is $\det(P_r)^2=1$.
At the function-space level the same map sends the coefficient
vector of $p(t)+yq(t)$, with $t=y^2$, to the two coefficient
vectors of $(p,q)$. Its exact norm identity is
\[
 \int_{\mathbb R}y^{2a}|p(y^2)+yq(y^2)|^2\,d\sigma(y)
 =\int_0^\infty|p(t)|^2\,d\nu_a(t)
  +\int_0^\infty|q(t)|^2\,d\nu_{a+1}(t).
\]
The cross term is odd. This proves the typed orthogonal direct-sum
map with both original parity sectors retained.

For the even integer $q=e(k+1)^2=2n$ specified by the packet,
the two original ranks are therefore exactly
\begin{equation}\tag{HCT33}
 \mathcal D_{a,q}=H_n^{(a)}H_n^{(a+1)},\qquad
 \mathcal D_{a,q+1}=H_{n+1}^{(a)}H_n^{(a+1)}.
\end{equation}
Only the stipulated integrality and evenness of $q$ are used here;
the packet parameter $e$ is left unchanged.
In particular
\begin{equation}\tag{HCT34}
 \frac{\mathcal D_{a,q+1}}{\mathcal D_{a,q}}=h_n^{(a)},\qquad
 \frac{\mathcal D_{a+l,q+1}/\mathcal D_{a,q+1}}
 {\mathcal D_{a+l,q}/\mathcal D_{a,q}}
 =\prod_{s=0}^{l-1}c_n^{(a+s)}.
\end{equation}
The empty product at $l=0$ is one.

Write $Z_a=\mathcal D_{a,q}\mathcal D_{a,q+1}$ while keeping
the two original ranks fixed. Equations (HCT29), (HCT33) give
\begin{align}
 Z_a&=H_n^{(a)}H_{n+1}^{(a)}(H_n^{(a+1)})^2,\tag{HCT35}\\
 \frac{Z_{a+1}}{Z_a}
 &=R_n^{(a)}R_{n+1}^{(a)}(R_n^{(a+1)})^2,\tag{HCT36}\\
 \frac{Z_{q+l}}{Z_q}
 &=\prod_{s=0}^{l-1}
 \left\{
 \left(\prod_{j=0}^{n-1}c_j^{(q+s)}\right)
 \left(\prod_{j=0}^{n}c_j^{(q+s)}\right)
 \left(\prod_{j=0}^{n-1}c_j^{(q+s+1)}\right)^2
 \right\}.\tag{HCT37}
\end{align}
Equation (HCT37) evaluates the paired determinant ratio at the
requested original exponents $a=q,q+l$ by the positive rational
evolution (HCT21)--(HCT23). Each of the four factors supplied by
the two full matrices remains visible. The $\gamma_0$ factors
in (HCT19)--(HCT24) cancel in this particular ratio because both
numerator and denominator have the same total matrix dimension;
they are present in the individual determinants throughout.

Finally the exact curvature of this same paired determinant is
\begin{equation}\tag{HCT38}
 \frac{Z_{a+2}Z_a}{Z_{a+1}^2}
 =(1+\kappa_n^{(a)})(1+\kappa_{n+1}^{(a)})
  (1+\kappa_n^{(a+1)})^2>1.
\end{equation}
For $l\geq1$ the consequent finite shift identity is
\begin{align}
 \log\frac{Z_{a+l+1}/Z_{a+1}}{Z_{a+l}/Z_a}
 =\sum_{s=0}^{l-1}\big[&\log(1+\kappa_n^{(a+s)})
 +\log(1+\kappa_{n+1}^{(a+s)})\notag\\
 &+2\log(1+\kappa_n^{(a+s+1)})\big]>0.\tag{HCT39}
\end{align}
These statements follow by adding (HCT27) for exactly the four
blocks in (HCT35), exponentiating for (HCT38), and telescoping
for (HCT39); they make no replacement of $q$ by an
independent large parameter. They provide the full finite return,
its positive contiguous increments, and its exact degree-step
comparison. A growth rate along the packet values of $q,k,l$
requires evaluating these explicit positive quantities; no such
rate is assumed in any identity above.
