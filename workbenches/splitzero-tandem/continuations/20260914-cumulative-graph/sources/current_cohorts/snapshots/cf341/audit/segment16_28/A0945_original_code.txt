def record(name, count):

def s_add(a,b,p=None):
    if a is None: return b
    if b is None: return a
    return (a+b)%p if p else a+b

def s_mul(a,b,p=None):
    if a is None or b is None: return None
    return (a*b)%p if p else a*b

def plus(a,b,p=None): return tuple(s_add(x,y,p) for x,y in zip(a,b))
def mult(a,b,p=None):
    n=len(a); out=[None]*n
    for i,x in enumerate(a):
        for j,y in enumerate(b):
            v=s_mul(x,y,p)
            if i+j>=n: v=s_mul(-1 if p is None else p-1,v,p)
            out[(i+j)%n]=s_add(out[(i+j)%n],v,p)
    return tuple(out)
def mask(a): return tuple(x is not None for x in a)
def amp(a): return tuple(0 if x is None else x for x in a)
def saturated(a): return tuple(None for x in a) if all(x is None for x in a) else amp(a)
def ring_mult(a,b,p=None): return amp(mult(amp(a),amp(b),p))
def support_mult(a,b):
