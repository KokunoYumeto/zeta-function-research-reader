\section{The signed Gamma return through its complete relation subspaces}
\label{sec:gsr-signed-gamma}

Retain exactly GCR1--GCR5: $k=4l+1$, $l\geq1$,
$c=k/2$, $e_k=1+k(m-1)$, $q=e_k(k+1)^2$, and
\[
\begin{gathered}
 \lambda_{a,b}=c+(2a-k)\delta+i(2b-k)\gamma,\quad
 \chi(S)=\prod_{a,b=0}^k(S-\lambda_{a,b})^{e_k},\\
 0<\delta<\tfrac12,\quad\gamma>2,\quad S=c+iy,\qquad
 f(S)=\prod_{j=0}^{l-1}(S-c-t_j),\quad t_j=2j+\tfrac12,\\
 d\sigma(y)=|\Gamma(1/4+iy/2)|^2\,dy/(2\pi),\qquad
 dm_{0,k}=\beta_k|f(c+iy)|^2d\sigma,\quad
 \beta_k=(2\pi)^{k/2}/(\sqrt2\,\Gamma(k/2)).
\end{gathered}\tag{GSR1}
\]
The original $h$, $v_h=(2\xi)/h$, complete Taylor unit,
remainder map $J_{\chi,N}$ and observation $\Lambda$ are the
unchanged ones of GCR1--GCR2 and GCR41--GCR44.
Let
\[
 N_0=q-1,\quad N_1=q,\quad N_2=2q-1,\quad N_3=2q,
 \qquad(\epsilon_0,\epsilon_1,\epsilon_2,\epsilon_3)=(1,1,-1,-1).
\]
The object calculated below is precisely
$\mathfrak T_k=\sum_a\epsilon_a
(\log\det G_{N_a}^0-\log\det G_{N_a}^\sigma)$
from GCR26. No source mass or packet coordinate is changed.

\subsection{The two exact low endpoints}

Use the full source determinant product and source bound
\[
\begin{gathered}
 D_{l,N}=\prod_{a=0}^N\prod_{r=0}^{2l-1}(a+r+\tfrac12),\qquad
 \mathscr U_{l,N}=\prod_{j=0}^{l-1}
                         \bigl(4(N+j+1)^2+t_j^2\bigr),\\
 X_{l,N}=(N+1)\log\mathscr U_{l,N}-\log D_{l,N}\geq0.
\end{gathered}\tag{GSR2}
\]
These are exactly GCR28--GCR30. Put
$A_N=\log\det G_N^f-\log\det G_N^\sigma$, where $G_N^0=\beta_kG_N^f$.
For $N\geq q-1$ let
\[
 r_N=N+1-q,\qquad
 R_N=
 \frac{\det\operatorname{Gram}_\sigma
                 (f\chi,f\chi S,\ldots,f\chi S^{r_N-1})}
      {\det\operatorname{Gram}_\sigma
                 (\chi,\chi S,\ldots,\chi S^{r_N-1})}.
\tag{GSR3}
\]
Both empty determinants at $r_N=0$ are one. GCR19 proves
the exact identity
\[
 A_N=\log D_{l,N}-\log R_N.
\tag{GSR4}
\]
In particular, define the full original moment ratio
\[
 \mathfrak r_\chi=
 \frac{\displaystyle\int_{\mathbb R}
              |f(c+iy)\chi(c+iy)|^2\,d\sigma(y)}
      {\displaystyle\int_{\mathbb R}
              |\chi(c+iy)|^2\,d\sigma(y)}>0.
\tag{GSR5}
\]
Then
\[
 A_{q-1}=\log D_{l,q-1},\qquad
 A_q=\log D_{l,q}-\log\mathfrak r_\chi.
\tag{GSR6}
\]
The first source has no relation directions and the second
has the single original direction $\chi$. This is an equality,
not an estimate which treats either kernel as having growing rank.
All moments are finite, since the original Gamma density has
exponential tails and both polynomials have finite degree.
Pointwise $|f(c+iy)|^2=\prod_j(y^2+t_j^2)\geq\prod_jt_j^2$;
the original polynomial bound GCR29 on the degree-$q$ input
$\chi$ proves
\[
 \prod_{j=0}^{l-1}t_j^2\leq\mathfrak r_\chi\leq\mathscr U_{l,q}.
\tag{GSR7}
\]
Consequently $|\log\mathfrak r_\chi|=O_m(k\log(k+2))$ along
the stated degrees. For example the upper logarithm in GSR7
is bounded by
$l\log(4(q+l)^2+4l^2)$, and the lower logarithm is at least
$-\log4$, since $t_0^2=1/4$ and $t_j^2>1$ for $j\geq1$.
In particular $\log\mathfrak r_\chi/(kq)\to0$.

The four copies of $q\log\beta_k$ cancel with their original
signs. Equations GSR4--GSR6 therefore give
\[
\begin{gathered}
 C_k=\log D_{l,q-1}+\log D_{l,q}-\log\mathfrak r_\chi
                     -\log D_{l,2q-1}-\log D_{l,2q},\\
 \boxed{\quad \mathfrak T_k=C_k+\log R_{2q-1}+\log R_{2q}.\quad}
\end{gathered}
\tag{GSR8}
\]
Thus the two high-degree determinants in this formula are the
complete original relation determinants.

\subsection{A finite Jacobi realization with the original source metric}

Let $b_n(y)$ and $\gamma_n$ be exactly GCR11, and let
$e_n(y)=b_n(y)/\sqrt{\gamma_n}$. For a positive integer $D$
define the real symmetric matrix $\mathcal J_D$ on coordinates
$0,\ldots,D-1$ by
\[
 (\mathcal J_D)_{a-1,a}=(\mathcal J_D)_{a,a-1}
       =\sqrt{a(a-\tfrac12)}\quad(1\leq a<D),
 \qquad(\mathcal J_D)_{a,a}=0.
\tag{GSR9}
\]
For $P\in\mathcal P_{D-1}$ in the original $S$ coordinate,
write
\[
 P(c+iy)=\sum_{n=0}^{D-1}z_n e_n(y),\qquad
 \mathcal C_D(P)=(z_0,\ldots,z_{D-1})^t .
\tag{GSR10}
\]
This is an explicit linear isometry from the original polynomial
source with its $\sigma$ norm to $\mathbb C^D$ with its Euclidean
norm. Its inverse is the polynomial
$S\mapsto\sum_nz_ne_n((S-c)/i)$. The orthonormal coordinate
frame has not altered $d\sigma$ or its mass $\gamma_0=\sqrt{2\pi}$.
The recurrence proves
$\mathcal C_D(((S-c)/i)P)=\mathcal J_D\mathcal C_D(P)$
whenever $\deg P\leq D-2$.

At a fixed endpoint $N$ put
\[
 D=N+l+1,\qquad
 \Phi_l(x)=\prod_{j=0}^{l-1}(x^2+t_j^2),\qquad
 F_l(x)=\prod_{j=0}^{l-1}(x+it_j).
\tag{GSR11}
\]
Every intermediate product of degree at most $l$ with an input
in $\mathcal P_N$ has degree at most $N+l=D-1$. Applying the
recurrence successively before reaching that last degree gives
the exact multiplication identity
\[
 \mathcal C_D(fP)=i^l F_l(\mathcal J_D)\mathcal C_D(P)
       \quad(P\in\mathcal P_N).
\tag{GSR12}
\]
Since $\mathcal J_D$ is self-adjoint,
$F_l(\mathcal J_D)^*F_l(\mathcal J_D)=\Phi_l(\mathcal J_D)>0$.
This equality concerns the squared norm of the exact degree-$l$
map in GSR12; it does not assert that multiplication by a
degree-$2l$ polynomial has no truncation on every source vector.

For the original relation space put
\[
 Z_N=
 \big(\mathcal C_D(\chi),\mathcal C_D(\chi S),\ldots,
                         \mathcal C_D(\chi S^{r_N-1})\big),\qquad
 B_N=Z_N(Z_N^*Z_N)^{-1/2}.
\tag{GSR13}
\]
At $r_N>0$ its columns are independent: cancellation of the
nonzero monic $\chi$ proves independence of the original
polynomials, and GSR10 is injective. Hence $B_N^*B_N=I_{r_N}$.
The exact relation determinant becomes
\[
 \boxed{\quad R_N=\det\bigl(B_N^*\Phi_l(\mathcal J_D)B_N\bigr).\quad}
\tag{GSR14}
\]
Indeed its denominator in GSR3 is $\det(Z_N^*Z_N)$; its
numerator, by GSR12, is
$\det(Z_N^*\Phi_l(\mathcal J_D)Z_N)$.
This proves GSR14 with the full relation subspace present.

The characteristic polynomial of $\mathcal J_D$ is $b_D$.
Expansion of the tridiagonal determinant along its last row
gives the monic recurrence with initial values $1,y$, exactly
GCR11. Its $D$ eigenvalues $x_{D,i}$ are real and simple.
To prove simplicity, the first coordinate of any eigenvector
determines the second by the first row, and then determines
every subsequent coordinate by the nonzero off-diagonal
recurrence; a zero first coordinate forces the whole vector
to vanish. Each eigenspace thus has dimension one, and
self-adjointness gives a full orthonormal eigenbasis.
Let
\[
 0<\pi_{D,1}\leq\cdots\leq\pi_{D,D}
\]
be the eigenvalues of $\Phi_l(\mathcal J_D)$, with multiplicity.
They are exactly the $D$ positive numbers
$\Phi_l(x_{D,i})$, ordered by $|x_{D,i}|$.
In its orthonormal eigenbasis the Cauchy--Binet formula gives
\[
 R_N=\sum_{\substack{I\subset\{1,\ldots,D\}\\|I|=r_N}}
       w_{N,I}\prod_{i\in I}\pi_{D,i},\qquad
 w_{N,I}\geq0,\quad\sum_Iw_{N,I}=1.
\tag{GSR15}
\]
Here $w_{N,I}$ is the squared determinant of the rows $I$
of the actual matrix $B_N$ in that eigenbasis. The last equality
is Cauchy--Binet applied to $B_N^*B_N=I$.
It follows that the finite spectral bounds are
\[
 s^-_{l,N}:=\sum_{i=1}^{r_N}\log\pi_{D,i}
       \leq\log R_N
       \leq\sum_{i=D-r_N+1}^D\log\pi_{D,i}
                       =:s^+_{l,N}.
\tag{GSR16}
\]
Empty sums are zero. All eigenvalues here come from the explicitly
specified original recurrence matrix; no packet-limit model has
been substituted for it.

\subsection{The actual positive weights and the original mixed determinant}

The weights in GSR15 can be written explicitly, so that its
comparison is tied to the complete original packet. Let $v_i$
be a real unit eigenvector of $\mathcal J_D$ at $x_{D,i}$,
with first coordinate positive, and put
\[
 \omega_{D,i}=\gamma_0(v_i)_0^2>0,\qquad
                    \sum_i\omega_{D,i}=\gamma_0.
\tag{GSR17}
\]
The eigenvector recurrence gives
$(v_i)_n=\sqrt{\omega_{D,i}}\,e_n(x_{D,i})$.
It follows directly from GSR10 that
$v_i^*\mathcal C_D(P)=
\sqrt{\omega_{D,i}}\,P(c+ix_{D,i})$ for every
$P\in\mathcal P_{D-1}$. This proves the evaluation formula
with its original mass, without introducing a probability
normalization of the source.

Order a subset $I$ by its distinct real nodes. The corresponding
minor of $Z_N$ is a product of the evaluation values of $\chi$,
the square roots of $\omega_{D,i}$, and the Vandermonde
determinant of $c+ix_{D,i}$. Thus
\[
 w_{N,I}=
 \frac{\displaystyle
   \prod_{i\in I}\omega_{D,i}|\chi(c+ix_{D,i})|^2
   \prod_{\substack{i<j\\i,j\in I}}(x_{D,j}-x_{D,i})^2}
 {\displaystyle
   \det\operatorname{Gram}_\sigma
               (\chi,\chi S,\ldots,\chi S^{r_N-1})}.
\tag{GSR18}
\]
The absolute value of the Vandermonde phase
$i^{r_N(r_N-1)/2}$ is one. Its original coordinate shift $c$
cancels in the differences, not in the original columns.
Every packet factor in this formula is explicitly
\[
 |\chi(c+ix)|^2
  =\prod_{a,b=0}^k
     \left(((2a-k)\delta)^2+
                    (x-(2b-k)\gamma)^2\right)^{e_k}>0.
\tag{GSR19}
\]
Positivity uses that $2a-k$ is nonzero for this odd $k$.
Thus every weight in GSR18 is strictly positive when $r_N>0$.
The complete multiplicity $e_k$ and every original mixed support
point enter this exact formula.

The weights sum to one because they are squared minors of
an isometric frame. They are coefficients in a finite
determinant identity, not replacement source masses. In
particular the original quadrature weights in GSR17 retain
their total $\sqrt{2\pi}$.

For a direct connection to the complete mixed matrix of GCR26,
let $U_f$ be the full jet multiplication matrix in GCR7,
and let $d_N$ be precisely GCR23. Subtracting the fixed
quotient formula in GCR25 and using GSR4 yields
\[
 \log d_N
     =2\log|\det U_f|-\log D_{l,N}+\log R_N.
\tag{GSR20}
\]
Every matrix in GSR14--GSR18 therefore maps to that same
$2l$ determinant by this exact scalar identity. The
degree-independent unit factor cancels only in the four signed
terms, giving GSR8 and GCR26 with identical values.
No mixed term in the positive factorization GCR24 has been
dropped; GSR20 computes its full determinant.

\subsection{A strictly narrower finite signed enclosure}

The scalar source inequality GCR29 and the determinant
compression argument GCR31--GCR32 also give
\[
 \log D_{l,N}-q\log\mathscr U_{l,N}
    \leq\log R_N\leq r_N\log\mathscr U_{l,N}.
\tag{GSR21}
\]
The lower side follows by applying
$\log\det G_N^f-\log\det G_N^\sigma\leq
q\log\mathscr U_{l,N}$ in GSR4.
The upper side is the direct determinant inequality on the
actual relation space, which has dimension $r_N$.
The pointwise lower bound for $|f|^2$ gives in addition
$\log R_N\geq r_N\sum_j\log t_j^2$.
Define the following entirely finite numbers:
\[
\begin{split}
 L_{l,N}
 &=\max\left\{s^-_{l,N},\
     \log D_{l,N}-q\log\mathscr U_{l,N},\
                       r_N\sum_{j=0}^{l-1}\log t_j^2\right\},\\
 U_{l,N}
 &=\min\left\{s^+_{l,N},\
                         r_N\log\mathscr U_{l,N}\right\}.
\end{split}\tag{GSR22}
\]
All entries are explicit products or eigenvalues of GSR9;
the full original moment in $C_k$ remains exactly GSR5.
Equations GSR8, GSR16 and GSR21 prove
\[
 \boxed{\quad
 C_k+L_{l,2q-1}+L_{l,2q}
       \leq\mathfrak T_k
       \leq C_k+U_{l,2q-1}+U_{l,2q}.
 \quad}\tag{GSR23}
\]
This interval is contained in GCR33. Indeed at each high
endpoint GSR22 includes both bounds used to prove GCR33;
at the low endpoints GSR6 supplies their exact values instead
of the two ranges used there.
Its upper side is strictly smaller for $l\geq1$.
To check this without relying on a spectral approximation,
the high upper choices in GSR22 give
\[
 C_k+U_{l,2q-1}+U_{l,2q}
 \leq A_{q-1}+A_q
       -q\log\mathscr U_{l,2q-1}
       -q\log\mathscr U_{l,2q}
       +X_{l,2q-1}+X_{l,2q}.
\]
In contrast, the upper side of GCR33 replaces
$A_{q-1}+A_q$ by
$q\log\mathscr U_{l,q-1}+q\log\mathscr U_{l,q}$.
Their difference is at least $X_{l,q-1}>0$.
For strict positivity, the row-sum estimate in GCR29 is strict
on the constant input at the first multiplication step:
$\|y\|_\sigma^2/\|1\|_\sigma^2=1/2<4q^2$ at $N=q-1$.
The later factors are positive and their stated upper bounds
remain valid, so the full source inequality has a strictly
smaller Rayleigh quotient on that nonzero vector.
Since all its relative eigenvalues are at most one, its
determinant is strictly smaller than one. This is exactly
$X_{l,q-1}>0$.

 \ker(\Lambda J_{\chi,N})
    =\chi\mathcal P_{N-q}\ \mathbin{\perp_{H_N^f}}\

\tag{GSR24}

The first summand is sent to the exact matrix image in the

\subsection{The resulting moving signed constants}

The complete finite proof JQR1--JQR39, included with this source,
gives an elementary counting law for the same matrices in GSR9.
Define the exact number $a\in(0,2)$ and the exact integral $I$ by
\[
 a\operatorname{arcosh}(2/a)+2\arcsin(a/2)=\pi/2,\qquad
 I=\frac1\pi\int_a^2\log x\,\operatorname{arcosh}(2/x)\,dx.
\tag{GSR25}
\]
Existence and uniqueness of $a$ follow from the strictly
increasing function in JQR13--JQR14. It is the middle absolute
eigenvalue cutoff of the proved limiting counting measure,
not a new parameter of the original quartet.
The positive and negative finite products in GSR16 satisfy
at exactly $N=2q-1,2q$, with $D=N+l+1$, $r_N=N+1-q$,
\[
\begin{split}
 s^+_{l,N}
     &=2lr_N\log D+2lD I+o(lD),\\
 s^-_{l,N}
     &=2lr_N\log D-2lD(1+I)+o(lD).
\end{split}\tag{GSR26}
\]
Here JQR22 is an explicit finite upper-product error;
JQR34 and JQR38--JQR39 give explicit finite full- and
lower-product errors. In particular no lower-tail logarithm
is inferred from weak convergence near a zero eigenvalue.
The auxiliary spectral coordinate $x_{D,i}/D$ is returned
to the original eigenvalue by every factor of $D$ in GSR26.

To verify the moving domains, the two pairs are precisely
\[
 (D,r_N)=(2q+l,q),\quad(2q+l+1,q+1).
\tag{GSR27}
\]
They obey $r_N/D\to1/2$, $l/D\to0$ and
$l\log(2(D+l))/D\to0$, since
$q=e_k(k+1)^2\geq(k+1)^2$ and $q=O_m(k^3)$.
These are the domains required in the fully finite JQR
proof. Both upper products and their complementary ranks
are thereby controlled at these actual degrees.

For explicit cancellation of the growing elementary terms,
GCR35 says
\[
 \log D_{l,N}=2l(N+1)(\log(N+1)-1)+\theta_{l,N+1},
\]
where its displayed bound for $\theta$ is
$O(l^2\log(N+2))$ on these degrees. GSR7 gives
$\log\mathfrak r_\chi=o(lq)$.
Using $N_a+1=q,q+1,2q,2q+1$ in GSR8 yields
\[
 \frac{C_k}{2lq}
          =-2\log q-4\log2+2+o(1).
\tag{GSR28}
\]
For instance replacing $(q+1)\log(q+1)$ by $q\log q$
costs $O(\log(q+1))$ by the mean value theorem; after
division by $q$ this tends to zero. The same reasoning
handles $2q+1$, and the finite $\theta$ errors divided
by $lq$ are $O(l\log q/q)\to0$. Thus the equality
retains, and then cancels, the terms in $\log q$.

The two upper-product expansions in GSR26 give
\[
 \frac{s^+_{l,2q-1}+s^+_{l,2q}}{2lq}
             =2\log q+2\log2+4I+o(1).
\tag{GSR29}
\]
Indeed $D/q\to2$, the two ranks are $q,q+1$, and
$D=2q+O(l)$, so
$\log D=\log(2q)+O(l/q)$.
The lower products instead give
\[
 \frac{s^-_{l,2q-1}+s^-_{l,2q}}{2lq}
             =2\log q+2\log2-4(1+I)+o(1).
\tag{GSR30}
\]
Insert these bounds in GSR8 using GSR16, and then use
$2l/k\to1/2$. The resulting signed conclusion is
\[
 \boxed{\quad
 -1-\log2-2I
 \leq\liminf_{\substack{k\to\infty\\k\equiv1\ (4)}}
                    \frac{\mathfrak T_k}{kq}
 \leq\limsup_{\substack{k\to\infty\\k\equiv1\ (4)}}
                    \frac{\mathfrak T_k}{kq}
 \leq1-\log2+2I.
 \quad}\tag{GSR31}
\]
This is a statement about the actual signed GCR26 value.
The fully finite estimate is GSR23, which is valid also
before any finite JQR cutoff guard holds.

The constants in GSR31 are strictly sharper than those
in GCR39, without relying on a decimal evaluation of $I$.
The counting measure of JQR6 has total mass one and second
moment $2/3$. Its restriction to $|x|\geq a$ has mass $1/2$.
Concavity of $\log$ on this restriction, applied to $x^2$,
therefore gives
\[
 I=\frac12\int_{|x|\geq a}\log x^2\,d\nu(x)
 \leq\frac14\log\left(2\int_{|x|\geq a}x^2\,d\nu(x)\right)
 \leq\frac14\log(4/3).
\tag{GSR32}
\]
This application only averages the explicitly derived spectral
counting law $\nu$; it does not rescale either original Gamma source.
Consequently the upper constant of GSR31 is at most
$1-\tfrac12\log3<1$, and its lower constant is at least
$-1-2\log2+\tfrac12\log3>-1-2\log2$.
GSR25 is the exact value of the sharper constants; no
numerical approximation is needed for these strict improvements.

The original source correction is still
$\Delta_k^\Gamma=\delta_k^\sigma-\mathfrak T_k$ by GCR27.
Using the already proved determinant mass-rate return
$\delta_k^\sigma=o(kq)$ gives
\[
 \boxed{\quad
 -1+\log2-2I
 \leq\liminf_{\substack{k\to\infty\\k\equiv1\ (4)}}
                       \frac{\Delta_k^\Gamma}{kq}
 \leq\limsup_{\substack{k\to\infty\\k\equiv1\ (4)}}
                       \frac{\Delta_k^\Gamma}{kq}
 \leq1+\log2+2I.
 \quad}\tag{GSR33}
\]
The interval in GSR31, like its finite version GSR23,
is an enclosure of the signed value, not its asserted value.
The actual packet weights in GSR18 remain present in GSR15
and GSR20. This proof establishes neither a nonzero leading
value nor a vanishing leading value for that weighted sum;
it retains the sum itself as the complete finite object.

\section{A finite Jacobi calculation for the moving upper spectral product}

The matrices and factors in this calculation have their original coefficients.
For an integer \(D\geq1\), let \(J_D\) be the real symmetric matrix on
\(\mathbb R^D\), with coordinates indexed by \(0,\ldots,D-1\), whose diagonal
is zero and whose two entries joining coordinates \(n-1,n\) are
\[
 a_n=\sqrt{n(n-\tfrac12)},\qquad 1\leq n\leq D-1.
 \tag{JQR1}
\]
Put \(A_D=J_D/D\), with the exact inverse relation \(J_D=D A_D\).
This is an auxiliary coordinate for the finite spectrum.  Every power of
\(D\) is retained, in particular in (JQR23); it does not replace the
original source coordinate \(c=k/2\) or the original source measure.
The eigenvalues of \(A_D\), counted with multiplicity, define
\(\mu_D=D^{-1}\sum_{j=1}^D\delta_{\lambda_j(A_D)}\).
For an integer \(l\geq1\), retain every factor in
\[
 t_j=2j+\tfrac12,\qquad
 \Phi_l(y)=\prod_{j=0}^{l-1}(y^2+t_j^2),\qquad
 B_l=\sum_{j=0}^{l-1}t_j^2
     =\frac{l(16l^2-12l-1)}{12}.
 \tag{JQR2}
\]
All eigenvalues of \(\Phi_l(J_D)\) are positive.  They are ordered by the
absolute values of the corresponding eigenvalues of \(J_D\), because
\(\Phi_l(y)\) is strictly increasing as a function of \(|y|\geq0\).
For \(0\leq r\leq D\), write \(\Pi^+_{D,l,r}\) for the product of its
largest \(r\) eigenvalues.  An empty product is one.

\paragraph{The exact finite walk estimate.}
The following bounds hold for every \(D\geq1\) and integer \(p\geq1\):
\[
 \operatorname{spec}(A_D)\subset[-2,2],\qquad
 \frac1D\operatorname{tr}A_D^{\,2p+1}=0,\qquad
 \left|\frac1D\operatorname{tr}A_D^{\,2p}
       -\frac{\binom{2p}{p}}{2p+1}\right|
 \leq \frac{\binom{2p}{p}(2p^2+3p+1)}D .
 \tag{JQR3}
\]
Indeed, \(0<a_n<n\) and every absolute row sum of \(J_D\) is less than
\(2D\), unless the row sum is zero.  The operator norm bound follows
from the maximum absolute row sum for a real symmetric matrix.  The
diagonal matrix with entries \((-1)^i\), \(0\leq i<D\), conjugates
\(J_D\) to \(-J_D\), giving zero trace for every odd power, including
the first power.

Here is the bound for even powers, including the boundary rows.  Set
\(C_p=\binom{2p}{p}\).  There are exactly \(C_p\) sequences of \(2p\)
nearest-neighbour steps with \(p\) positive and \(p\) negative steps.
Starting at a coordinate \(i\) with \(p\leq i\leq D-1-p\), all these
closed walks stay in the matrix.  An edge used by any such walk has
index \(n\) satisfying \(|n-i|\leq p\).  The elementary inequalities
\[
 n-\tfrac12\leq a_n\leq n,\qquad
 \left|\frac{a_n}{D}-\frac{i}{D}\right|
 \leq\frac{p+\tfrac12}{D}
 \tag{JQR4}
\]
follow by squaring the lower inequality and using \(n\geq1\).
Every factor in a walk product and every comparison factor \(i/D\)
belongs to \([0,1]\).  The telescoping identity for a difference of
products therefore bounds the difference between a walk product and
\((i/D)^{2p}\) by \(p(2p+1)/D\).
There are at most \(2p\) remaining, boundary starting coordinates.
At one such coordinate, both the actual sum of valid walk products
and \(C_p(i/D)^{2p}\) lie in \([0,C_p]\); their difference in absolute
value is at most \(C_p\).  This assertion still applies when
\(D\leq2p\), in which case every row may be a boundary row.  Consequently,
\[
 \left|\frac1D\operatorname{tr}A_D^{\,2p}
       -\frac{C_p}{D}\sum_{i=0}^{D-1}(i/D)^{2p}\right|
 \leq\frac{C_p(2p^2+3p)}D.
 \tag{JQR5}
\]
The increasing function \(s^{2p}\) has its left Riemann sum between
its integral minus \(1/D\) and its integral.  Its integral on
\([0,1]\) is \(1/(2p+1)\).  Combining this fact with (JQR5) proves
(JQR3).

\paragraph{The limiting measure and an explicit finite discrepancy.}
Define a probability measure \(\nu\) as the image of
\(ds\,d\theta/\pi\) on \([0,1]\times[0,\pi]\) under
\((s,\theta)\mapsto2s\cos\theta\).  For fixed \(s>0\), its conditional
density is
\(\mathbf1_{\{|x|<2s\}}/(\pi\sqrt{4s^2-x^2})\).
Integrating in \(s\) gives
\[
 d\nu(x)=
 \frac1{2\pi}\operatorname{arcosh}\!\left(\frac2{|x|}\right)
 \mathbf1_{\{0<|x|<2\}}\,dx.
 \tag{JQR6}
\]
There is no atom at zero or at either endpoint.  The integral defining
the probability measure also proves that this logarithmically singular
density is integrable.  Its odd moments vanish, and
\[
 \int x^{2p}\,d\nu(x)=\frac{\binom{2p}{p}}{2p+1},\qquad p\geq0.
 \tag{JQR7}
\]
For completeness, integration in \(s\) supplies \(1/(2p+1)\);
the average of \((2\cos\theta)^{2p}\) on \([0,\pi]\) is
\(\binom{2p}{p}\).  The latter follows by taking the constant coefficient
of \((e^{i\theta}+e^{-i\theta})^{2p}\) over a full period; the integrand
has period \(\pi\).

For every integer \(L\geq1\), put
\[
 \omega_{D,L}=\frac4{\sqrt L}
               +\frac{2(L+1)^2\,2^L}{D}.
 \tag{JQR8}
\]
For every real Lipschitz function \(f\) on \([-2,2]\), with Lipschitz
constant \(\operatorname{Lip}(f)\), one has the fully finite inequality
\[
 \left|\int f\,d\mu_D-\int f\,d\nu\right|
 \leq \operatorname{Lip}(f)\,\omega_{D,L}.
 \tag{JQR9}
\]
Here is a proof that does not assume a limiting spectral theorem.
Replace \(f\) by \(f-f(0)\), which leaves the difference of integrals
unchanged and gives \(\|f\|_\infty\leq2\operatorname{Lip}(f)\).
Its Bernstein polynomial on \([-2,2]\) is
\[
 P_Lf(x)=\sum_{j=0}^L f(4j/L-2)\binom Lj
       \left(\frac{x+2}{4}\right)^j
       \left(\frac{2-x}{4}\right)^{L-j}.
 \tag{JQR10}
\]
At \(x=4t-2\), this is the expectation of \(f(4X/L-2)\) for
\(X\) binomial with parameters \(L,t\).  Cauchy--Schwarz and
\(\operatorname{Var}(X/L)=t(1-t)/L\) show that
\(\|P_Lf-f\|_\infty\leq2\operatorname{Lip}(f)/\sqrt L\).

Write \(P_Lf=\sum_{h=0}^L b_hx^h\).
For polynomials, the coefficient expression
\(\sum_h|b_h|2^h\) is submultiplicative.  Each of the two linear factors
\((x+2)/4\), \((2-x)/4\) has this expression equal to one.
The binomial expansion in (JQR10) therefore gives
\[
 \sum_{h=0}^L|b_h|2^h\leq2^L\|f\|_\infty.
 \tag{JQR11}
\]
For each nonconstant even \(h=2p\leq L\), (JQR3) bounds the moment
error by \((h+1)^2\,2^h/D\), since
\(\binom{2p}{p}\leq2^{2p}\) and
\(2p^2+3p+1\leq(2p+1)^2\).  The constant and odd errors are zero.
Thus the polynomial integral error is at most
\((L+1)^2\,2^L\|f\|_\infty/D\).
Adding the two Bernstein approximation errors proves (JQR9).
In particular, for \(D\geq4\), the explicit choice
\[
 L_D=\left\lfloor\tfrac12\log_2D\right\rfloor,\qquad
 \omega_D=\omega_{D,L_D}
 \tag{JQR12}
\]
has \(\omega_D\to0\).  For example, its second term is at most
\(2(L_D+1)^2/\sqrt D\).  These estimates apply to the moving dimension
\(D\) itself; no other packet or matrix dimension has been fixed.

\paragraph{The absolute-value distribution and its middle cutoff.}
Let \(F(t)=\nu(\{|x|\leq t\})\).  Extend \(F\) by zero to \(t<0\)
and by one to \(t>2\).  On \([0,2]\) it is
\[
 F(t)=\frac{t\operatorname{arcosh}(2/t)+2\arcsin(t/2)}{\pi},
 \qquad F(0)=0.
 \tag{JQR13}
\]
The formula follows by integrating the density in (JQR6);
differentiation on \((0,2)\) gives
\(F'(t)=\operatorname{arcosh}(2/t)/\pi>0\).
The endpoint values are \(F(0)=0\), \(F(2)=1\), using
\(t\operatorname{arcosh}(2/t)\to0\) at zero.
There is consequently a unique \(a\in(0,2)\) such that
\[
 a\operatorname{arcosh}(2/a)+2\arcsin(a/2)=\frac{\pi}{2}.
 \tag{JQR14}
\]
This number is independent of \(D,l,r\).

Set \(F_D(t)=\mu_D(\{|x|\leq t\})\).  For every \(t\in\mathbb R\)
and every \(\eta>0\), the finite Lipschitz estimate implies
\[
 F(t-\eta)-\frac{\omega_{D,L}}{\eta}
 \leq F_D(t)
 \leq F(t+\eta)+\frac{\omega_{D,L}}{\eta}.
 \tag{JQR15}
\]
To prove the upper bound, majorize the indicator of \(|x|\leq t\)
by a function equal to one up to \(t\), decreasing linearly to zero
between \(t\) and \(t+\eta\), and zero thereafter, as a function of
\(|x|\).  This function has Lipschitz constant at most \(1/\eta\);
its \(\nu\)-integral is at most \(F(t+\eta)\).  The lower bound uses
the corresponding minorant between \(t-\eta\) and \(t\).  When a
cutoff falls outside \([0,2]\), the same clipped ramp argument applies.

For the remaining finite estimates, fix
\[
 c_*=\frac a2,\qquad
 H_{c_*}=\max\{|\log c_*|,\log2\},\qquad
 \theta=\frac rD,\qquad
 I_{\mathrm{top}}=\frac1\pi
       \int_a^2\log x\,\operatorname{arcosh}(2/x)\,dx.
 \tag{JQR16}
\]
Suppose \(1\leq r\leq D\) and that the following explicit finite
inequality holds for some \(L\geq1\):
\[
 F(3a/4)+\frac{4\omega_{D,L}}a<1-\frac rD.
 \tag{JQR17}
\]
Applying (JQR15) with \(t=c_*,\eta=a/4\) proves
\(F_D(c_*)<1-r/D\).  Hence more than \(r\) eigenvalues, counted with
multiplicity, have \(|\lambda_j(A_D)|>c_*\); in particular all the
largest \(r\) absolute values do.  A weak inequality in the counting
statement would suffice, but the strict finite guard (JQR17)
avoids any endpoint convention.

\paragraph{A finite upper-order logarithmic sum.}
Order the absolute eigenvalues as
\(y_{D,1}\geq\cdots\geq y_{D,D}\geq0\).
Under (JQR17),
\[
 \left|\frac1D\sum_{j=1}^r\log y_{D,j}-I_{\mathrm{top}}\right|
 \leq \frac{\omega_{D,L}}{c_*}
       +H_{c_*}\left|\frac rD-\frac12\right|.
 \tag{JQR18}
\]
This estimate follows from an exact variational formula for an ordered
finite sum, which we prove here.  Let
\(g_{c_*}(x)=\log\max(c_*,|x|)\), and, for any probability measure \(\zeta\)
on \([-2,2]\), set
\[
 {\cal Q}_\theta(\zeta)
 =\inf_{u\in[\log c_*,\log2]}
       \left\{\theta u+\int(g_{c_*}(x)-u)_+\,d\zeta(x)\right\}.
 \tag{JQR19}
\]
For \(\zeta=\mu_D\) and \(\theta=r/D\), it equals
\(D^{-1}\sum_{j=1}^r g_{c_*}(y_{D,j})\).
Indeed, at any \(u\) between the \(r\)-th and the \((r+1)\)-st
ordered values of \(g_{c_*}\), the terms above \(u\) give exactly this
sum after addition of \(ru/D\).  At any other \(u\), including ties,
one has
\[
 ru+\sum_{j=1}^D(g_{c_*}(y_{D,j})-u)_+
 \ \geq\ \sum_{j=1}^r g_{c_*}(y_{D,j}),
 \tag{JQR20}
\]
by retaining only the first \(r\) summands and using
\((v-u)_+\geq v-u\).  The endpoint cases \(r=0,D\) follow by
taking \(u=\log2,\log c_*\), respectively.

For every \(u\), the positive-part function in (JQR19) has Lipschitz
constant at most \(1/c_*\).  Equation (JQR9) and the elementary bound
for the difference of two infima therefore give
\[
 |{\cal Q}_\theta(\mu_D)-{\cal Q}_\theta(\nu)|
       \leq\omega_{D,L}/c_*,\qquad
 |{\cal Q}_\theta(\nu)-{\cal Q}_{1/2}(\nu)|
       \leq H_{c_*}|\theta-\tfrac12|.
 \tag{JQR21}
\]
The second bound uses \(|u|\leq H_{c_*}\) throughout the infimum.
Since \(F(a)=1/2\) and \(\nu\) has no atom at \(a\) or \(-a\),
the minimizing cutoff in (JQR19) for \(\nu,\theta=1/2\) is
\(u=\log a\).  To verify it directly, the derivative in \(u\) of
the expression there, wherever needed away from its only possible
atom at \(\log c_*\), is
\(1/2-\nu(g_{c_*}>u)\); its sign changes from negative to positive
at \(\log a\).  Alternatively, integrating (JQR20)'s pointwise
inequality on the set \(|x|\geq a\), of measure \(1/2\), gives the
same minimum without differentiation.  Its value is exactly
\(\int_{|x|\geq a}\log|x|\,d\nu(x)=I_{\mathrm{top}}\).
Under (JQR17), the selected finite values also satisfy
\(g_{c_*}(y_{D,j})=\log y_{D,j}\).  This proves (JQR18).

\paragraph{The complete moving product, with a finite error.}
For every \(D,l,r,L\) in the preceding finite domains satisfying
(JQR17), one has
\[
 \begin{split}
 \left|\log\Pi^+_{D,l,r}
       -2lr\log D-2lD I_{\mathrm{top}}\right|
 &\leq 2lD\left\{\frac{\omega_{D,L}}{c_*}+
           H_{c_*}\left|\frac rD-\frac12\right|\right\}
       +\frac{rB_l}{D^2c_*^2}.
 \end{split}
 \tag{JQR22}
\]
No factor of \(\Phi_l\) has been deleted in this estimate.
For a selected absolute eigenvalue \(y\geq c_*\), its exact contribution
is
\[
 \log\Phi_l(Dy)
 =2l\log D+2l\log y+
    \sum_{j=0}^{l-1}\log\left(1+\frac{t_j^2}{D^2y^2}\right).
 \tag{JQR23}
\]
The final sum is nonnegative and at most \(B_l/(D^2c_*^2)\), by
\(\log(1+s)\leq s\) for \(s\geq0\).  Summing (JQR23) and applying
(JQR18) proves (JQR22).

In particular, along any sequence with
\[
 D\longrightarrow\infty,\qquad
 l\geq1,\qquad \frac lD\longrightarrow0,\qquad
 1\leq r\leq D,\qquad \frac rD\longrightarrow\frac12,
 \tag{JQR24}
\]
the guard (JQR17) eventually holds with \(L=L_D\), because
\(F(3a/4)<F(a)=1/2\).
Dividing (JQR22) by \(2lD\), its last term becomes
\(\theta B_l/(2lD^2c_*^2)\), which tends to zero since
\(B_l/l=(16l^2-12l-1)/12\).
Thus
\[
 \boxed{\displaystyle
 \log\Pi^+_{D,l,r}
     =2lr\log D+2lD I_{\mathrm{top}}+o(lD).}
 \tag{JQR25}
\]
The error conclusion follows from explicit bounds in (JQR22), not
from an asymptotic theorem at fixed \(l\), fixed \(r\), or fixed \(D\).

\paragraph{Application of the domains at the two upper source degrees.}
For the original subsequence \(k=4l+1\), fix an integer \(m\geq1\)
and retain the program's degree
\[
 q=(k+1)^2[1+k(m-1)].
 \tag{JQR26}
\]
For \(N=2q+\varepsilon\), with
\(\varepsilon\in\{-1,0\}\), the matrix dimension and relation rank
in the delegated comparison are exactly
\[
 D=N+l+1=2q+l+\varepsilon+1,\qquad
 r=N+1-q=q+\varepsilon+1,\qquad
 \frac rD-\frac12=\frac{\varepsilon+1-l}{2D}.
 \tag{JQR27}
\]
They satisfy (JQR24), because \(q\geq(k+1)^2\) and
\(l=(k-1)/4\).  Equation (JQR25) therefore applies separately to
both original upper source degrees.  It applies also to the top
\(D-r\) eigenvalues, since their rank fraction tends to \(1/2\).
This statement is about the given finite matrix spectrum.  The map
from the original relation subspace into this matrix and the determinant
comparison for that subspace must be retained in its surrounding
calculation; no subspace identification is being inferred from a
matching rank.

\paragraph{An exact full-product calculation and the lower product.}
There is also an elementary finite identity for the full determinant
that avoids a logarithm at a zero eigenvalue of \(J_D\).
Let \(b_n(y)\) be the monic polynomials defined by
\[
 b_0(y)=1,\quad b_1(y)=y,\quad
 b_{n+1}(y)=y b_n(y)-n(n-\tfrac12)b_{n-1}(y)\quad(n\geq1).
 \tag{JQR28}
\]
Expansion of the tridiagonal determinant along its last row proves
that \(b_D(y)=\det(yI_D-J_D)\).  As a formal power series in \(z\),
the recurrence implies
\[
 B(y,z)=\sum_{n\geq0}\frac{b_n(y)}{n!}z^n
  =(1+iz)^{-1/4-iy/2}(1-iz)^{-1/4+iy/2}.
 \tag{JQR29}
\]
To check every sign, multiply the recurrence by \(z^n/n!\) and
sum.  One obtains
\((1+z^2)\partial_z B=(y-z/2)B\), with \(B(y,0)=1\).
The right side of (JQR29) solves this equation with this initial
coefficient, so equality follows recursively for every coefficient.
All powers here are their formal series with constant term one.

At the original factor \(y=it_j=i(2j+1/2)\), this gives exactly
\[
 B(it_j,z)=(1+iz)^j(1-iz)^{-j-1/2},\qquad
 b_D(it_j)=i^D D!\,A_{D,j},
 \quad
 A_{D,j}=\sum_{s=0}^{\min(j,D)}
   \binom js\frac{(j+\tfrac12)_{D-s}}{(D-s)!}.
 \tag{JQR30}
\]
Here \((u)_n=u(u+1)\cdots(u+n-1)\), with \((u)_0=1\).
Every coefficient in \(A_{D,j}\) is positive, and the following
finite bounds hold for \(D\geq1\), \(0\leq j<l\):
\[
 \frac1{2D+1}\leq A_{D,j}\leq[2(D+l)]^j.
 \tag{JQR31}
\]
For the lower bound, retain the \(s=0\) term and use
\[
 \frac{(j+\tfrac12)_D}{D!}\geq
 \frac{(\tfrac12)_D}{D!}
 =4^{-D}\binom{2D}{D}\geq\frac1{2D+1}.
 \tag{JQR32}
\]
The last inequality follows because the central binomial coefficient
is the largest of the \(2D+1\) nonnegative coefficients summing to
\(4^D\); the ratio of consecutive coefficients verifies maximality.
For the upper bound, for \(0\leq n\leq D\),
\[
 \frac{(j+\tfrac12)_n}{n!}
 \leq\frac{(j+1)_n}{n!}=\binom{n+j}{j}
 \leq(D+j)^j.
 \tag{JQR33}
\]
The last inequality holds also at \(j=0\), when both sides are one;
for \(j\geq1\), write the binomial coefficient as a product of \(j\)
factors at most \(D+j\), divided by \(j!\geq1\).
Summing \(\binom js\) over the actual truncated range gives at most
\(2^j\), proving (JQR31).

Since \(J_D\) has real eigenvalues,
\(\det(J_D^2+t_j^2I_D)=|b_D(it_j)|^2\).
Equations (JQR30)--(JQR31), with all \(l\) factors present, now give
\[
 \begin{split}
 \log\det\Phi_l(J_D)
   &=2l\log(D!)+2\sum_{j=0}^{l-1}\log A_{D,j},\\
 2l\log(D!)-2l\log(2D+1)
   &\leq\log\det\Phi_l(J_D)\\
   &\leq2l\log(D!)+l(l-1)\log[2(D+l)].
 \end{split}
 \tag{JQR34}
\]
For explicit finite remainders, monotonicity of \(\log x\) gives
\[
 D\log D-D+1\leq\log(D!)
       \leq D\log D-D+1+\log D.
 \tag{JQR35}
\]
Consequently the full determinant differs from
\(2lD(\log D-1)\) by a quantity between
\(2l-2l\log(2D+1)\) and
\(2l(1+\log D)+l(l-1)\log[2(D+l)]\).
In addition to (JQR24), the sufficient condition for these full
remainders to be \(o(lD)\) is
\[
 \frac{l\log[2(D+l)]}{D}\longrightarrow0.
 \tag{JQR36}
\]
This stronger condition holds at each of the actual dimensions
(JQR27), where \(D\geq2(k+1)^2\) and \(l=(k-1)/4\).
It is not silently replaced by the weaker condition \(l/D\to0\).

Let \(\Pi^-_{D,l,r}\) denote the product of the smallest \(r\)
eigenvalues of \(\Phi_l(J_D)\).
The exact identity
\(\Pi^-_{D,l,r}\Pi^+_{D,l,D-r}=\det\Phi_l(J_D)\),
combined with (JQR25), (JQR34)--(JQR36), proves at the actual
dimensions, and more generally under (JQR24) and (JQR36),
\[
 \boxed{\displaystyle
 \log\Pi^-_{D,l,r}
   =2lr\log D-2lD(1+I_{\mathrm{top}})+o(lD).}
 \tag{JQR37}
\]
This deduction retains possible zero eigenvalues of \(J_D\) inside
the strictly positive factors \(J_D^2+t_j^2I_D\).
For the completely finite version, assume the guard (JQR17) with
rank \(D-r\), and define
\[
 E^-_{D,l,r,L}
 =2lD\left\{\frac{\omega_{D,L}}{c_*}+
           H_{c_*}\left|\frac rD-\frac12\right|\right\}
       +\frac{(D-r)B_l}{D^2c_*^2}.
 \tag{JQR38}
\]
Subtracting the upper-rank estimate (JQR22), with rank \(D-r\),
from the two finite bounds following (JQR35) gives
\[
 \begin{split}
 2l-2l\log(2D+1)-E^-_{D,l,r,L}
 &\leq
 \log\Pi^-_{D,l,r}
    -\{2lr\log D-2lD(1+I_{\mathrm{top}})\}\\
 &\leq
 2l(1+\log D)+l(l-1)\log[2(D+l)]+E^-_{D,l,r,L}.
 \end{split}
 \tag{JQR39}
\]
Thus both upper and lower spectral products have explicit finite
enclosures.  No lower-tail logarithmic approximation is needed.
