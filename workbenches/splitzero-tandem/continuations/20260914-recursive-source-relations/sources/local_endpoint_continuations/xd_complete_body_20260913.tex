\section{Exact determinant of the polynomial--exponential period comparison}
\label{sec:exponential-period-determinant-exact}

Fix the original monic cyclic polynomial
\[
 \chi(S)=S^q+\sum_{a=0}^{q-1}c_aS^a,\qquad q\geq1,
 \quad d=q+1,
 \quad\Phi_t(S)=\frac{S^d}{d}
       +\sum_{a=0}^{q-1}\frac{c_aS^{a+1}}{a+1}-tS.
\]
Fix $u\in\mathbb C^*$ and one argument of $u$. All the following
comparisons retain this $u$, the original coordinate $S$, and the
ordered remainder basis $1,S,\ldots,S^{q-1}$. Set
\[
 D_{u,t}P=uP'+(\chi-t)P,
 \qquad H_{u,t}=\mathbb C[S]/D_{u,t}\mathbb C[S].
\]
This is a quotient of coefficient vector spaces. Let
$\operatorname{red}_{u,t}$ be its unique representative of degree
less than $q$. Let $A(t)$ be the multiplication matrix of $S$ on
$\mathbb C[S]/(\chi-t)$ in the same ordered basis, so
\[
 A(t)e_b=e_{b+1}\quad(0\leq b<q-1),\qquad
 A(t)e_{q-1}=(-c_0+t)e_0-\sum_{a=1}^{q-1}c_ae_a.
 \tag{XD1}
\]
The empty sum convention in this display retains the case $q=1$.

Use precisely the oriented rays and differences
\[
 \theta_j=\frac{\arg u+\pi}{d}+\frac{2\pi j}{d},\quad
 \ell_j:\ S=re^{i\theta_j},\ 0\leq r<\infty,
 \quad\Gamma_j=-\ell_0+\ell_j\quad(1\leq j\leq q).
 \tag{XD2}
\]
Write
\[
 \Pi_{jb}(u,c,t)=\int_{\Gamma_j}e^{\Phi_t(S)/u}S^b\,dS,
 \qquad 1\leq j\leq q,\quad0\leq b<q.
 \tag{XD3}
\]

\subsection{Uniform estimates on the fixed unbounded contours}

For a compact set $K\subset\mathbb C^{q+1}$ of the parameters
$(c_0,\ldots,c_{q-1},t)$, put
\[
 M_K=\sum_{a=0}^{q-1}\frac{\sup_K|c_a|}{a+1}+\sup_K|t|,
 \quad R_K=\max\{1,2dM_K\},\quad\alpha=\frac1{2d|u|}.
 \tag{XD4}
\]
For every ray in (XD2) its leading term satisfies
$S^d/(du)=-r^d/(d|u|)$. For $r\geq1$ every other exponent
has degree at most $d-1$, so
\[
 \operatorname{Re}\frac{\Phi_t(re^{i\theta_j})}{u}
 \leq-\frac{r^d}{d|u|}+\frac{M_Kr^{d-1}}{|u|}
 \leq-\alpha r^d\qquad(r\geq R_K).
 \tag{XD5}
\]
Every parameter derivative of the integrand is its original exponential
times a polynomial in $S$: a derivative in $c_a$ contributes
$S^{a+1}/((a+1)u)$ and a derivative in $t$ contributes $-S/u$.
For any fixed derivative order its absolute value on the tail is
bounded by $Cr^L e^{-\alpha r^d}$, with $C,L$ uniform on $K$.
The required majorant is integrable, since substitution
$x=\alpha r^d$ gives
\[
 \int_0^\infty r^L e^{-\alpha r^d}\,dr
 =\frac1d\alpha^{-(L+1)/d}
    \Gamma\!\left(\frac{L+1}{d}\right)<\infty.
 \tag{XD6}
\]
The remaining finite interval is uniformly bounded by continuity.
Thus (XD3) converges absolutely and is entire in all the lower
coefficients and $t$; every parameter derivative may be taken under
the integral. The same bound gives
$e^{\Phi_t(S)/u}P(S)\longrightarrow0$ at the infinite endpoint
of each ray, for every polynomial $P$. Consequently
\[
 \int_{\Gamma_j}e^{\Phi_t/u}D_{u,t}P\,dS
 =u\int_{\Gamma_j}d(e^{\Phi_t/u}P)=0.
 \tag{XD7}
\]
Indeed the integral of that derivative on each outward ray is
$-e^{\Phi_t(0)/u}P(0)=-P(0)$; the coefficients $-1,+1$ of
the two rays cancel these values. These estimates handle the
noncompact contours directly, including parameter values at which
critical points coincide.

\subsection{An exact trace comparison for twisted reduction}

For $1\leq m\leq q$ and $0\leq b<q$, ordinary monic division gives
unique polynomials $Q_{m,b},R_{m,b}$ satisfying
\[
 S^{m+b}=(\chi-t)Q_{m,b}+R_{m,b},\qquad
 \deg R_{m,b}<q.
 \tag{XD8}
\]
If $m+b<q$, take $Q_{m,b}=0$. Otherwise
$\deg Q_{m,b}=m+b-q\leq q-1$, and the differential itself gives
\[
 \operatorname{red}_{u,t}(S^{m+b})
   =R_{m,b}-uQ_{m,b}'.
 \tag{XD9}
\]
Both terms on the right have degree below $q$. In every nonzero
case,
\[
 \deg Q_{m,b}'\leq m+b-q-1\leq b-1.
 \tag{XD10}
\]
Thus the coefficient of $S^b$ in the derivative correction vanishes.
Let $T_m(u,c,t)$ have column $b$ equal to the coordinates of the
left side of (XD9). The columns $R_{m,b}$ are exactly those of
$A(t)^m$. Although (XD9) retains a possibly nonzero correction
matrix, (XD10) proves the exact trace identity
\[
 \operatorname{Tr}T_m(u,c,t)=\operatorname{Tr}A(t)^m
       \qquad(1\leq m\leq q).
 \tag{XD11}
\]

Set $B_a=T_{a+1}/((a+1)u)$ for $0\leq a<q$. Differentiation
of (XD3), followed by (XD7)--(XD9), now proves
\[
 \partial_{c_a}\Pi=\Pi B_a,\qquad
 \partial_t\Pi=-\Pi A(t)/u,
 \quad\operatorname{Tr}B_a
  =\frac{\operatorname{Tr}A(t)^{a+1}}{(a+1)u}.
 \tag{XD12}
\]
All the coefficient matrices here have polynomial entries in $c,t$
for this fixed nonzero $u$.

\subsection{The determinant, including every monomial factor}

Define the scalar polynomial in the original coefficients
\[
 \mathcal F(c,t)=\operatorname{Tr}\Phi_t(A(t))
 =\frac{\operatorname{Tr}A(t)^{q+1}}{q+1}
 +\sum_{a=0}^{q-1}\frac{c_a\operatorname{Tr}A(t)^{a+1}}{a+1}
 -t\operatorname{Tr}A(t).
 \tag{XD13}
\]
There is no choice or enumeration of roots in this definition.
For any differentiable matrix $A$ and integer $n\geq1$, the product
rule and cyclic invariance of trace give
\[
 d\operatorname{Tr}A^n
 =\sum_{r=0}^{n-1}\operatorname{Tr}(A^r(dA)A^{n-1-r})
 =n\operatorname{Tr}(A^{n-1}dA).
 \tag{XD14}
\]
Because $A(t)$ acts on the actual quotient by $\chi-t$, it satisfies
$\chi(A(t))-tI_q=0$. Applying (XD14) to (XD13) therefore gives
\[
 d\mathcal F
 =\sum_{a=0}^{q-1}
    \frac{\operatorname{Tr}A(t)^{a+1}}{a+1}\,dc_a
   -\operatorname{Tr}A(t)\,dt
   +\operatorname{Tr}\bigl((\chi(A(t))-tI_q)dA(t)\bigr)
 =\sum_{a=0}^{q-1}
    \frac{\operatorname{Tr}A(t)^{a+1}}{a+1}\,dc_a
   -\operatorname{Tr}A(t)\,dt.
 \tag{XD15}
\]

At $c=t=0$ the period matrix is
\[
 \Pi^{\mathrm{mon}}_{j,l-1}(u)
 =a_l(u)e^{il\theta_0}(\zeta^{jl}-1),\qquad
 a_l(u)=\frac{(d|u|)^{l/d}}d\Gamma(l/d),\quad
 \zeta=e^{2\pi i/d},\quad1\leq j,l\leq q.
 \tag{XD16}
\]
This follows by substituting $x=r^d/(d|u|)$ on each oriented ray
in (XD3). In particular all $a_l(u)$ are positive real numbers.
For $V_{jl}=\zeta^{jl}-1$ one may include $j=0$ in the Gram sum
because its summand is zero. The complete root-of-unity sums give
\[
 (V^*V)_{lm}
 =\sum_{j=0}^{d-1}
    (\zeta^{-jl}-1)(\zeta^{jm}-1)
 =d\delta_{lm}+d.
 \tag{XD17}
\]
Here $1\leq l,m<d$, so the two individual nonconstant sums vanish,
and the first sum is $d\delta_{lm}$. The all-ones rank-one matrix
has eigenvalue $q$ on the all-ones vector and zero on its
$(q-1)$-dimensional perpendicular complement. It follows that
\[
 \det(V^*V)=d^q(1+q)=d^d>0,
 \qquad
 C_q(u):=|\det\Pi^{\mathrm{mon}}(u)|^2
 =d^d\prod_{l=1}^{q}
       \left[\frac{(d|u|)^{l/d}}d\Gamma(l/d)\right]^2.
 \tag{XD18}
\]
The complex determinant itself retains the oriented phase:
$\det\Pi^{\mathrm{mon}}=\det V\prod_{l=1}^q
a_l(u)e^{il\theta_0}$.

We claim the exact global formula
\[
 \boxed{\det\Pi(u,c,t)=\det\Pi^{\mathrm{mon}}(u)
                 \exp\bigl(\mathcal F(c,t)/u\bigr).}
 \tag{XD19}
\]
To prove it without assuming invertibility along the path, join
$(0,0)$ to $(c,t)$ by $(sc,st)$, $0\leq s\leq1$.
Equations (XD12) give $\Pi'=\Pi B(s)$, with continuous $B(s)$,
and (XD15) gives $\operatorname{Tr}B(s)=\mathcal F'(sc,st)/u$.
For every square matrix, including a singular one,
\[
 \frac{d}{ds}\det\Pi
 =\operatorname{Tr}(\operatorname{adj}(\Pi)\Pi B)
 =\det\Pi\operatorname{Tr}B.
 \tag{XD20}
\]
Consequently $\det\Pi\,\exp(-\mathcal F/u)$ has derivative zero.
At the starting point $A$ is nilpotent and $\mathcal F(0,0)=0$,
so its value is $\det\Pi^{\mathrm{mon}}(u)$. This proves (XD19).
Equation (XD18) proves its nonvanishing for every $c,t$, including
every repeated-root fibre. In particular the period map
\[
 H_{u,t}\longrightarrow\mathbb C^q,
 \quad[P]\longmapsto\Pi\operatorname{red}_{u,t}(P),
 \tag{XD21}
\]
is an isomorphism with inverse given by $\Pi^{-1}$ followed by
the specified degree-below-$q$ section.

\subsection{Exact determinant transport to the arithmetic representative}

Retain the original positive Gram $G_N$ on the cyclic coefficient
space $E$, its determinant $V_N=\det G_N$, and the original maps
$r_N:E\longrightarrow\mathscr B^{\widehat\otimes k}$ and
$j_E$ on the admitted test-function domain with $j_Er_N=I_E$.
Identify $E$ with $H_{u,t}$ by their specified ordered remainder
bases. Then the actual maps
\[
 r_N(E)\longrightarrow\mathbb C^q,\quad x\longmapsto\Pi j_Ex,
 \qquad
 \mathbb C^q\longrightarrow r_N(E),\quad y\longmapsto r_N\Pi^{-1}y
 \tag{XD22}
\]
are inverse: their composites are respectively $r_Nj_E$ on
$r_N(E)$ and $\Pi(j_Er_N)\Pi^{-1}$ on $\mathbb C^q$.
On the full admitted domain the first observation has precisely
$\ker j_E$ as its kernel; all original theta boundaries killed by
$j_E$ remain in that kernel.

With the explicitly fixed coordinate metric $I_q$ on the target,
put
\[
 H_\Pi=\Pi^*\Pi,\qquad B_N^{\mathrm{per}}=G_N^{-1}H_\Pi.
 \tag{XD23}
\]
These operators satisfy
$G_NB_N^{\mathrm{per}}=(B_N^{\mathrm{per}})^*G_N=H_\Pi>0$.
Taking determinants in (XD19) gives the exact identities
\[
 \det H_\Pi=C_q(u)
       \exp\!\left(2\operatorname{Re}\frac{\mathcal F(c,t)}u\right),
 \tag{XD24}
\]
\[
 \boxed{\log\det B_N^{\mathrm{per}}
 =\log C_q(u)+2\operatorname{Re}\frac{\mathcal F(c,t)}u-\log V_N.}
 \tag{XD25}
\]
Every determinant in (XD24)--(XD25) is positive real; these
logarithms therefore require no branch choice. At the same $u,c,t$
for two admitted degrees $i\leq j$, subtraction proves
\[
 \boxed{\log\det B_j^{\mathrm{per}}-
        \log\det B_i^{\mathrm{per}}=\log(V_i/V_j).}
 \tag{XD26}
\]
Thus the common polynomial--exponential factor is now calculated
exactly while the full original endpoint cost is retained by the
displayed pair of positive comparison operators. No uniform upper
estimate on $V_i/V_j$ is inferred from this determinant identity.
