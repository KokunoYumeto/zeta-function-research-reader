\section{Original objects, adjoints and the parity map}
Use exactly the packet and measure in NG1:
\begin{equation}\tag{SD1}
\begin{gathered}
0<\delta<\tfrac12,\quad\gamma>2,\quad m_0\ge1,\quad
k\ge5,\quad k\equiv1\pmod4,\\
e=1+k(m_0-1),\quad q=e(k+1)^2,\quad c=k/2,\\
Q(y)=\prod_{a,b=0}^k
\bigl[y-(2b-k)\gamma+i(2a-k)\delta\bigr]^e,\\
\dd\mu_k(y)=w_h^{*k}(y)\dd y,\qquad
w_h(y)=|(2\xi/h)(1/2+iy)|^2/(2\pi).
\end{gathered}
\end{equation}
The stipulated full quartet divisor $h$, its arithmetic unit and all
multiplicities remain in this measure. The original coordinate is
$S=c+iy$; the map $\C[S]/(\chi)\to E=\C[y]/(Q)$ substitutes this exact
coordinate, with $Q=i^{-q}\chi(c+iy)$. The even degree $q$ is divisible
by four. The complete root multiset is closed under negation and
conjugation, has no real root, and lies in $|y|\le R_k$, where
$R_k=k\sqrt{\delta^2+\gamma^2}$. In particular $Q$ is positive on $\R$.

For each $q-1\le N\le2q-1$, let $J_N:\PP_{\le N}\to E$ be the
actual remainder map and $L_N:E\to\PP_{\le N}$ its attained minimum
section in $L^2(\mu_k)$. Its image $\mathcal V_N$ is the orthogonal
complement of $Q\PP_{<d}$, $d=N+1-q$. The quotient metric $G_N$ makes
$L_N$ an isometry. Let $M$ be multiplication by $y$ on $E$ and let
$C_N=L_N^{-1}P_{\mathcal V_N}yL_N$. Throughout, $\dagger$ means the
adjoint in this particular $G_N$, not a coefficient-space adjoint.
If $p_{N+1}$ is the real monic source orthogonal polynomial, NG21 gives
\begin{equation}\tag{SD2}
M=C_N+r_N\ell_N,\qquad r_N=[p_{N+1}],\qquad
\ell_N(v)=[y^N]L_Nv,\qquad C_N^\dagger=C_N.
\end{equation}
Indeed, subtract $\ell_N(v)p_{N+1}$ from $yL_Nv$ before its degree-$N$
projection, then take its remainder. This proves both the sign and the
map in SD2 in the original metric.

The even original measure makes $\Pi[P(y)]=[P(-y)]$ a unitary involution.
It preserves the remainder map, its minimum section, and the relation
space. Thus
\begin{equation}\tag{SD3}
\Pi C_N\Pi=-C_N,\qquad
\Pi r_N=(-1)^{N+1}r_N,\qquad
\ell_N\Pi=(-1)^N\ell_N,\qquad \ell_Nr_N=0.
\end{equation}
Write $v_N=\ell_N^\dagger$ and
$\kappa_N=\norm{r_N}_{G_N}\norm{v_N}_{G_N}=\norm{r_N\ell_N}_{G_N}$.
Both factors are nonzero. A zero remainder of $p_j$ would make the
nonreal roots of $Q$ roots of a source orthogonal polynomial, whose
roots are real. Moreover, the remainder of $p_N$ is nonzero and its
minimum section has a nonzero degree-$N$ coefficient: the identity
$\langle[p_N],x\rangle_{G_N}=\omega_N[y^N]L_Nx$ proves this directly.
The real-root property follows from the elementary orthogonality
argument: the product of the real sign-changing factors, if of degree
less than $j$, would have a nonzero integral against $p_j$. This is the
standard orthogonal-polynomial fact recorded by Koornwinder and
coauthors \cite{DLMF}; its hypotheses hold for the full measure here.

\section{An exact determinant with the full roots and boundary vectors}
For $a>0$ define the following actual positive operators and scalars:
\begin{equation}\tag{SD4}
D_a=C_N^2+a^2I,\qquad
A_a=r_N^\dagger D_a^{-1}r_N,\qquad
B_a=\ell_ND_a^{-1}\ell_N^\dagger,\qquad
p_N^C(z)=\det(zI-C_N).
\end{equation}
The symbol $p_N^C$ is a characteristic polynomial of degree $q$, and
is distinct from the source polynomial $p_{N+1}$.

\begin{theorem}[Original regularized determinant]
For every cutoff and every $a>0$,
\begin{equation}\tag{SD5}
\boxed{\det(M^\dagger M+a^2I)
=|Q(ia)|^2+a^2|p_N^C(ia)|^2 A_aB_a.}
\end{equation}
Every term is computed in the original attained metric and every root
and repeated factor of $Q$ remains in the first term.
\end{theorem}
\begin{proof}
For this proof write $C=C_N$, $r=r_N$, $v=\ell_N^\dagger$ and
$h=v^\dagger D_a^{-1}Cr$. The difference between $M^\dagger M+a^2I$
and $D_a$ is
$Crv^\dagger+vr^\dagger C+\norm r^2vv^\dagger$.
The determinant identity $\det(I+UV)=\det(I+VU)$ follows by eliminating
either diagonal block of $\left(\begin{smallmatrix}I&U\\-V&I\end{smallmatrix}\right)$.
Apply it with columns $Cr,v$ and the two-by-two coefficient matrix
$\left(\begin{smallmatrix}0&1\\1&\norm r^2\end{smallmatrix}\right)$.
Expanding that two-by-two determinant gives
\[
\frac{\det(M^\dagger M+a^2I)}{\det D_a}
=|1+h|^2+
\bigl(\norm r^2-r^\dagger CD_a^{-1}Cr\bigr)B_a
=|1+h|^2+a^2A_aB_a.
\]
There has been no estimate or discarded term in this expansion.
Parity gives $v^\dagger D_a^{-1}r=0$, because $D_a$ preserves the two
opposite parity spaces in SD3. The rank-one determinant formula,
proved by the same block elimination, therefore gives
\[
\frac{Q(ia)}{p_N^C(ia)}
=1-v^\dagger(iaI-C)^{-1}r
=1+v^\dagger(C+iaI)D_a^{-1}r=1+h.
\]
Finally $\det D_a=|p_N^C(ia)|^2$, since $C$ is selfadjoint. Substitute
these two equalities in the preceding expansion to prove SD5.
\end{proof}

This formula also identifies the contribution at zero exactly:
\begin{equation}\tag{SD6}
\det(M^\dagger M)=|Q(0)|^2
=\prod_{a,b=0}^k\bigl[(2b-k)^2\gamma^2+(2a-k)^2\delta^2\bigr]^e.
\end{equation}
Here one can take determinants of $M$ directly; no invertibility of
$C_N$ at zero is assumed.

\section{The finite logarithmic error and the action receiver}
Put $a=q\sqrt t$, for an arbitrary fixed $t>0$, and define
\begin{equation}\tag{SD7}
\begin{gathered}
F_N(t)=\log\det(I+M^\dagger M/(tq^2)),\qquad
L_N^{\rm sp}(t)=\log\det(I+C_N^2/(tq^2)),\\
\alpha_N(t)=q^2 A_a/\norm{r_N}^2,\quad
\beta_N(t)=q^2 B_a/\norm{\ell_N^\dagger}^2,\qquad
\zeta_N(t)=\frac{|Q(ia)/p_N^C(ia)|^2}{a^2A_aB_a}.
\end{gathered}
\end{equation}
These definitions do not rescale the original metric. They specify the
arguments of dimensionless spectral tests, just as in NG23.
SD5 gives the complete finite equality
\begin{equation}\tag{SD8}
F_N(t)=L_N^{\rm sp}(t)+2\log\kappa_N-2\log q
+\log(t\alpha_N(t)\beta_N(t))+\log(1+\zeta_N(t)).
\end{equation}
Let $B=5\max\{1,C_{\rm mult}(h)\}$, where the original multiplication
constant is CAI22/NG3. The degree of $L_Nv$ is at most $2q-1$, so that
NG3 proves $\norm{C_N}\le Bq$ throughout this window. Consequently
\begin{equation}\tag{SD9}
\frac1{t+B^2}\le\alpha_N(t),\beta_N(t)\le\frac1t,
\qquad
0\le\zeta_N(t)\le
\frac{q^2}{\kappa_N^2}\frac{(t+B^2)^2}{t}
\exp\!\left(\frac{R_k^2}{tq}\right).
\end{equation}
For the last inequality, pair opposite roots of $Q$ to obtain
$|Q(ia)|\le a^q(1+R_k^2/a^2)^{q/2}$. The paired real eigenvalues of
$C_N$ give $|p_N^C(ia)|\ge a^q$, even if a zero eigenvalue occurs.
Finally $A_aB_a\ge\kappa_N^2/[q^4(t+B^2)^2]$. These prove SD9 with
the complete root polynomial, without a missing region or a root limit.

We now prove the uniform lower bound needed to evaluate this error.
The trace of SD2 squared, using $(r_N\ell_N)^2=0$, is
\begin{equation}\tag{SD10}
\Tr M^2=\Tr C_N^2+2\ell_N C_Nr_N
=\frac{qk(k+2)}3(\gamma^2-\delta^2).
\end{equation}
The plus sign here follows from SD2. Changing the convention to
$M=C_N-\mathsf R_N$ changes the named rank-one vector accordingly.
The last equality sums the squared full roots in SD1; the mixed term
vanishes and $(k+1)^{-1}\sum_{a=0}^k(2a-k)^2=k(k+2)/3$.

NG20 and GM18--20 prove, if $d/q\to s\in(0,1]$,
\[
q^{-3}\Tr C_N^2\longrightarrow\mathfrak m(s)>1/24.
\]
The same limit at $s=0$ holds even when $d\to\infty$ and $d/q\to0$.
Indeed the source compression has dimension $q+d$ and uniformly
bounded spectrum after division by $q$. Its relation compression has
dimension $d$, so its divided second moment is $O(d/q)$. The rank-two
off-diagonal comparison in NG19--20 has divided error $O(1/q)$ for
the squared test on that fixed compact interval. The source moment
therefore tends to $\mathfrak m(0)=2/3$. The explicit GM18 formula
is continuous at zero and on $[0,1]$. A contradiction subsequence,
then a convergent subsequence of $d/q$, proves uniform convergence to
these moments over the full finite window. In particular, eventually
\[
\Tr C_N^2\ge q^3/48,\qquad |\Tr M^2|\le q^3/96
\]
for every cutoff. By SD10 and $|\ell_N C_Nr_N|\le Bq\kappa_N$,
\begin{equation}\tag{SD11}
\boxed{\kappa_N\ge q^2/(192B)}
\end{equation}
eventually uniformly in $N$. This uses the proved spectral limit, and
does not claim an effective numerical threshold in $k$.

The minimum--maximum characterization of singular values gives
\begin{equation}\tag{SD12}
|s_1(M)-\kappa_N|\le Bq,\qquad s_j(M)\le Bq\quad(j\ge2).
\end{equation}
The first statement is the two triangle inequalities for the operator
norm. For the second restrict the domain to $\ker\ell_N$ (codimension
one), where $M=C_N$, and use the singular-value variational formula.
Thus one singular value exceeds the scale $q$ by a factor tending to
infinity, and all the others are bounded by $Bq$. In particular
$\log s_1(M)^2=2\log\kappa_N+O_h(q^{-1})$ uniformly.

\section{The full root product at zero}
Here the repeated roots can be evaluated without using a metric limit.
Define
\begin{equation}\tag{SD15}
\mathfrak g(\delta,\gamma)=
\tfrac12\log(\delta^2+\gamma^2)-\tfrac32
+\tfrac12\left[\frac\delta\gamma\arctan\frac\gamma\delta
+\frac\gamma\delta\arctan\frac\delta\gamma\right].
\end{equation}
At every fixed original multiplicity,
\begin{equation}\tag{SD16}
\boxed{\log|Q(0)|=q\log k+q\mathfrak g(\delta,\gamma)
+O_h(q\log k/k).}
\end{equation}
To prove it, divide each root in SD1 by $k$. Its logarithmic modulus is
$f(x,y)=\tfrac12\log(\delta^2x^2+\gamma^2y^2)$ on the grid
$(x,y)=((2a-k)/k,(2b-k)/k)$. The empirical average of $f$ tends to
one quarter of its integral over $[-1,1]^2$, with error
$O_h(\log k/k)$. Here are bounds at its singularity. Voronoi cells
of this grid have diameter at most $3/k$. Outside the cells within
$6/k$ of the origin, $|\nabla f|\le C_h/(|x|+|y|)$; integration
of this bound against a cell diameter gives $O_h(1/k)$, since this
radial inverse is integrable in two dimensions. The finitely many
remaining cells have total error $O_h(\log k/k^2)$, by scaling them
by $k$ and integrating $|\log\sqrt{x^2+y^2}|$ on a bounded square.
No grid point is the origin, because $k$ is odd. The Voronoi weights
differ from the uniform weights $1/(k+1)^2$ by total variation
$O(1/k)$; all sampled absolute values are $O_h(\log k)$. This proves
the claimed error for the empirical average, including boundary cells.
Multiplication by the full $q=e(k+1)^2$ proves the error in SD16.

For the exact integral, evenness reduces it to $I=\int_0^1\int_0^1f$.
Homogeneity gives $xf_x+yf_y=1$ away from zero. Integrate
$\partial_x(xf)+\partial_y(yf)=2f+1$ on the square with a small
quarter disk removed. The new boundary term tends to zero as its
radius does, since it is $O(r^2|\log r|)$. Hence
$2I+1=\int_0^1f(1,y)\dd y+\int_0^1f(x,1)\dd x$.
Direct integration of the two logarithms gives SD15.

\section{The necessary small singular values in the original metric}
Write $\lambda_{j,N}=s_j(M)^2/q^2$ in decreasing order. Set
\[
\log^+x=\max\{\log x,0\},\qquad \log^-x=\max\{-\log x,0\}.
\]
Define the actual logarithmic loss
$\mathcal D_N=\sum_{j=1}^q\log^-\lambda_{j,N}$.
By SD6,
\begin{equation}\tag{SD17}
\sum_{j=1}^q\log\lambda_{j,N}=2\log|Q(0)|-2q\log q.
\end{equation}
The perturbation between $M^\dagger M$ and $C_N^2$ has rank at most
two: its image lies in the span of $C_Nr_N$ and $v_N$.
For Hermitian matrices differing by rank $r$, the eigenvalue counts
below any threshold differ by at most $r$: intersect the spectral
subspace with the kernel of their difference and apply the variational
principle, then interchange the matrices. This proves the count bound
here, including multiplicities and endpoint conventions.

After removing the one large singular value in SD12, all remaining
$\lambda_{j,N}$ lie in $[0,B^2]$. Apply the count bound to
$\min\{\log^+x,\log^+(B^2)\}$, whose variation is finite. Removing
the top singular value costs at most the constant $\log^+(B^2)$.
NG20, including the $s=0$
extension proved above, gives uniformly for $d/q\in[0,1]$
\begin{equation}\tag{SD18}
\frac1q\sum_{j=2}^q\log^+\lambda_{j,N}
=b(d/q)+o_h(1),\qquad
b(s)=\int\log^+(y^2)\dd\nu_s(y).
\end{equation}
The full density is the proved NG20/GM16 probability
\[
\dd\nu_s(y)=\left[h_0\left(\frac y{1+s}\right)
-s|y|\rho_s(y^2)\right]\dd y,\quad
h_0(y)=\frac1{2\pi}\operatorname{arcosh}(2/|y|)1_{0<|y|<2}.
\]
The exact $\rho_s$ and its elliptic endpoints are given and proved in
the complete NG/GM sources accompanying this note. The integral in
SD18 is finite; it is an explicit one-dimensional quadrature of that
proved density. Uniformity follows by the same subsequence argument
as SD11 and continuity of these compactly supported measures.

Use $\log^-x=\log^+x-\log x$ in SD17 and the top singular value in
SD12. This yields the evaluated logarithmic loss
\begin{equation}\tag{SD19}
\boxed{\begin{aligned}
\mathcal D_N={}&2\log\kappa_N+2(q-1)\log q-2q\log k
-2q\mathfrak g(\delta,\gamma)+qb(d/q)+o_h(q).
\end{aligned}}
\end{equation}
The error is uniform over the original window. Unlike a weak spectral
limit applied to $\log x$ at zero, this calculation uses the exact
full determinant at zero, and uses the spectral limit only on the
bounded positive part. There is no assumption of uniform integrability
of the negative logarithm.

There is also an entirely finite bound for the smallest singular value.
For $0<\eta<1$, let
\begin{equation}\tag{SD20}
 m_{N,\eta}=\min\{q,\,2+\#\{j:|\lambda_j(C_N)|<q\sqrt\eta\}\}.
\end{equation}
If $q>R_k/\sqrt\eta$, the exact finite formula is
\begin{equation}\tag{SD21}
\boxed{\frac{s_q(M)^2}{q^2}
\le\eta\exp\!\left(
\frac{2\log|Q(0)|-2q\log q-q\log\eta}{m_{N,\eta}}
\right).}
\end{equation}
This displayed guard is a direct check on specified finite data, not
an additional unproved analytic estimate. To verify the formula, put
$n=\#\{j:\lambda_{j,N}<\eta\}$. The determinant and the guard give
$\sum\log\lambda_{j,N}<q\log\eta$, so $n\ge1$. The count bound
gives $n\le m_{N,\eta}$. Since all remaining values are at least
$\eta$, SD17 is at least $n\log\lambda_{q,N}+(q-n)\log\eta$.
Solving for $\log\lambda_{q,N}$ and replacing $n$ by its upper bound
in the resulting negative fraction proves SD21.

\begin{theorem}[Uniform decay below every inverse power]
For every fixed $A>0$, the original attained metrics satisfy
\begin{equation}\tag{SD22}
\boxed{\max_{q-1\le N\le2q-1}s_q(M;G_N)/q\le q^{-A}}
\end{equation}
for all sufficiently large admissible $k$, at every fixed $m_0$.
Every algebraic eigenvalue of this same $M$ has modulus at least
$\sqrt{\delta^2+\gamma^2}>2$.
\end{theorem}
\begin{proof}
The last statement follows directly from SD1: $2a-k$ and $2b-k$ are
nonzero odd integers. For the first statement the positivity proof
GM17 gives $\nu_s(y)\le h_0(y/(1+s))\le h_0(y/2)$ for $0\le s\le1$.
Consequently
\begin{equation}\tag{SD23}
\nu_s\{|y|<\sqrt\eta\}
\le\frac{\sqrt\eta}{\pi}
\left(1+\log\frac8{\sqrt\eta}\right)=:\mathfrak a(\eta)
\longrightarrow0.
\end{equation}
This uses $\operatorname{arcosh}u\le\log(2u)$ and integrates the
logarithm explicitly. The measures have no atoms. NG20 and the
subsequence argument therefore give $m_{N,\eta}/q\le
\mathfrak a(\eta)+o_h(1)$ uniformly in $N$ for each fixed $\eta$.
Meanwhile $\log(q/R_k)/\log q\to1/2$ if $m_0=1$ and $2/3$ if
$m_0>1$, and $\log|Q(0)|\le q\log R_k$. Choose $\eta$ with
$\mathfrak a(\eta)<1/(8A+8)$. Eventually $m_{N,\eta}/q<1/(4A+4)$,
the guard in SD21 holds, and $\log(q/R_k)>\tfrac13\log q$.
In SD21 the numerator is at most
$-\tfrac23q\log q+q|\log\eta|$.
Its ratio to $m_{N,\eta}$ is eventually smaller than $-2A\log q$,
because $\tfrac23(4A+4)>2A$. Exponentiating and taking square roots
proves SD22, with the harmless fixed prefactor absorbed by the strict
inequality in that coefficient. No effective value of the threshold
is inferred from the limiting spectral theorem.
\end{proof}

\section{The new metric handoff and the determinant it controls}

\partial_s\Tr(aI+X_s)^{-1}

\end{aligned}

\section{Two attained one-dimensional quotients control characteristic values}
Let $\lambda,\eta$ be positive measures on $[0,\infty)$ with infinite
support and all moments needed below. For the application they are the
literal pushforwards under $x=y^2/q^2$ of either $d\mu_k,d\sigma$, or
$Q_k^2d\mu_k,Q_k^2d\sigma$. These maps preserve the full measure masses.
Fix an integer $n\ge1$, $a\ge1$, and suppose the following inequalities
have been proved on degrees at most $n$:
\begin{equation}\tag{NG4}
 \ell\norm F_\eta^2\le\norm F_\lambda^2\le u\norm F_\eta^2,
 \qquad \int x|F|^2\dd\zeta\le B_\zeta\norm F_\zeta^2
 \quad(\zeta=\lambda,\eta).
\end{equation}
In our application NG2--3 prove them with
$L=q+2n+2$ for the relation measures, or $L=2n+2$ for the source measures,
and
\[
 B_\lambda=[C_{\rm mult}(h)(2L+\lfloor k/3\rfloor)/q]^2,
 \qquad B_\eta=[2(L+1)/q]^2.
\]
Increasing $L$ in these expressions covers every polynomial that follows.
Write $\Lambda=\log(u/\ell)$ and
\begin{equation}\tag{NG5}
 \Lambda_a^{\rm rat}=\Lambda+
       2\log\frac{(a+B_\lambda)(a+B_\eta)}{a^2}.
\end{equation}
Let $p_n^\zeta$ be the monic degree-$n$ orthogonal polynomial of $\zeta$.
Then
\begin{equation}\tag{NG6}
 \left|\log|p_n^\lambda(-a)|-\log|p_n^\eta(-a)|\right|
 \le\frac12\Lambda_a^{\rm rat}
                   +\frac32\log(1+B_\eta/a).
\end{equation}
Here and below the bounds are finite statements, before any limit.

\paragraph{Proof of the determinant and convexity identities.}
Set $f_\zeta(t)=\log\det[\int x^{i+j}(x+a)^t\dd\zeta]_{i,j<n}$.
Expanding two Vandermonde determinants and integrating proves
\[
 e^{f_\zeta(t)}=\frac1{n!}\int\prod_{i<j}(x_i-x_j)^2
                              \prod_i(x_i+a)^t\dd\zeta(x_i).
\]
The same expansion with $\prod_i(z-x_i)$ yields a monic polynomial
orthogonal to every degree less than $n$: integrating against $z^j\dd\zeta(z)$
and antisymmetrizing the $n+1$ variables gives a determinant with a
repeated row when $j<n$. Uniqueness of the monic orthogonal polynomial
therefore gives
$f_\zeta(1)-f_\zeta(0)=\log|p_n^\zeta(-a)|$.
The first identity is Andr\'eief's integration formula
\cite[Eq.~(1.7) and Section~2.2]{forrester}. The monic-polynomial identity
is Heine's formula; its moment determinant agrees with
\cite[18.2.29]{dlmf}. The expansions above prove the exact convention used here.
Differentiating the positive integral shows that $f_\zeta''$ is the
variance of $\sum_i\log(x_i+a)$ and is nonnegative. In our measures
all such differentiations on bounded $t$ intervals are justified by the
exponential tail and $x+a\ge1$. Convexity gives
\begin{equation}\tag{NG7}
 \tfrac12[f_\zeta(0)-f_\zeta(-2)]
 \le f_\zeta(1)-f_\zeta(0)
 \le\tfrac12[f_\zeta(2)-f_\zeta(0)].
\end{equation}

\paragraph{Proof that no dimension multiplies $\Lambda$.}
The spaces $V=\PP_{<n}$ and $(x+a)V$ share
$(x+a)\PP_{<n-1}$. Use this common polynomial basis in both Grams
and take its Schur complement. Its determinant cancels, leaving the
ratio of the attained squared norms of $1$ and $(x+a)x^{n-1}$ modulo
the common subspace. The triangular monic changes of basis have
determinant of absolute value one. NG4 passes to each affine minimum,
so
\begin{equation}\tag{NG8}
 |[f_\lambda(2)-f_\lambda(0)]-[f_\eta(2)-f_\eta(0)]|\le\Lambda.
\end{equation}
For $V$ and $V/(x+a)$ the common space is $\PP_{<n-1}$, and every
element of their sum is $F/(x+a)$ with $\deg F\le n$. Jensen's inequality
in $|F|^2\dd\zeta/\norm F_\zeta^2$ proves
\[
 (a+B_\zeta)^{-2}\norm F_\zeta^2
 \le\norm{F/(x+a)}_\zeta^2\le a^{-2}\norm F_\zeta^2.
\]
Thus on that entire rational space the comparison constants are
$\ell a^2/(a+B_\lambda)^2$ and $u(a+B_\eta)^2/a^2$.
The same two Schur complements give
\begin{equation}\tag{NG9}
 |[f_\lambda(0)-f_\lambda(-2)]-[f_\eta(0)-f_\eta(-2)]|
 \le\Lambda_a^{\rm rat}.
\end{equation}
This argument concerns the whole rational sum, not only its polynomial
intersection.

\paragraph{Proof of the finite reference convexity gap.}
Put $\eta_j=(x+a)^j\eta$ for $j=-2,-1,0,1$.
Every degree-$n$ orthogonal polynomial for these measures has all its
zeros in $(0,B_\eta]$. Indeed its zeros are the eigenvalues of the
compression of $x$ to $\PP_{<n}$, as follows directly from its three-term
recurrence. For $j\le0$, reweighting a probability by the decreasing
function $(x+a)^j$ cannot increase its mean of $x$: the difference
has the sign of
$\frac12\iint(x-y)((x+a)^j-(y+a)^j)\dd\pi(x)\dd\pi(y)\le0$.
For $j=1$, that mean is at most the mean with weight $(x+a)^2$, by
the same covariance identity with the increasing extra factor. The
latter is bounded by $B_\eta$ on the actual polynomial $(x+a)F$ of
degree at most $n$. This proves the claimed common bound.

For a positive measure $\zeta$, Christoffel's exact formula is
\[
 p_n^{(x+a)\zeta}(x)=
 \frac{p_{n+1}^{\zeta}(x)-
       [p_{n+1}^{\zeta}(-a)/p_n^{\zeta}(-a)]p_n^{\zeta}(x)}{x+a}.
\]
Divisibility, monicity, and orthogonality prove it without a limit.
The original and transformed zeros strictly alternate as
$x_1<x'_1<x_2<x'_2<\cdots<x_n<x'_n$.
To verify the direction, at $x_i$ the numerator is
$-b_n p_{n-1}(x_i)$, with $b_n>0$ from the recurrence, and $x_i+a>0$.
Successive signs alternate; the sign at the largest old zero is
negative and the new polynomial is positive at large positive $x$.
All $n$ new roots therefore occupy the indicated intervals. The
underlying reality, simplicity and interlacing follow from the same
positive three-term recurrence (or the sign-change orthogonality test).
Consequently each successive slope
$d_j=f_\eta(j+1)-f_\eta(j)$ satisfies
\[
 0\le d_{j+1}-d_j\le\log(1+B_\eta/a),\qquad j=-2,-1,0.
\]
The upper bound telescopes the alternating roots against the increasing
function $\log(a+x)$. In NG7 the upper reference slope exceeds $d_0$
by at most half this bound; the lower slope is below $d_0$ by at most
three halves. Combining NG8--9 proves NG6 in both directions.

\paragraph{The complete free-energy bound with a test function.}
Fix $a\ge1$ and $\phi(x)=\log(a+x)$. Set
\[
 F(t)=\sup_{\pi\in\mathcal P((0,\infty))}
 \left\{\iint\log|x-y|\dd\pi(x)\dd\pi(y)
            -\int V_s\dd\pi+t\int\phi\dd\pi\right\},
\]
using its bounded-above kernel to define the value at infinite energy.
For the actual $f_k(t)=\log\det\operatorname{Gram}_{\PP_{<n}}
 [(x+a)^t\eta_k]$, we claim
\begin{equation}\tag{NG14}
 \limsup\frac{f_k(nt)-2qn\log q}{n^2}\le F(t),\qquad
 \liminf\frac{f_k(0)-2qn\log q}{n^2}\ge F(0).
\end{equation}
Here is a proof including both ends of the original integration domain.
After the Vandermonde identity, NG12--13 bound the exponent above by
the logarithmic pair interaction minus the one-body potential with
coefficients tending to
\[
 V_{s,\epsilon,t}(x)=\pi\sqrt x/s-(4/s)\log(\sqrt x+\epsilon)-t\phi(x)
 \quad(r_k\le\epsilon).
\]
The discarded positive prefactor has logarithm $O(n\log q+n\log n)$,
hence is $o(n^2)$. Dividing the pair sum by $n(n-1)$ changes the
potential coefficients by $o(1)$. For fixed $\epsilon>0$, the error
is bounded by $o(1)(1+\sqrt x)$: use
$|\log(\sqrt x+\epsilon)|\le |\log\epsilon|+\sqrt x/\epsilon$
and $\log(a+x)\le\log a+2\sqrt{x/a}$.
Thus, for any small $\eta>0$, the exponent is bounded using
$V_{s,\epsilon,t}-\eta(1+\sqrt x)$ for all sufficiently large $k$.

Compactify $[0,\infty)$ by its point at infinity. The corresponding
kernel $K=\log|x-y|-(V(x)+V(y))/2$ tends uniformly to $-\infty$ at
infinity in either variable and at the diagonal, since
$K\le r(x)+r(y)$, $r(x)=\log(1+x)-V(x)/2$, and $r\to-\infty$.
Its truncation $K_T=\max(K,-T)$ is continuous on the compact square.
For the empirical probability $L_n$, the full off-diagonal sum obeys
$n^{-2}\sum_{i\ne j}K(x_i,x_j)\le\iint K_T\dd L_n^2+T/n$.
Integration against the probability $\omega^n$ preserves the maximum
of the right side. The maxima of the truncated energies decrease to
the maximum of the untruncated energy: choose maximizing measures,
pass to a weakly convergent subsequence, compare with every fixed
truncation, and then decrease that truncation. This argument also
applies when $\eta\downarrow0$ and $\epsilon\downarrow0$.
The kernels decrease pointwise to that for $V_s-t\phi$; at zero
the limit is $-\infty$, while at infinity confinement is uniform.
It follows that the successive maxima decrease to $F(t)$.
This proves the upper assertion, including the neighborhood of zero
where replacing the full root product by $x^q$ would be unjustified.

For the lower assertion restrict all variables to the support
$[u_s^2,v_s^2]$ of the actual equilibrium measure $\rho_s\dd x$.
NG13 and the lower Gamma bound make the log-weight error $O_h(k+\log q)$
per variable. The coefficient error $q/n-2/s=o(1)$ contributes an
additional $o(n)$ per variable on that fixed compact interval. The
combined error is $o(n)$ since $k/q\to0$.
Change the probability measure $\omega^n$ to $\rho_s^{\otimes n}$
and apply Jensen's inequality. The resulting logarithm is at least
$n(n-1)\iint\log|x-y|\rho_s(x)\rho_s(y)\dd x\dd y
-n^2\int V_s\rho_s\dd x-o(n^2)$.
The entropy cost is $O(n)$ because the density is bounded on a compact
positive interval with square-root decay at its edges. Its logarithmic
energy is finite there. This proves the lower assertion of NG14.

\section{Endpoints, density and equilibrium}
For $0<s\le1$, set
\begin{equation}\tag{GM1}
 \begin{gathered}
 V_s(x)=\frac\pi s\sqrt x-\frac2s\log x\quad(x>0),\qquad
 \\
 K(\kappa)=\int_0^{\pi/2}\frac{\dd\theta}{\sqrt{1-\kappa^2\sin^2\theta}},
 \quad E(\kappa)=\int_0^{\pi/2}\sqrt{1-\kappa^2\sin^2\theta}\dd\theta.
 \end{gathered}

The definitions of $K,E$ agree exactly with \cite[19.2.4, 19.2.5,
19.2.8(a,b)]{Carlson}. Put $r=\sqrt{1-\kappa^2}$. There is exactly one
$r\in(0,1)$ for which
\begin{equation}\tag{GM2}
 \frac{rK}{E}=\frac1{1+s},\qquad
 u=\frac2K,\quad v=\frac{2(1+s)}E,\quad r=\frac uv,
 \quad a=u^2,\quad b=v^2.
\end{equation}
Indeed, differentiation under the defining integrals and integration of
$d(\sin\theta\cos\theta/\sqrt{1-\kappa^2\sin^2\theta})$ give
$E_\kappa'=(E-K)/\kappa$ and
$K_\kappa'=E/[\kappa(1-\kappa^2)]-K/\kappa$.
They imply
\begin{equation}\tag{GM3}
 \frac{d}{dr}\frac{rK}E
 =\frac{(K-E)(E-r^2K)}{(1-r^2)E^2}>0.
\end{equation}
Both factors are positive by the defining integrals. Dominated convergence
gives $rK/E\to0$ at $r\downarrow0$ and $rK/E\to1$ at $r\uparrow1$:
for the first limit the integrand of $rK$ is bounded by one and tends
to zero except at the last endpoint, while $E\to1$.
This proves existence, uniqueness and continuous dependence.

With $R_w=\sqrt{(a+w)(b+w)}$ define
\begin{equation}\tag{GM4}
 \rho_s(x)=\frac{\sqrt{(b-x)(x-a)}}{4\pi s x}
       \int_0^\infty\frac{\sqrt w}{(x+w)R_w}\dd w\quad(a<x<b),
 \qquad \rho_s(x)=0\quad(x\notin(a,b)).
\end{equation}
Here is a derivation verifying its probability mass and the original
equilibrium problem. It also specifies the transform needed for its moment.
Write $\alpha=2/s$, $\beta=\pi/s$, $m=(a+b)/2$, $R_0=uv$ and
$R(z)=\sqrt{(z-a)(z-b)}$, with $R(z)/z\to1$ at infinity.
The substitution $x=a\cos^2\theta+b\sin^2\theta$ gives
\begin{equation}\tag{GM5}
 \int_a^b\frac{V_s'(x)}{\sqrt{(b-x)(x-a)}}\dd x=0,
 \qquad
 \int_a^b\frac{xV_s'(x)}{\sqrt{(b-x)(x-a)}}\dd x=2\pi.
\end{equation}
The required elementary integrals, with numerators $x^{-1/2}$,
$x^{-1}$ and $x^{1/2}$, are $2K/v$, $\pi/(uv)$ and $2vE$.
The substitution $w=xt^2$ gives
$\int_0^\infty[\sqrt w(x+w)]^{-1}\dd w=\pi/\sqrt x$.
Tonelli's theorem and (GM5) now imply
\begin{equation}\tag{GM6}
 \frac\alpha{R_0}=\frac\beta{2\pi}
       \int_0^\infty\frac{\dd w}{\sqrt wR_w},\qquad
 \int_0^\infty\frac{1-w/R_w}{\sqrt w}\dd w=2vE.
\end{equation}
Thus $h(x):=2\pi\rho_s(x)/\sqrt{(b-x)(x-a)}$ has the two forms
\begin{equation}\tag{GM7}
 h(x)=\frac\alpha{xR_0}-\frac\beta{2\pi}
       \int_0^\infty\frac{\dd w}{\sqrt w(x+w)R_w}
 =\frac\beta{2\pi x}\int_0^\infty\frac{\sqrt w}{(x+w)R_w}\dd w>0.
\end{equation}

 \frac1\pi\int_a^b\frac{\sqrt{(b-x)(x-a)}}{x+w}\dd x=m+w-R_w,
 \quad
 \frac1\pi\int_a^b\frac{\sqrt{(b-x)(x-a)}}{(x+w)(z-x)}\dd x
 =1-\frac{R_w+R(z)}{z+w}=:H_w(z).

\begin{equation}\tag{GM9}
 \int_a^b\rho_s(x)\dd x
 =\frac\alpha{2R_0}(m-R_0)-\frac\beta{4\pi}

\begin{equation}\tag{GM18}
 \boxed{\mathfrak m(s)=\int y^2\dd\nu_s(y)
 =\frac23(1+s)^3-\frac{(1+s)(u^2+v^2)-2uv}{6}.}
\end{equation}
In particular $\mathfrak m(0)=2/3$ and
$\mathfrak m(1)=(16-u^2-v^2+uv)/3$.
For the endpoint limit $s\downarrow0$, (GM2) gives
$r\uparrow1$ and $u,v\to4/\pi$; the formula agrees continuously
with this definition at zero.

\section{Finite lower bounds, with no fitted elliptic constants}
First, $r(1)>1/8$. The elementary tangent substitution gives
\[
 K=\int_0^\infty\frac{\dd t}{\sqrt{(1+t^2)(1+r^2t^2)}}
 \le2\operatorname{arsinh}(1)+\log(1/r).
\]
For this bound split the integral at $1$ and $1/r$, use the change
$t=1/(rw)$ in the final part, and bound the middle integrand by $1/t$.
At $r=1/8$ this is less than $39/10<4$: the elementary inequalities
$\operatorname{arsinh}(1)<9/10$ and $\log2<7/10$ follow from
$\sqrt2<99/70$ and the positive Taylor sums through degree five for
$e^{9/10}$ and $e^{7/10}$ respectively.
Thus $rK/E<1/2$ at $r=1/8$ and (GM3) proves the assertion.
Also $K>E$ gives $r(1)<1/2$.

If $r=r(1)\ge3/20$, the decreasing function $1-r+r^2$ on $(0,1/2)$
and $v<4$ give $v^2(1-r+r^2)<349/25$.
If $1/8<r\le3/20$, differentiation under the integral in (GM1) gives
\begin{equation}\tag{GM19}
 E(r)-1=\int_0^r t\int_0^{\pi/2}
 \frac{\sin^2\theta}{\sqrt{\cos^2\theta+t^2\sin^2\theta}}
 \dd\theta\dd t
 \ge\frac{r^2(K-E)}{2(1-r^2)}.
\end{equation}
Use $K=E/(2r)$ and
$r(1-2r)/[4(1-r^2)]\ge7/320$ on this interval.
It follows that $E\ge320/313$ and
\[
 v^2(1-r+r^2)\le\frac{57}{4}\left(\frac{313}{320}\right)^2
 <\frac{349}{25}.
\]
Both cases prove $\mathfrak m(1)>17/25$ with exact rational constants.

The same argument proves the claimed uniform bound. For $0<s\le1$,
(GM3) gives $r(s)\ge r(1)>1/8$; $K$ decreases with $r$, so $K<4$
and $u>1/2$. Furthermore $v\ge u$, $v\le2(1+s)$, and
$(1+s)u/v=E/K<1$. With $T=1+s$, the exact identity
\begin{equation}\tag{GM20}
 6\mathfrak m(s)-uv=T(4T^2-v^2)+u(v-Tu)\ge0
\end{equation}
therefore gives $\mathfrak m(s)\ge uv/6>1/24$. The assertion is also
true at $s=0$, where its value is $2/3$.

\section{An explicit finite series certificate strengthening the endpoint}
The accompanying exact/radius computation strengthens the finite endpoint
bound to
\begin{equation}\tag{GM21}
 \frac{16010167095}{10^{11}}<r(1)<\frac{16010167096}{10^{11}},
 \qquad \boxed{\mathfrak m(1)>\frac{10249}{10000}.}
\end{equation}
Here are the full mathematical error bounds used by the certificate.
For $z=1-r^2\in(0,1)$ and
$a_n=(\binom{2n}n/4^n)^2$, expansion of the defining integrands yields
\begin{align}\tag{GM22}
 \frac{2K}{\pi}&=\sum_{n=0}^{J}a_nz^n+R_K,
 &0<R_K\le\frac{z^{J+1}}{1-z},\notag\\
 \frac{2E}{\pi}&=1-\sum_{n=1}^{J}\frac{a_nz^n}{2n-1}-R_E,
 &0<R_E\le\frac{z^{J+1}}{(2J+1)(1-z)}.\notag
\end{align}
For $K$ use the binomial series and the cosine moment in (GM15).
For $E$ the binomial coefficient is the negative of the corresponding
coefficient for $K$ divided by $2n-1$. Termwise integration is justified
by uniform convergence because $z<1$. The tail bounds use $0<a_n\le1$.
The recurrence $a_0=1$, $a_{n+1}=a_n((2n+1)/(2n+2))^2$ retains every
coefficient. The certificate uses $J=2048$ and outward-rounded real
interval arithmetic with 240-bit midpoints. It proves that $rK/E-1/2$
is negative at the lower rational in (GM21), positive at the upper,
and then evaluates (GM18) on the entire resulting interval.
The ratio test is independent of $\pi$. The endpoints $u=2/K$,
$v=4/E$ use a rigorous $\pi$ interval obtained from
\[
 \pi=16\arctan(1/5)-4\arctan(1/239).
\]
For completeness, the tangent addition formula gives
$\tan(4\arctan(1/5)-\arctan(1/239))=1$ exactly;
the angle is in $(0,\pi/2)$, proving the identity.
Alternating arctangent sums with their first omitted terms bound both
arctangents (respectively 24 and 8 terms). Thus no floating point
approximation to an unbounded remainder is used. All rational endpoints,
interval enclosures and pass/fail comparisons are saved in the check
receipt. The general bound (GM20) remains an analytic proof over the
entire continuum; sampled parameter checks are not substituted for it.

\section{The complete signed Gaussian expression}
Let
\begin{equation}\tag{GM23}
 H_s(t)=\int e^{-ty^2}\dd\nu_s(y),\qquad
 \mathfrak A(t)=2[-H_0'(t)+H_1'(t)]\quad(t\ge0).
\end{equation}
All these derivatives exist by compact support. If
$I_0(z)=\pi^{-1}\int_0^\pi e^{z\cos\theta}\dd\theta$, the arcsine
mixture (GM15) gives exactly
\begin{equation}\tag{GM24}
 H_0(t)=\int_0^1e^{-2tw^2}I_0(2tw^2)\dd w,\qquad
 H_s(t)=(1+s)H_0((1+s)^2t)-s\int e^{-tx}\rho_s(x)\dd x.
\end{equation}
The second identity is the two-to-one pushforward $x=y^2$ of (GM16),
including its full mass. Since $\nu_0,\nu_1$ are positive probabilities
with squared supports in $[0,4]$ and $[0,16]$, respectively,
\[
 -H_0'(1/1024)\le2/3,\qquad
 -H_1'(1/1024)\ge\frac{63}{64}\mathfrak m(1).
\]
Here $e^{-x/1024}\ge1-x/1024\ge63/64$ on the second support.
The completely analytic bound $\mathfrak m(1)>17/25$ proves the incoming
$\mathfrak A(1/1024)<-13/2400$. The finite-series certificate strengthens
this to
\begin{equation}\tag{GM25}
 \boxed{\mathfrak A(1/1024)
 <\frac43-\frac{63}{32}\frac{10249}{10000}
 =-\frac{657061}{960000}.}
\end{equation}
This statement is about the explicitly constructed equilibrium receiver.
Its transfer to the original finite arithmetic observation matrices
requires the actual spectral convergence argument. This note proves
neither that convergence nor the individual projected phase signs.
