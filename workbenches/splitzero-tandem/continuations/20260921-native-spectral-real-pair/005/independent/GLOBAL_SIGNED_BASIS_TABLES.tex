\subsection{Every multiplication coefficient}
The following table gives every product other than multiplication by $e_0=1$.
Commutativity gives the entries below the diagonal; each right-hand side is in the unchanged eight-section basis.
\paragraph{Even products}
\begin{align}
e_1 e_1 &=A e_{2} - e_{1}\tag{GB13a}\\
e_1 e_2 &=A e_{3} - B e_{1}\tag{GB13b}\\
e_1 e_3 &=- A D e_{0} - C e_{1}\tag{GB13c}\\
e_2 e_2 &=- A D e_{0} - B e_{2} - C e_{1} + e_{3}\tag{GB13d}\\
e_2 e_3 &=- C e_{2} - D e_{0} - D e_{1}\tag{GB13e}\\
e_3 e_3 &=- B D e_{0} - C e_{3} - D e_{2}\tag{GB13f}
\end{align}
\paragraph{Even--odd products}
\begin{align}
e_1 o_0 &=A o_{1} - o_{0}\tag{GB14a}\\
e_1 o_1 &=A o_{2} - B o_{0}\tag{GB14b}\\
e_1 o_2 &=A o_{3} - C o_{0}\tag{GB14c}\\
e_1 o_3 &=- D o_{0}\tag{GB14d}\\
e_2 o_0 &=A o_{2} - B o_{0}\tag{GB14e}\\
e_2 o_1 &=A o_{3} - B o_{1} - C o_{0} + o_{2}\tag{GB14f}\\
e_2 o_2 &=- C o_{1} - D o_{0} + o_{3}\tag{GB14g}\\
e_2 o_3 &=- D o_{1}\tag{GB14h}\\
e_3 o_0 &=A o_{3} - C o_{0}\tag{GB14i}\\
e_3 o_1 &=- C o_{1} - D o_{0} + o_{3}\tag{GB14j}\\
e_3 o_2 &=B o_{3} - C o_{2} - D o_{1}\tag{GB14k}\\
e_3 o_3 &=- D o_{2}\tag{GB14l}
\end{align}
\paragraph{Odd products}
\begin{align}
o_0 o_0 &=A^{2} C e_{0} + 4 A^{2} e_{3} - A e_{2} + e_{1} \left(- 2 A B + 1\right)\tag{GB15a}\\
o_0 o_1 &=- 2 A B e_{2} + 3 A e_{3} + e_{0} \left(- 4 A^{2} D + A C\right) + e_{1} \left(- 3 A C + B\right)\tag{GB15b}\\
o_0 o_2 &=2 A B e_{3} - 3 A C e_{2} + e_{0} \left(A B C - 3 A D\right) + e_{1} \left(- 4 A D + C\right)\tag{GB15c}\\
o_0 o_3 &=A C e_{3} - 4 A D e_{2} + D e_{1} + e_{0} \left(- 2 A B D + A C^{2}\right)\tag{GB15d}\\
o_1 o_1 &=e_{0} \left(- 7 A D + C\right) + e_{1} \left(- 4 A D + 2 B^{2} - 2 C\right) + e_{2} \left(- 3 A C - B\right) + e_{3} \left(- 2 A B + 3\right)\tag{GB15e}\\
o_1 o_2 &=e_{0} \left(- 2 A B D + B C - 3 D\right) + e_{1} \left(2 B C - 3 D\right) + e_{2} \left(- 4 A D - 2 C\right) + e_{3} \left(- 3 A C + 2 B\right)\tag{GB15f}\\
o_1 o_3 &=2 B D e_{1} - 3 D e_{2} + e_{0} \left(- A C D - 2 B D + C^{2}\right) + e_{3} \left(- 4 A D + C\right)\tag{GB15g}\\
o_2 o_2 &=e_{0} \left(3 A C D + B^{2} C - 5 B D\right) + e_{1} \left(- 2 B D + 3 C^{2}\right) + e_{2} \left(- B C - 3 D\right) + e_{3} \left(- 4 A D + 2 B^{2} - 2 C\right)\tag{GB15h}\\
o_2 o_3 &=- 2 B D e_{2} + 3 C D e_{1} + e_{0} \left(4 A D^{2} - 2 B^{2} D + B C^{2} - C D\right) + e_{3} \left(B C - 3 D\right)\tag{GB15i}\\
o_3 o_3 &=- C D e_{2} + 4 D^{2} e_{1} + e_{0} \left(- 3 B C D + C^{3} + 3 D^{2}\right) + e_{3} \left(- 2 B D + C^{2}\right)\tag{GB15j}
\end{align}
\subsection{The complete trace blocks}
\begin{equation}T_{ee}=\left[\begin{matrix}8 & -2 & - 4 B & - 6 C\\-2 & - 4 A B + 2 & - 6 A C + 2 B & - 8 A D + 2 C\\- 4 B & - 6 A C + 2 B & - 8 A D + 4 B^{2} - 4 C & 4 B C - 6 D\\- 6 C & - 8 A D + 2 C & 4 B C - 6 D & - 4 B D + 6 C^{2}\end{matrix}\right],\qquad T_{eo}=0.\tag{GB16}\end{equation}
The symmetric odd block has the following ten entries; $T_{ji}=T_{ij}$ supplies the other six.
\begin{align}
(T_{oo})_{00}&=- 16 A^{2} C + 8 A B - 2\tag{GB17a}\\
(T_{oo})_{01}&=- 32 A^{2} D + 8 A B^{2} - 4 A C - 2 B\tag{GB17b}\\

(T_{oo})_{03}&=2 A C^{2} - 2 D\tag{GB17d}\\
(T_{oo})_{11}&=24 A B C - 48 A D - 6 C\tag{GB17e}\\
(T_{oo})_{12}&=18 A C^{2} - 18 D\tag{GB17f}\\
(T_{oo})_{13}&=16 A C D - 8 B D + 2 C^{2}\tag{GB17g}\\
(T_{oo})_{22}&=48 A C D - 24 B D + 6 C^{2}\tag{GB17h}\\
(T_{oo})_{23}&=32 A D^{2} - 8 B^{2} D + 2 B C^{2} + 4 C D\tag{GB17i}\\
(T_{oo})_{33}&=- 8 B C D + 2 C^{3} + 16 D^{2}\tag{GB17j}
\end{align}
