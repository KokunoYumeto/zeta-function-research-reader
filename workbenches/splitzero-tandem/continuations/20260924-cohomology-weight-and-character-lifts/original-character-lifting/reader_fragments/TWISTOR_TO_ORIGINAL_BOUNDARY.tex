\section{The odd twistor dynamics on the original annulus object}
\label{sec:twistor-original-boundary}

We apply the concrete dynamics in Alain Connes and Caterina Consani,
\href{https://arxiv.org/abs/2609.00299v1}{\emph{The Absolute Twistor
Line and the Geometry of $\overline{\operatorname{Spec}\mathbb Z}$}},
Proposition \texttt{prop:twistor\_frobenius}, to the exact companion
and annulus space of
\href{https://github.com/KokunoYumeto/zeta-function-research-reader/blob/3c78cfbd9e194d35117b33277c421c6564ce3dbf/workbenches/splitzero-tandem/continuations/20260923-full-prime-support/033/MIXED_PRIME_ENERGY_AND_BOUNDARY.tex}{033,
SBT1--23}. This section proves the receiving maps and their complete
metric defects. Neither a spherical-algebra point functor nor an
arithmetic cohomology comparison is silently substituted for these maps.

\subsection{The actual circle map and its linear pullback}

For every positive odd integer $d$ put
\begin{equation}\tag{TOB1}
 \chi_d=(-1)^{(d-1)/2},\qquad m_d=\chi_d d,\qquad
 f_d(z)=\chi_d z^{m_d}\quad(z\in\mathbb C^\times).
\end{equation}
At $0$ and $\infty$ use the unique continuous extension to
$\mathbb P^1(\mathbb C)$. Thus $f_d=z^d$ when $d\equiv1\pmod4$,
and $f_d=-z^{-d}$ when $d\equiv3\pmod4$, exactly as in the source.
For odd $c,d$, the equality $\chi_{cd}=\chi_c\chi_d$ and the oddness
of $m_c$ give
\[
 f_c(f_d(z))=\chi_c\chi_d^{m_c}z^{m_cm_d}
            =\chi_{cd}z^{m_{cd}}=f_{cd}(z).
\]
The formula extends at both poles. The real structure
$\sigma(z)=-1/\overline z$ satisfies
$f_d\sigma=\sigma f_d$: substituting either sign of $\chi_d$
proves this equality on $\mathbb C^\times$, and continuity proves
it at the poles.

Give $S^1$ its probability Haar measure $dt/(2\pi)$, and write
$e_j(z)=z^j$, $j\in\mathbb Z$. Define the pullback
\begin{equation}\tag{TOB2}
 V_d:L^2(S^1)\longrightarrow L^2(S^1),\qquad
 (V_d\varphi)(z)=\varphi(f_d(z)),\qquad
 V_de_j=\chi_d^je_{m_dj}.
\end{equation}
It is an isometry: substitution in the integral of a trigonometric
polynomial, or orthogonality of the displayed distinct basis vectors,
proves the norm equality on their span; that span is dense, so the
operator extends uniquely to all of $L^2$. Pullback by the measurable
degree-$d$ circle covering is the same extension, since that covering
preserves Haar measure. The adjoint and range projection are exactly
\begin{equation}\tag{TOB3}
 V_d^*e_j=
 \begin{cases}\chi_d^{j/m_d}e_{j/m_d},&m_d\mid j,\\0,&m_d\nmid j,
 \end{cases}
 \qquad
 V_dV_d^*e_j=\mathbf1_{d\mid j}e_j.
\end{equation}
For $S e_j=e_{j+1}$, direct substitution yields
\begin{equation}\tag{TOB4}
 V_dS=\chi_d S^{m_d}V_d,\qquad V_cV_d=V_{cd}.
\end{equation}
These identities include all signs for the $3\pmod4$ case.

We use the same coefficient operator on $E$-valued distributions
$E^{\mathbb Z}$, paired with finitely supported covectors:
\begin{equation}\tag{TOB5}
 (V_dx)_j=
 \begin{cases}\chi_d^{j/m_d}x_{j/m_d},&m_d\mid j,\\0,&m_d\nmid j.
 \end{cases}
\end{equation}
Every pairing is a finite sum; no convergence hypothesis is needed.
Equation (TOB4) continues to hold by its coordinate calculation.
On $\ell^2(\mathbb Z;E,G_N)$ the same map is an isometry.

\subsection{An exact finite receiver with the original metric}

Keep the original parameters and operators:
\begin{equation}\tag{TOB6}
\begin{gathered}
 k\ge17,\quad k\equiv1\pmod4,\quad q=(k+1)^2,
 \quad\delta,\gamma>0,\quad q-1\le N\le2q,\\
 \lambda_{ab}=(2b-k)\gamma-i(2a-k)\delta,\quad
 Q_k(y)=\prod_{a,b=0}^k(y-\lambda_{ab}),\\
 E=\mathbb C[y]/(Q_k),\quad M[v]=[yv],\quad
 F_p=e^{i(\log p)M},\quad \widetilde F_p=p^{k/2}F_p,
 \quad\langle v,w\rangle_N=v^*G_Nw.
\end{gathered}
\end{equation}
The positive matrix $G_N$ is the full original metric at the stated
source cutoff. The four cutoffs $q-1,q,2q-1,2q$ are included separately.
The new integer $d$ is neither a source cutoff nor the amplification
parameter $k$. Let $P_{ab}$ and $\ell_{ab}$ be the original Lagrange
projectors and vectors. Thus
\begin{equation}\tag{TOB7}
 F_p\ell_{ab}=\omega_{ab}\ell_{ab},\qquad
 \omega_{ab}=p^{(2a-k)\delta}e^{i(2b-k)\gamma\log p},\qquad
 (\mathcal J_pv)_j=F_p^{-j}v.
\end{equation}
The original embedding satisfies $S\mathcal J_p=\mathcal J_pF_p$
and $\operatorname{ev}_0\mathcal J_p=I$ by substitution.

Define
\begin{equation}\tag{TOB8}
 A_d=\chi_d F_p^{\chi_d},\qquad E_d=E^{\oplus d},\qquad
 G_{N,d}=\operatorname{diag}(G_N,\ldots,G_N),
 \qquad \mathcal W_d=V_d\mathcal J_p.
\end{equation}
For every integer $\ell$,
\begin{equation}\tag{TOB9}
 (\mathcal W_dv)_{d\ell}=A_d^{-\ell}v,
 \qquad(\mathcal W_dv)_j=0\quad(d\nmid j),
 \qquad S^d\mathcal W_d=\mathcal W_dA_d.
\end{equation}
For $\chi_d=1$ this is (TOB5) directly. For $\chi_d=-1$,
$j/m_d=-\ell$, so the coefficient is
$(-1)^{-\ell}F_p^\ell v=(-F_p^{-1})^{-\ell}v$.
This proves both cases without changing any original coordinate.

Let the receiving companion and embedding be
\begin{equation}\tag{TOB10}
\begin{split}
 C_d(v_0,\ldots,v_{d-1})&=(A_dv_{d-1},v_0,\ldots,v_{d-2}),\\
 \mathcal I_d(v)&=\sum_{r=0}^{d-1}S^r\mathcal W_dv_r,
 \qquad (\mathcal I_dv)_{d\ell+r}=A_d^{-\ell}v_r
 \quad(0\le r<d).
\end{split}
\end{equation}
Then
\begin{equation}\tag{TOB11}
\begin{gathered}
 \operatorname{ev}_{[0,d-1]}\mathcal I_d=I_{E_d},\qquad
 S\mathcal I_d=\mathcal I_d C_d,\qquad
 C_d^d=\operatorname{diag}(A_d,\ldots,A_d),\\
 \operatorname{im}\mathcal I_d
 =\{x\in E^{\mathbb Z}:x_{j-d}=A_dx_j\text{ for all }j\}.
\end{gathered}
\end{equation}
The first identity follows at the $d$ indicated indices. Shifting
the sum in (TOB10), and using its last term (TOB9), proves the second.
Following each block through $d$ steps proves the third. In the
displayed recurrence, division with remainder $j=d\ell+r$ forces
$x_j=A_d^{-\ell}x_r$ for positive and negative $\ell$, proving the
last equality and its inverse. The image is also the smallest
$S$- and $S^{-1}$-stable linear subspace containing
$\mathcal W_d(E)$: its $d$ shifts generate the image, and
(TOB9), with invertible $A_d$, proves stability in both directions.
In particular its dimension is exactly $dq$, with all original
$q$ labels retained.

The original metric is transported by the exact first-block evaluation:
\begin{equation}\tag{TOB12}
 \langle\mathcal I_dv,\mathcal I_dw\rangle_{d,N}
   =\sum_{r=0}^{d-1}v_r^*G_Nw_r.
\end{equation}
This is not the infinite sum of annulus norms. Each original copy
has precisely its original norm; no averaging by $d$ is performed.
Direct subtraction gives the full defect
\begin{equation}\tag{TOB13}
\begin{split}
 \|C_dv\|_{N,d}^2-\|v\|_{N,d}^2
   &=v_{d-1}^*(A_d^*G_NA_d-G_N)v_{d-1},\\
 \|p^{k/2}C_dv\|_{N,d}^2-p^k\|v\|_{N,d}^2
   &=p^kv_{d-1}^*(A_d^*G_NA_d-G_N)v_{d-1}.
\end{split}
\end{equation}
The cancellation concerns the unchanged first $d-1$ summands only;
the last block is the complete remaining quadratic form.
Equivalently, the singular values of $C_d$ in $G_{N,d}$ consist
of $(d-1)q$ copies of $1$ and the $q$ singular values of $A_d$ in
$G_N$. Indeed $C_d$ is a unitary block permutation after the
block diagonal operator $\operatorname{diag}(I,\ldots,I,A_d)$.
This proves
\begin{equation}\tag{TOB14}
 \|C_d\|_{N,d}=\max\{1,\|A_d\|_N\},\qquad
 \|C_d^{-1}\|_{N,d}=\max\{1,\|A_d^{-1}\|_N\}
 \quad(d>1).
\end{equation}
For $d=1$, the corresponding norms are just $\|F_p\|_N$ and
$\|F_p^{-1}\|_N$. Since $\delta>0$, these latter norms already
exceed $1$, by the original eigenvectors of largest and smallest
modulus, respectively.

For the last-block vector $v=(0,\ldots,0,\ell_{ab})$, the exact
right side in (TOB13) is
\begin{equation}\tag{TOB15}
 p^k\bigl(p^{2\chi_d(2a-k)\delta}-1\bigr)
                         \|\ell_{ab}\|_N^2.
\end{equation}
At every odd $d$, both signs occur as $a$ varies. For either fixed
residue class modulo four this expression is independent of $d$.
Thus the original metric defect has not vanished in a growing
covering degree.

\subsection{All roots, labels, phases, and finite boundaries}

Put $b_d=(1-\chi_d)/2$. The eigenvectors and eigenvalues of $C_d$
are explicitly
\begin{equation}\tag{TOB16}
\begin{gathered}
 \zeta_{ab,t}=
 \exp\!\left(\frac{\chi_di(\log p)\lambda_{ab}
                              +i\pi b_d+2\pi it}{d}\right),
 \quad 0\le t<d,\\
 v_{ab,t}=(\ell_{ab},\zeta_{ab,t}^{-1}\ell_{ab},\ldots,
                   \zeta_{ab,t}^{-(d-1)}\ell_{ab}),\\
 C_dv_{ab,t}=\zeta_{ab,t}v_{ab,t},\qquad
 |\zeta_{ab,t}|=p^{\chi_d(2a-k)\delta/d}.
\end{gathered}
\end{equation}
The first formula specifies the phase explicitly, without a choice
of a new logarithm. Its $d$th power is the eigenvalue
$\chi_d\omega_{ab}^{\chi_d}$ of $A_d$. This proves the second
line by substitution in (TOB10). For each fixed $(a,b)$ the $d$
roots are distinct and the corresponding Vandermonde matrix is
invertible. The original $P_{ab}$ split $E_d$ into their $d$-block
spaces, so these $dq$ vectors form a basis even if roots from
different original labels have equal phases or eigenvalues.
Their full Gram entries are
\begin{equation}\tag{TOB17}
 \langle v_{ab,t},v_{ce,u}\rangle_{N,d}
 =\langle\ell_{ab},\ell_{ce}\rangle_N
   \sum_{r=0}^{d-1}
        (\overline{\zeta_{ab,t}}\zeta_{ce,u})^{-r}.
\end{equation}
All original off-diagonal metric entries remain in this formula.

The original annulus functions and operator are
\begin{equation}\tag{TOB18}
 h_j(\mathbf x)=p^{kj/2}(1-p^{-1})^{-k/2}
       \mathbf1_{(p^j\mathbb Z_p^\times)^k}(\mathbf x),\qquad
 \mathscr T_{p,k}=p^{k/2}\mathscr U_p^{\otimes k},
 \quad \mathscr U_pF(x)=\sqrt p F(x/p).
\end{equation}
Here Haar measure gives $\mu(\mathbb Z_p)=1$. The support in
$h_j$ has measure $p^{-kj}(1-p^{-1})^k$; these functions are
orthonormal, and substitution proves
$\mathscr T_{p,k}h_j=p^{k/2}h_{j+1}$.
For an integer $H\ge0$ define
\begin{equation}\tag{TOB19}
 I_{d,H}v=\sum_{j=-H}^Hh_j\otimes(\mathcal I_dv)_j.
\end{equation}
Its squared norm is exactly
$\sum_{j=-H}^H\|A_d^{-\lfloor j/d\rfloor}v_{j\bmod d}\|_N^2$,
where the remainder lies in $\{0,\ldots,d-1\}$ also for negative
$j$. The complete boundary is
\begin{equation}\tag{TOB20}
\begin{split}
 B_{d,H}v&=(\mathscr T_{p,k}\otimes I)I_{d,H}v
                       -I_{d,H}(p^{k/2}C_dv)\\
 &=p^{k/2}\bigl(h_{H+1}\otimes(\mathcal I_dv)_H
                  -h_{-H}\otimes(\mathcal I_dv)_{-H-1}\bigr),\\
 \|B_{d,H}v\|^2&=p^k\bigl(
       \|(\mathcal I_dv)_H\|_N^2+
       \|(\mathcal I_dv)_{-H-1}\|_N^2\bigr).
\end{split}
\end{equation}
The shift relation (TOB11) cancels every common summation index;
the two remaining annuli are disjoint, proving the norm formula.
For $d>2$, this boundary map need not be injective: it observes
only the two indicated residue blocks. Its kernel is exactly the
subspace whose entries at residues $H\bmod d$ and
$(-H-1)\bmod d$ vanish. This retains a new boundary kernel that
cannot be removed by citing the injective $d=1$ boundary.

For the eigenvector $v_{ab,t}$, (TOB10) gives the full sequence
$(\mathcal I_dv_{ab,t})_j=\zeta_{ab,t}^{-j}\ell_{ab}$.
With $r=|\zeta_{ab,t}|$, equation (TOB20) becomes
\begin{equation}\tag{TOB21}
 \frac{\|B_{d,H}v_{ab,t}\|^2}{\|I_{d,H}v_{ab,t}\|^2}
 =p^k\frac{r^{-2H}+r^{2H+2}}{\sum_{j=-H}^Hr^{-2j}}
 \longrightarrow p^k|r^2-1|
 =p^k\left|p^{2\chi_d(2a-k)\delta/d}-1\right|.
\end{equation}
For $r>1$ divide numerator and denominator by $r^{2H}$; for
$r<1$ divide by $r^{-2H}$. The resulting geometric series prove
the displayed limits. They are positive for every finite $d$,
since $k$ is odd. As $d$ tends to infinity they tend to zero, but
the exact $d$-step return (TOB11) and original defect (TOB15)
remain. These statements concern different, explicitly related
operators; the root estimate has not been substituted for its
unchanged return.

For a nonzero $v\in E_d$ define
\begin{equation}\tag{TOB22}
 R_{p,d}(v)=\frac{\delta\log p}{d}
       \max\{|2a-k|:P_{ab}v_r\ne0\text{ for some }r\}.
\end{equation}
Then
\begin{equation}\tag{TOB23}
 \lim_{H\to\infty}\frac1H
   \log\max_{|j|\le H}\|(\mathcal I_dv)_j\|_N
 =R_{p,d}(v)>0.
\end{equation}
For the upper bound expand each of the $d$ blocks with the original
$P_{ab}$ and use $|\lfloor j/d\rfloor|\le H/d+1$. The finite
sum is bounded by a constant times $\exp(HR_{p,d})$.
For the lower bound choose a nonzero projected block attaining the
maximum. Choose $j=d\ell+r$ within $[-H,H]$ with the sign of
$\ell$ that makes $A_d^{-\ell}$ grow on that block and with
$|\ell|\ge H/d-2$. Projection bounds its norm from below by
the growing projected norm divided by $\|P_{ab}\|_N$.
The constants are independent of $H$ and their logarithms disappear
after division by $H$. This proves (TOB23), retaining the original
metric. Consequently the image contains no nonzero
polynomial-growth sequence for any fixed finite $d$.

\subsection{Exact observation-kernel and support transport}

Retain the original isometric section $J:B\to E$,
$P_B=JJ^\dagger$, $Q=I-P_B$, and $K=\ker\Lambda=QE$.
Let $J_d=\operatorname{diag}(J,\ldots,J)$, and define
$\widehat C_d=J_d^\dagger C_dJ_d$. It is (TOB10) with the last
corner $A_d$ replaced by $J^\dagger A_dJ$. Therefore
\begin{equation}\tag{TOB24}
 C_dJ_d-J_d\widehat C_d
  :(b_0,\ldots,b_{d-1})\longmapsto
       (QA_dJb_{d-1},0,\ldots,0).
\end{equation}
This follows by multiplying each block, using $J^\dagger J=I$.
Applying the injective $\mathcal I_d$ gives the full corresponding
defect for the distribution comparison. More explicitly,
\begin{equation}\tag{TOB25}
 p^{k/2}S\mathcal I_dJ_d
     -\mathcal I_dJ_d(p^{k/2}\widehat C_d)
 =p^{k/2}\mathcal I_d
       (QA_dJ\operatorname{pr}_{d-1},0,\ldots,0).
\end{equation}
The finite-cutoff version is the sum of this compressed-kernel term
and the two annulus terms (TOB20); thus neither contribution is
lost by compression.

For the full bounded distributive lattice $L$, every vector space
$V$ here has its synchronized carrier
$V^\uparrow=\{(0,\lambda):\lambda\in L\}\cup
\{(v,1_L):v\in V\}$. A linear map $T$ lifts as
$(v,\lambda)\mapsto(Tv,\lambda)$. Substitution in the addition
and scalar multiplication laws proves linearity over $G_L(\mathbb C)$.
Applying this to (TOB10)--(TOB11) proves that the first-block
evaluation recovers every $(0,\lambda)$ and every nonzero original
vector. In (TOB24)--(TOB25), zero amplitude in the exact defect is
still a labelled zero. All these identities therefore have exact
full-$L$ lifts, including the observation-kernel inclusion and the
growing finite receiver; no free-semimodule basis change is assumed.

\subsection{The complete boundary window restores the original exponent}

A single boundary in (TOB20) reads at most two residue blocks.
The exact joint map through a complete window reads every block.
For an integer $h\ge0$ put
\begin{equation}\tag{TOB26}
 \mathcal B_{d,h}v=(B_{d,dh}v,B_{d,dh+1}v,\ldots,
                              B_{d,dh+d-1}v).
\end{equation}
Give its target the direct-sum Hilbert norm. For $H=dh+r$,
equation (TOB10) gives
$x_H=A_d^{-h}v_r$ and
$x_{-H-1}=A_d^{h+1}v_{d-1-r}$. Summing the exact two-annulus
identity proves
\begin{equation}\tag{TOB27}
 \|\mathcal B_{d,h}v\|^2
 =p^k\sum_{r=0}^{d-1}
       \bigl(\|A_d^{-h}v_r\|_N^2+
                    \|A_d^{h+1}v_r\|_N^2\bigr).
\end{equation}
In particular this joint map is injective for every $h$: every
summand is nonnegative, and invertibility of $A_d$ implies that
a zero sum forces each $v_r=0$.

For comparison with exactly the same complete blocks, define
\begin{equation}\tag{TOB28}
 L_{d,h}v=\sum_{\ell=-h}^h\sum_{r=0}^{d-1}
                  h_{d\ell+r}\otimes A_d^{-\ell}v_r,
 \qquad
 \|L_{d,h}v\|^2=\sum_{\ell=-h}^h\sum_{r=0}^{d-1}
                              \|A_d^{-\ell}v_r\|_N^2.
\end{equation}
This is the explicitly specified annulus interval
$[-dh,dh+d-1]$, not a redefinition of the symmetric truncation
(TOB19). Its norm formula follows from orthonormality of all the
different annuli in (TOB18).
For any nonzero $v$ with each block in the same original
one-dimensional subspace $\mathbb C\ell_{ab}$, put
$R=|\omega_{ab}|^{\chi_d}=p^{\chi_d(2a-k)\delta}$.
Equations (TOB27)--(TOB28) yield
\begin{equation}\tag{TOB29}
 \frac{\|\mathcal B_{d,h}v\|^2}{\|L_{d,h}v\|^2}
 =p^k\frac{R^{-2h}+R^{2h+2}}{\sum_{\ell=-h}^hR^{-2\ell}}
 \longrightarrow p^k|R^2-1|
 =p^k\left|p^{2\chi_d(2a-k)\delta}-1\right|>0.
\end{equation}
The common factor $\sum_r\|v_r\|_N^2$ cancels from this ratio;
no metric was changed. The geometric-series proof after (TOB21)
applies to $R$. The formula includes every $v_{ab,t}$ of (TOB16).
It is independent of the covering degree within each residue
class modulo four. Therefore the full boundary window, with all
residue classes restored, retains the original nonzero radial
defect exactly, while the one-step eigenvalue radius in (TOB16)
has its explicitly proved $1/d$ exponent. Equations
(TOB10), (TOB26) and (TOB28) are the connecting maps between
these two quantities.
