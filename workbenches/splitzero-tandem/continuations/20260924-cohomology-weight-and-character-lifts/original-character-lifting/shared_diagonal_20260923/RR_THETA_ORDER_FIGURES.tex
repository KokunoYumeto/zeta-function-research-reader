\begin{figure}[ht]
\centering
\includegraphics[width=\textwidth]{shared_diagonal_20260923/RR_SECTIONS_AND_THETA.png}
\caption{RDT10--20, for the displayed original divisor. Left: the exact
degree-one nonzero amplitude count. At $R=1$ its full zero fibre remains
$L$, and at level $r_+$ it remains $L^r$ with join folds. Right: a finite
plot of the theta transform of the \emph{entire} counting measure.
Global strict positivity is proved by the original Gaussian and Poisson
identities in RDT15 and RDT19, not by this plot. The series is evaluated
using those identities and twelve positive-side summands; for $z\ge1$
the omitted absolute terms are bounded by
$2\pi^2\sum_{n\ge13}(nz)^4e^{-\pi(nz)^2}$, and for $z<1$ by
$z^{-1}$ times that bound at $z^{-1}$. Original human sources:
Connes--Consani, \texttt{xdefn}, \texttt{xdefnr}, \texttt{RRZ};
Connes--Consani--Marcolli, \texttt{degmodifyV}.
The accompanying script reproduces both panels without changing $N,a,b$.}
\end{figure}
